37/// Splits the input at each occurrence of the (multi-character) delimiter.
38/// A trailing token is always emitted, so the result contains
39/// (number of delimiters + 1) tokens (e.g. "a," -> {"a",""}, "" -> {""}).
40inline std::vector<std::string> split(const std::string& str, const std::string& delim) {
41 std::vector<std::string> tokens;
42 size_t pos_start = 0, pos_end, delim_len = delim.length();
43 while ((pos_end = str.find(delim, pos_start)) != std::string::npos) {
44 tokens.push_back(str.substr(pos_start, pos_end - pos_start));
45 pos_start = pos_end + delim_len;
46 }
47 tokens.push_back(str.substr(pos_start));
48 return tokens;
49}

59 std::vector<std::string> tokens;

64 pos_start = pos_end + 1;
