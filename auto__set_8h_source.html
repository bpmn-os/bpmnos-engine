13/**
14 * @brief Orders tuples by increasing value of the first component; equal values are kept in insertion order.
15 */
16struct ascending {
17 template <typename Tuple>
18 bool operator()(const Tuple& lhs, const Tuple& rhs) const {
19 // First component only: equal-first-component tuples compare equivalent and coexist in the multiset,
20 // ordered by insertion — a deterministic, allocation-independent tiebreak.
21 return std::get<0>(lhs) < std::get<0>(rhs);
22 }
23};

37/// Satisfied by a type that can order two tuples of the given type (data element types like weak_ptr are not).
38template <typename C, typename Tuple>
39concept TupleComparator = requires (const C& c, const Tuple& a, const Tuple& b) {
40 { c(a, b) } -> std::convertible_to<bool>;
