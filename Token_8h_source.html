179 /**
180 * Returns a merged status from the status of each token
181 **/
182 template <typename TokenPtr>
183 static BPMNOS::Values mergeStatus(const std::vector<TokenPtr>& tokens) {
184 assert( !tokens.empty() );
185 size_t n = tokens.front()->status.size();
186 BPMNOS::Values result;
187 result.resize(n);
188 result[(int)BPMNOS::Model::ExtensionElements::Index::Timestamp] = tokens.front()->status[(int)BPMNOS::Model::ExtensionElements::Index::Timestamp];
189
190 for ( size_t i = 0; i < n; i++ ) {
191 for ( auto& token : tokens ) {
192 if ( i == (int)BPMNOS::Model::ExtensionElements::Index::Timestamp ) {
193 if ( result[i].value() < token->status[i].value() ) {
194 result[i] = token->status[i];
195 }
196 }
197 else if ( !result[i].has_value() ) {
198 result[i] = token->status[i];
199 }
200 else if ( token->status[i].has_value() && token->status[i].value() != result[i].value() ) {
201 result[i] = std::nullopt;
202 break;
203 }
204 }
205 }
206 return result;
207 }
