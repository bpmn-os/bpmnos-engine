19/**
20 * @brief An iterable collection of evaluated candidate decisions, ordered by reward (best first; infeasible last).
21 *
22 * Iterating (`for ( auto c : candidates )`) evaluates the candidates for the current state — captured from the
23 * SystemState notice — and yields them best first. `candidates` holds only weak references (each decision's Event
24 * and Evaluation); `decisions` owns them, so a candidate drops once its Evaluation is reset (the re-evaluation
25 * signal) or its decision dies; `addDecision`/`addCandidate` couple listing with ownership. A specialization
26 * implements evaluateCandidates and decides how many to create — a first-feasible one only the first feasible, a
27 * best-of-all one all of them.
28 *
29 * WeakPtrs... are the candidate's weak identifiers, e.g. < std::weak_ptr<const Token>,
30 * std::weak_ptr<const DecisionRequest> > (and a std::weak_ptr<const Message> for message delivery).
31 */
32template <typename... WeakPtrs>
33class Candidates : public Observer {
34public:
35 /// The evaluator every candidate of this collection is evaluated through, owned here because this is
36 /// where it is read: the collection builds its decisions with it and holds it for as long as it may.
37 Candidates(std::shared_ptr<Evaluator> evaluator) : evaluator(std::move(evaluator)) {}
38 virtual ~Candidates() = default;
39
40 /// Subscriptions are made by the specializations; the base does nothing.
41 virtual void connect(Notifier* /*notifier*/) {}
42
43 /// Binds the per-run system state (specializations override and call this).
44 void notice(const Observable* observable) override {
45 if ( observable->getObservableType() == Observable::Type::SystemState ) {
46 systemState = static_cast<const SystemState*>(observable);
47 }
48 }
49
50 /// Evaluation is lazy — iterating triggers it.
51 auto begin() { evaluateCandidates(); return candidates.begin(); }
52 auto end() { return candidates.end(); }
53
54protected:
55 /// Specialization hook: produce the candidates for the current `systemState`.
56 virtual void evaluateCandidates() = 0;
57
58 /// Take ownership of a decision without listing it as a candidate. Touching `begin()` lazily erases the expired front of `decisions`.
59 void addDecision(WeakPtrs... weak_ptrs, std::shared_ptr<Decision> decision) {
60 decisions.emplace_back(weak_ptrs..., std::move(decision));
61 decisions.begin();
62 }
63
64 /// Take ownership of a decision and list it as a candidate in one step (descending by reward, best first; infeasible (-infinity) last; cannot list a candidate without an owner).
65 void addCandidate(WeakPtrs... weak_ptrs, std::shared_ptr<Decision> decision) {
66 auto reward = decision->reward();
67 candidates.emplace( reward.has_value() ? (double)reward.value() : -std::numeric_limits<double>::infinity(),
68 weak_ptrs..., decision->weak_from_this(), decision->evaluation );
69 addDecision(weak_ptrs..., std::move(decision));
70 }
71
72 /// Drop all owned decisions and the candidates referring to them (per-call reset for stateless sources).
73 void clearDecisions() {
74 candidates.clear();
75 decisions.clear();
76 }
77
78 std::shared_ptr<Evaluator> evaluator; ///< Evaluates every candidate; owned here, so a decision built with it is never built through a dangling one.
79 const SystemState* systemState = nullptr; ///< Per-run system state (bound by notice).
80 auto_set< double, descending, WeakPtrs..., std::weak_ptr<Event>, std::weak_ptr<Evaluation> > candidates; ///< Reward-ordered candidate decisions, descending (best first; infeasible last); holds only weak references.
81 auto_list< WeakPtrs..., std::shared_ptr<Decision> > decisions; ///< Sole owner of the decisions referenced by `candidates`.
82};
