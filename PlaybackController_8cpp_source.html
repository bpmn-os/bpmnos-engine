24PlaybackController::PlaybackController(const nlohmann::ordered_json& log, std::shared_ptr<Evaluator> evaluator)
25 : evaluator(std::move(evaluator))
26{
27 if ( !log.is_array() ) {
28 throw std::runtime_error("PlaybackController: the log must be an array of records");
29 }
30
31 // returns the string a decision states for a field it requires
32 auto field = [](const nlohmann::ordered_json& object, const std::string& name) -> std::string {
33 if ( !object.contains(name) || !object.at(name).is_string() ) {
34 throw std::runtime_error("PlaybackController: decision without field '" + name + "'");
35 }
36 return object.at(name).get<std::string>();
37 };
38
39 // decisions before the first clock tick belong to the time the run starts
40 BPMNOS::number time = std::numeric_limits<BPMNOS::number>::lowest();
41 for ( auto& record : log ) {
42 if ( !record.is_object() || !record.contains("event") ) {
43 // neither a clock tick nor a decision
44 continue;
45 }
46 auto& event = record.at("event");
47
48 if ( event.contains("event") && event.at("event") == "clocktick" ) {
49 if ( !event.contains("time") || !event.at("time").is_number() ) {
50 throw std::runtime_error("PlaybackController: clock tick without time");
51 }
52 time = BPMNOS::number(event.at("time").get<double>());
53 continue;
54 }
55
56 if ( !event.contains("decision") ) {
57 // an event the engine carries out itself
58 continue;
59 }
60
61 LoggedDecision decision;
62 decision.time = time;
63 auto kind = field(event, "decision");
64 if ( kind == "entry" ) {
65 decision.kind = LoggedDecision::Kind::Entry;
66 }
67 else if ( kind == "choice" ) {
68 decision.kind = LoggedDecision::Kind::Choice;
69 if ( !event.contains("choices") || !event.at("choices").is_object() ) {
70 throw std::runtime_error("PlaybackController: choice without field 'choices'");
71 }
72 decision.choices = event.at("choices");
73 }
74 else if ( kind == "messagedelivery" ) {
75 decision.kind = LoggedDecision::Kind::MessageDelivery;
76 if ( !event.contains("message") || !event.at("message").is_object() ) {
77 throw std::runtime_error("PlaybackController: message delivery without field 'message'");
78 }
79 decision.message = event.at("message");
80 decision.origin = field(decision.message, "origin");
81 if ( decision.message.contains("header") && decision.message.at("header").contains("sender") && decision.message.at("header").at("sender").is_string() ) {
82 decision.sender = decision.message.at("header").at("sender").get<std::string>();
83 }
84 }
85 else {
86 throw std::runtime_error("PlaybackController: decision of unknown kind '" + kind + "'");
87 }
88 decision.processId = field(event, "processId");
89 decision.instanceId = field(event, "instanceId");
90 decision.nodeId = field(event, "nodeId");
91
92 decisions[time].push_back(std::move(decision));
93 }
94
95 // the decisions are ordered by their time, and within it by the log
96 for ( auto& [_, decisionsAtTime] : decisions ) {
97 for ( auto& decision : decisionsAtTime ) {
98 sequence.push_back(&decision);
99 }
100 }
101 dispatched.assign(sequence.size(), false);
102 due.assign(sequence.size(), std::nullopt);
103}

194std::string PlaybackController::performerKey(const SystemState* systemState, const LoggedDecision& decision, const BPMN::Node* node) {
195 auto performer = node->as<BPMN::FlowNode>()->parent->as<BPMNOS::Model::SequentialAdHocSubProcess>()->performer;
196
197 // returns true if the scope lies within the performer
198 auto isWithinPerformer = [performer](const BPMN::Node* scope) {
199 for ( auto flowNode = scope->represents<BPMN::FlowNode>(); flowNode; flowNode = flowNode->parent->represents<BPMN::FlowNode>() ) {
200 if ( flowNode->parent == performer ) {
201 return true;
202 }
203 }
204 return false;
205 };
206
207 // the instance of the decision is the instance of the performer, followed by the disambiguations of the event
208 // subprocesses and multi-instance activities within the performer, which are removed
209 std::string instance = decision.instanceId;
210 for ( auto position = instance.rfind(StateMachine::delimiters[0]); position != std::string::npos; position = instance.rfind(StateMachine::delimiters[0]) ) {
211 auto disambiguation = instance.substr(position + 1);
212 auto scope = findNode(systemState, disambiguation.substr(0, disambiguation.rfind(StateMachine::delimiters[1])));
213 if ( !scope || !isWithinPerformer(scope) ) {
214 break;
215 }
216 instance = instance.substr(0, position);
217 }
218 return instance + "/" + performer->id;
219}

227std::shared_ptr<Decision> PlaybackController::createDecision(const SystemState* systemState, const LoggedDecision& decision) {
228 // returns true if the token is the deciding token the log states
229 auto isDeciding = [&decision](const Token* token) {
230 return
231 token->node->id == decision.nodeId &&
232 token->owner->root->scope->id == decision.processId &&
233 BPMNOS::to_string(token->getInstanceId(),STRING) == decision.instanceId
234 ;
235 };
236
237 if ( decision.kind == LoggedDecision::Kind::Entry ) {
238 for ( auto& [token_ptr, request_ptr] : systemState->pendingEntryDecisions ) {
239 auto token = token_ptr.lock();
240 auto request = request_ptr.lock();
241 if ( token && request && isDeciding(token.get()) ) {
242 return std::make_shared<EntryDecision>(request.get(), evaluator.get());
243 }
244 }
245 }
246 else if ( decision.kind == LoggedDecision::Kind::Choice ) {
247 for ( auto& [token_ptr, request_ptr] : systemState->pendingChoiceDecisions ) {
248 auto token = token_ptr.lock();
249 auto request = request_ptr.lock();
250 if ( token && request && isDeciding(token.get()) ) {
251 // the values are given by attribute name and are taken in the order of the choices of the decision task
252 auto extensionElements = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
253 std::vector<BPMNOS::number> values;
254 for ( auto& choice : extensionElements->choices ) {
255 auto& name = choice->attribute->name;
256 if ( !decision.choices.contains(name) ) {
257 return nullptr;
258 }
259 auto& value = decision.choices.at(name);
260 if ( value.is_string() ) {
261 values.push_back( BPMNOS::to_number(value.get<std::string>(), STRING) );
262 }
263 else if ( value.is_boolean() ) {
264 values.push_back( BPMNOS::number( value.get<bool>() ? 1 : 0 ) );
265 }
266 else {
267 values.push_back( BPMNOS::number( value.get<double>() ) );
268 }
269 }
270 return std::make_shared<ChoiceDecision>(request.get(), values, evaluator.get());
271 }
272 }
273 }
274 else {
275 for ( auto& [token_ptr, request_ptr] : systemState->pendingMessageDeliveryDecisions ) {
276 auto token = token_ptr.lock();
277 auto request = request_ptr.lock();
278 if ( !token || !request || !isDeciding(token.get()) ) {
279 continue;
280 }
281 auto& senderCandidates = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->messageCandidates;
282 for ( auto& message : systemState->messages ) {
283 if (
284 message->state != Message::State::CREATED ||
285 message->origin->id != decision.origin ||
286 !std::ranges::contains(senderCandidates, message->origin) ||
287 !message->matches(request->recipientHeader)
288 ) {
289 continue;
290 }
291 // the message agrees with the header and the content the log states
292 auto messageObject = message->jsonify();
293 bool agrees = true;
294 if ( decision.message.contains("header") ) {
295 auto& header = decision.message.at("header");
296 for ( auto it = header.begin(); it != header.end(); ++it ) {
297 if ( !it.value().is_null() && ( !messageObject["header"].contains(it.key()) || messageObject["header"][it.key()] != it.value() ) ) {
298 agrees = false;
299 }
300 }
301 }
302 if ( decision.message.contains("content") && messageObject["content"] != decision.message.at("content") ) {
303 agrees = false;
304 }
305 if ( agrees ) {
306 return std::make_shared<MessageDeliveryDecision>(request.get(), message.get(), evaluator.get());
307 }
308 }
309 }
310 }
311 return nullptr;
312}

314std::shared_ptr<Event> PlaybackController::dispatchEvent(const SystemState* systemState) {
315 if ( mismatch ) {
316 return nullptr;
317 }
318 auto now = systemState->getTime();
319
320 // a decision becomes due at the first time reached that is not earlier than its own time
321 for ( size_t i = 0; i < sequence.size() && sequence[i]->time <= now; i++ ) {
322 if ( !dispatched[i] && !due[i].has_value() ) {
323 due[i] = now;
324 }
325 }
326
327 // performers whose next entry is listed before the entry considered
328 std::unordered_set<std::string> busyPerformers;
329 for ( size_t i = 0; i < sequence.size() && due[i].has_value(); i++ ) {
330 if ( dispatched[i] ) {
331 continue;
332 }
333 auto& decision = *sequence[i];
334
335 if ( decision.kind == LoggedDecision::Kind::Entry ) {
336 auto node = findNode(systemState, decision.nodeId);
337 if ( !node || !BPMNOS::Model::Model::isSequentialActivity(node) ) {
338 return stop(&decision, now, "entry of an activity whose entry is not decided");
339 }
340 // the entries of a performer keep the order of the log
341 if ( !busyPerformers.insert(performerKey(systemState, decision, node)).second ) {
342 continue;
343 }
344 }
345
346 auto event = createDecision(systemState, decision);
347 if ( !event ) {
348 if ( decision.kind == LoggedDecision::Kind::Choice ) {
349 // a choice whose request is pending but which lacks the value of a choice attribute is never dispatched
350 auto node = findNode(systemState, decision.nodeId);
351 if ( node ) {
352 for ( auto& choice : node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->choices ) {
353 if ( !decision.choices.contains(choice->attribute->name) ) {
354 return stop(&decision, now, "choice without value of attribute '" + choice->attribute->name + "'");
355 }
356 }
357 }
358 }
359 continue;
360 }
361
362 event->evaluate();
363 if ( !event->reward().has_value() ) {
364 // a decision that is infeasible now waits until it becomes feasible, as any controller has to
365 continue;
366 }
367
368 dispatched[i] = true;
369 dispatchedCount++;
370 if ( decision.time != std::numeric_limits<BPMNOS::number>::lowest() && now > decision.time ) {
371 // the run does not reach the time of the decision
372 deviations.push_back( Deviation{ &decision, now } );
373 }
374 return event;
375 }
376
377 return nullptr;
