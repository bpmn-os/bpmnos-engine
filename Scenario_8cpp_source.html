23BPMNOS::Values Scenario::evaluateGlobals(const std::unordered_map<const Attribute*, BPMNOS::number>& globalValueMap) {
24 Values result(model->attributes.size());
25
26 // Set CSV-provided globals
27 for (auto& [attribute, value] : globalValueMap) {
