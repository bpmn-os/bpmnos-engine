36std::shared_ptr<Decision> FirstBisectionalChoice::determineBestChoices(std::shared_ptr<const DecisionRequest> request) {
37 this->clearDecisions();
38
39 auto token = request->token;
40 assert( token->node );
41 assert( token->node->represents<BPMNOS::Model::DecisionTask>() );
42 assert( token->node->extensionElements->represents<BPMNOS::Model::ExtensionElements>() );
43 auto extensionElements = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
44 assert( extensionElements->choices.size() );
45
46 token_ptr = token->weak_from_this();
47 request_ptr = request->weak_from_this();
48
49 if ( extensionElements->choices.size() > 1 ) {
50 return bestEnumeratedChoice(request);
51 }
52
53 auto& choice = extensionElements->choices[0];
54
55 if ( !choice->enumeration.empty() ) {
56 return bestEnumeratedChoice(request);

61 }
62

72std::shared_ptr<Decision> FirstBisectionalChoice::bestEnumeratedChoice(std::shared_ptr<const DecisionRequest> request) {
73 auto token = request->token;
74 auto decisionTask = token->node->as<BPMNOS::Model::DecisionTask>();
75
76 auto alternativeChoices = decisionTask->enumerateAlternatives(token->status, *token->data, token->globals);
77 std::shared_ptr<Decision> bestDecision = nullptr;
78 for ( auto& choices : alternativeChoices ) {
79 auto decision = std::make_shared<ChoiceDecision>(request.get(), std::move(choices), evaluator);
80 decision->evaluate();
81 auto reward = decision->reward();
82 if ( reward.has_value() && ( !bestDecision || reward.value() > bestDecision->reward().value() ) ) {
83 bestDecision = decision;
84 }
85 // add every alternative to the reward-ordered candidates (infeasible last), taking ownership
86 this->addCandidate( token_ptr, request_ptr, std::move(decision) );
87 }
88 return bestDecision;

101std::tuple<size_t, FirstBisectionalChoice::Candidate, size_t> FirstBisectionalChoice::findFeasible(size_t first, size_t last) {
102 const size_t npos = std::numeric_limits<size_t>::max();
103 if ( first > last ) return { npos, { npos, nullptr }, npos };
104
105 // BFS on intervals to find any feasible
106 std::queue<std::pair<size_t, size_t>> intervals;
107 intervals.push({first, last});
108
109 while ( !intervals.empty() ) {
110 auto [l, r] = intervals.front();
111 intervals.pop();
112
113 size_t mid = l + (r - l) / 2;
114 auto candidate = evaluate(mid);
115 if ( candidate.isFeasible() ) {
116 // Found feasible - return with nearest known infeasible bounds
117 size_t nearestLeft = (mid > first) ? l : npos;
118 size_t nearestRight = (mid < last) ? r : npos;
119 return { nearestLeft, candidate, nearestRight };
120 }
121
122 // mid is infeasible - continue searching both halves
123 if ( mid > l ) intervals.push({l, mid - 1});
124 if ( mid < r ) intervals.push({mid + 1, r});
125 }
126
127 return { npos, { npos, nullptr }, npos };
128}

130void FirstBisectionalChoice::findBetweenFeasibleAndFeasible(Candidate left, Candidate right) {
131 if ( right.index <= left.index + 1 ) return;
132
133 size_t mid = left.index + (right.index - left.index) / 2;
134 auto candidate = evaluate(mid);
135
136 if ( !candidate.isFeasible() ) {
137 // Mid is infeasible - search both sides
138 findBetweenFeasibleAndInfeasible(left, mid);
139 findBetweenInfeasibleAndFeasible(mid, right);
140 }
141 else {
142 // Mid is feasible
143 if ( candidate.reward() > best.reward() ) {
144 best = candidate;
145 }
146 // Check if mid is worse than both boundaries -> peak is at a boundary
147 if ( candidate.reward() <= left.reward() && candidate.reward() <= right.reward() ) {
148 return; // Peak is at one of the boundaries, already checked
149 }
150 // Continue toward better side
151 if ( left.reward() >= right.reward() ) {
152 findBetweenFeasibleAndFeasible(left, candidate);
153 }
154 else {
155 findBetweenFeasibleAndFeasible(candidate, right);
156 }
157 }

160void FirstBisectionalChoice::findBetweenFeasibleAndInfeasible(Candidate feasible, size_t infeasibleIndex) {
161 while ( infeasibleIndex > feasible.index + 1 ) {
162 size_t mid = feasible.index + (infeasibleIndex - feasible.index + 1) / 2;
163 auto candidate = evaluate(mid);
164
165 if ( !candidate.isFeasible() ) {
166 // Still infeasible - narrow right boundary
167 infeasibleIndex = mid;
168 }
169 else if ( candidate.reward() <= best.reward() ) {
170 // Feasible but worse - done (past peak)
171 break;
172 }
173 else {
174 // Feasible and better - new best, continue toward infeasible

176 feasible = candidate;

202std::shared_ptr<Decision> FirstBisectionalChoice::discreteBisection(std::shared_ptr<const DecisionRequest> request, const BPMNOS::Model::Choice* choice) {
203 const size_t npos = std::numeric_limits<size_t>::max();
204
205 // Set member state for this search
206 token = request->token;
207 values = choice->getEnumeration(token->status, *token->data, token->globals);
208 best = { npos, nullptr };
209
210 if ( values.empty() ) {
211 return nullptr;
212 }
213
214 // Check both boundaries
215 auto left = evaluate(0);
216 auto right = (values.size() > 1) ? evaluate(values.size() - 1) : Candidate{ npos, nullptr };
217
218
219 if ( left.isFeasible() ) {
220 best = left;
221 }
222 if ( right.isFeasible() && (!best.isFeasible() || right.reward() > best.reward()) ) {
223 best = right;
224 }
225
226 if ( left.isFeasible() && right.isFeasible() ) {
227 // Both feasible - search between them
228 findBetweenFeasibleAndFeasible(left, right);
229 }
230 else if ( left.isFeasible() && !right.isFeasible() ) {
231 // Left feasible, right infeasible - search toward right
232 findBetweenFeasibleAndInfeasible(left, values.size() - 1);
233 }
234 else if ( !left.isFeasible() && right.isFeasible() ) {
235 // Left infeasible, right feasible - search toward left
236 findBetweenInfeasibleAndFeasible(0, right);
237 }
238 else {
239 // Both infeasible - find any feasible first
240 if ( values.size() <= 2 ) {
241 return nullptr;
242 }
243 auto [leftInfeasible, candidate, rightInfeasible] = findFeasible(1, values.size() - 2);
244 if ( !candidate.isFeasible() ) {
245 return nullptr;
246 }
247 best = candidate;
248
249 // Search both directions from feasible point
250 if ( leftInfeasible != npos ) {
251 findBetweenInfeasibleAndFeasible(leftInfeasible, best);
252 }
253 if ( rightInfeasible != npos ) {
254 findBetweenFeasibleAndInfeasible(best, rightInfeasible);
255 }
256 }
257
258 return best.decision;
259}
