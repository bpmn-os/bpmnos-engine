47void DataUpdateEvent::validate(const BPMNOS::Model::Attribute* attribute, bool isObject) const {
48 if ( attribute->category != BPMNOS::Model::Attribute::Category::DATA ) {
49 throw std::invalid_argument("DataUpdateEvent: '" + attribute->id + "' is no data or global attribute");
50 }
51 if ( attribute->isObject() != isObject ) {
52 throw std::invalid_argument("DataUpdateEvent: '" + attribute->id + "' is " + ( isObject ? "no object" : "an object" ));
53 }
54 // the time is changed only by a clock tick, and the instance by nothing
55 if ( !attribute->isObject() && attribute->isGlobal && attribute->index == BPMNOS::Model::ExtensionElements::Index::Time ) {
56 throw std::invalid_argument("DataUpdateEvent: the time must not be updated");
57 }
58 if ( attribute->id == BPMNOS::Keyword::Instance ) {
59 throw std::invalid_argument("DataUpdateEvent: the instance must not be updated");
60 }
61 if ( attribute->isImmutable ) {
62 throw std::invalid_argument("DataUpdateEvent: '" + attribute->id + "' is immutable");
63 }
64 if ( !scope ) {
65 if ( !attribute->isGlobal ) {
66 throw std::invalid_argument("DataUpdateEvent: '" + attribute->id + "' is no global attribute");
67 }
68 return;
69 }
70 // the attributes a scope declares are the data of the data objects within it
71 auto extensionElements = scope->extensionElements ? scope->extensionElements->as<BPMNOS::Model::ExtensionElements>() : nullptr;
72 auto declares = [attribute](const std::vector< std::unique_ptr<BPMNOS::Model::Attribute> >& declared) {
73 return std::ranges::any_of(declared, [attribute](const std::unique_ptr<BPMNOS::Model::Attribute>& candidate) { return candidate.get() == attribute; });
74 };
75 if ( !extensionElements || !( declares(extensionElements->data) || declares(extensionElements->dataObjects) ) ) {
76 throw std::invalid_argument("DataUpdateEvent: '" + attribute->id + "' is no data attribute of '" + scope->id + "'");
77 }

84StateMachine* DataUpdateEvent::owner() const {
85 if ( !scope ) {
86 return systemState->stateMachine.get();

88 // the archive holds the state machines of the process instances, of the instances of multi-instance activities and
89 // of the instances of non-interrupting event subprocesses, each by its identifier; any other state machine shares
90 // the identifier of one of them and is found below it
91 auto it = systemState->archive.find( (long unsigned int)instance );
92 if ( it == systemState->archive.end() ) {
93 return nullptr;
94 }
95 auto archived = it->second.lock();
96 if ( !archived ) {
97 return nullptr;
98 }
99 return find(archived.get());

102StateMachine* DataUpdateEvent::find(StateMachine* stateMachine) const {
103 if ( stateMachine->scope == scope ) {
104 return stateMachine;
105 }
106 // a subprocess and an interrupting event subprocess share the identifier of the state machine they belong to, whereas
107 // every state machine with an identifier of its own is archived and reached from the archive
108 for ( auto& token : stateMachine->tokens ) {
109 if ( token->owned && token->owned->instance == instance ) {
110 if ( auto found = find(token->owned.get()) ) {
111 return found;
112 }
113 }
114 }
115 if ( auto& interrupting = stateMachine->interruptingEventSubProcess; interrupting && interrupting->instance == instance ) {
116 return find(interrupting.get());
117 }
118 return nullptr;
119}
