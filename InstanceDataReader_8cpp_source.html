11InstanceDataReader::InstanceDataReader(const BPMNOS::Model::Model* model, const std::string& instanceFileOrString, const std::vector<std::string>& columns)
12 : model(model)
13{
14 enum { INSTANCE, NODE, INITIALIZATION, OTHER };
15 if ( columns.size() < OTHER ) {
16 throw std::logic_error("InstanceDataReader: at least three columns are required");
17 }
18
19 BPMNOS::CSVReader reader(instanceFileOrString, ";");
20 auto table = reader.read();
21 if ( table.empty() ) {
22 throw std::runtime_error("InstanceDataReader: table '" + instanceFileOrString + "' is empty");
23 }
24
25 // the table names the first three or more of the columns given
26 auto& header = table.front();
27 if ( header.size() < OTHER || header.size() > columns.size() ) {
28 throw std::runtime_error("InstanceDataReader: expected between " + std::to_string((size_t)OTHER) + " and " + std::to_string(columns.size()) + " columns, got " + std::to_string(header.size()));
29 }
30 for ( size_t i = 0; i < header.size(); i++ ) {
31 if ( !std::holds_alternative<std::string>(header[i]) || std::get<std::string>(header[i]) != columns[i] ) {
32 throw std::runtime_error("InstanceDataReader: column " + std::to_string(i + 1) + " must be named '" + columns[i] + "'");
33 }
34 }
35
36 for ( auto& cells : table | std::views::drop(1) ) {
37 if ( cells.empty() ) {
38 continue;
39 }
40 if ( cells.size() != header.size() ) {
41 throw std::runtime_error("InstanceDataReader: inconsistent number of cells");
42 }
43 for ( size_t i = 0; i < OTHER; i++ ) {
44 if ( !std::holds_alternative<std::string>(cells[i]) ) {
45 throw std::runtime_error("InstanceDataReader: illegal cell in column '" + columns[i] + "'");
46 }
47 }
48 auto& instanceIdentifier = std::get<std::string>(cells[INSTANCE]);
49 auto& nodeId = std::get<std::string>(cells[NODE]);
50
51 Row row{0, nullptr, std::get<std::string>(cells[INITIALIZATION]), std::vector<std::string>(columns.size() - OTHER)};
52 for ( size_t i = OTHER; i < cells.size(); i++ ) {
53 row.cells[i - OTHER] = std::holds_alternative<std::string>(cells[i]) ?
54 std::get<std::string>(cells[i]) :
55 std::to_string((double)std::get<BPMNOS::number>(cells[i]));
56 }
57
58 if ( instanceIdentifier.empty() && nodeId.empty() ) {
59 // the value of a global attribute
60 for ( size_t i = OTHER; i < cells.size(); i++ ) {
61 if ( !row.cells[i - OTHER].empty() ) {
62 throw std::runtime_error("InstanceDataReader: a global attribute must not have a value in column '" + columns[i] + "'");
63 }
64 }
65 if ( !row.initialization.empty() ) {
66 rows.push_back(std::move(row));
67 }
68 continue;
69 }
70 if ( instanceIdentifier.empty() ) {
71 throw std::runtime_error("InstanceDataReader: instance required for node '" + nodeId + "'");
72 }
73
74 row.instanceId = (size_t)BPMNOS::to_number(instanceIdentifier, STRING);
75 row.node = findNode(nodeId);
76 if ( !processes.contains(row.instanceId) ) {
77 // the first row of an instance names its process
78 auto process = row.node->represents<BPMN::Process>();
79 if ( !process ) {
80 throw std::runtime_error("InstanceDataReader: the first row of instance '" + instanceIdentifier + "' must name a process, not '" + nodeId + "'");
81 }
82 processes[row.instanceId] = process;
83 }
84 rows.push_back(std::move(row));
85 }

88const BPMN::Node* InstanceDataReader::findNode(const std::string& nodeId) const {
89 for ( auto& process : model->processes ) {

100std::pair<std::string, std::string> InstanceDataReader::splitInitialization(const std::string& initialization) {
101 auto position = initialization.find(":=");
102 if ( position == std::string::npos ) {
103 throw std::runtime_error("InstanceDataReader: initialization must have the form 'attribute := expression', got '" + initialization + "'");
104 }
105 auto trim = [&initialization](std::string text) {
106 auto start = text.find_first_not_of(" \t");
107 if ( start == std::string::npos ) {
108 throw std::runtime_error("InstanceDataReader: incomplete initialization '" + initialization + "'");
109 }
110 auto end = text.find_last_not_of(" \t");
111 return text.substr(start, end - start + 1);
112 };
113 return { trim(initialization.substr(0, position)), trim(initialization.substr(position + 2)) };
114}

116std::pair<const BPMNOS::Model::Attribute*, std::string> InstanceDataReader::lookupAttribute(const BPMN::Node* node, const std::string& initialization) const {
117 // the engine creates event subprocesses and compensation activities itself, with the values the model
118 // assigns to their attributes, so the data may give them none
119 auto activity = node->represents<BPMN::Activity>();
120 if ( node->represents<BPMN::EventSubProcess>() || ( activity && activity->isForCompensation ) ) {
121 throw std::runtime_error("InstanceDataReader: no values may be given for event subprocess or compensation activity '" + node->id + "'");
122 }
123
124 auto [attributeName, expression] = splitInitialization(initialization);
125 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
126 if ( !extensionElements->attributeRegistry.contains(attributeName) ) {
127 // the attributes a guidance declares take only the values the model assigns to them
128 for ( auto guidance : { &extensionElements->entryGuidance, &extensionElements->exitGuidance, &extensionElements->choiceGuidance, &extensionElements->messageDeliveryGuidance } ) {
129 if ( guidance->has_value() && guidance->value()->attributeRegistry.contains(attributeName) ) {
130 throw std::runtime_error("InstanceDataReader: attribute '" + attributeName + "' of node '" + node->id + "' belongs to a guidance and takes no value from the data");
131 }
132 }
133 throw std::runtime_error("InstanceDataReader: node '" + node->id + "' has no attribute '" + attributeName + "'");
134 }
135
136 auto attribute = extensionElements->attributeRegistry[attributeName];
137 if ( attribute->expression ) {
138 throw std::runtime_error("InstanceDataReader: attribute '" + attributeName + "' is assigned by the model and must not be given");
139 }
140 return {attribute, expression};
141}

143void InstanceDataReader::evaluateGlobal(const std::string& initialization, const LIMEX::Handle<double>& handle) {
144 auto [attributeName, expressionString] = splitInitialization(initialization);
145
146 const BPMNOS::Model::Attribute* attribute = nullptr;
147 for ( auto& globalAttribute : model->attributes ) {
148 if ( globalAttribute->name == attributeName ) {
149 attribute = globalAttribute.get();
150 break;
151 }
152 }
153 if ( !attribute ) {
154 throw std::runtime_error("InstanceDataReader: unknown global attribute '" + attributeName + "'");
155 }
156 if ( attribute->expression ) {
157 throw std::runtime_error("InstanceDataReader: global attribute '" + attributeName + "' is assigned by the model and must not be given");
158 }
159
160 BPMNOS::Model::Expression expression(handle, BPMNOS::InputEncoder::fragment(expressionString), model->attributeRegistry);
161 // the global attributes are the first data attributes
162 BPMNOS::Values globalValues(model->attributes.size());
163 for ( auto& [globalAttribute, value] : globals ) {
164 globalValues[globalAttribute->index] = value;
165 }
166 for ( auto referencedAttribute : expression.variables ) {
167 if ( !globals.contains(referencedAttribute) ) {
168 throw std::runtime_error("InstanceDataReader: global attribute '" + attributeName + "' refers to global attribute '" + referencedAttribute->name + "' without a value");
169 }
170 }
171
172 auto value = expression.execute(BPMNOS::Values{}, globalValues);
173 if ( !value.has_value() ) {
174 throw std::runtime_error("InstanceDataReader: failed to evaluate global attribute '" + attributeName + "'");
175 }
176 globals[attribute] = convert(value.value(), attribute->type);
177}

179BPMNOS::number InstanceDataReader::evaluate(size_t instanceId, const BPMN::Node* node, const std::string& expressionString, ValueType type, const LIMEX::Handle<double>& handle) const {
180 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
181 BPMNOS::Model::Expression expression(handle, BPMNOS::InputEncoder::fragment(expressionString), extensionElements->attributeRegistry);
182
183 BPMNOS::Values status(extensionElements->attributeRegistry.statusAttributes.size());
184 BPMNOS::Values data(extensionElements->attributeRegistry.dataAttributes.size());
185 // the global attributes are the first data attributes
186 for ( auto& [attribute, value] : globals ) {
187 data[attribute->index] = value;
188 }
189 auto instanceValues = values.find(instanceId);
190 if ( instanceValues != values.end() ) {
191 for ( auto& [attribute, value] : instanceValues->second ) {
192 if ( attribute->category == BPMNOS::Model::Attribute::Category::STATUS ) {
193 status[attribute->index] = value;
194 }
195 else if ( attribute->category == BPMNOS::Model::Attribute::Category::DATA ) {
196 data[attribute->index] = value;
197 }
198 }
199 }
200
201 for ( auto attribute : expression.variables ) {
202 bool known = ( attribute->category == BPMNOS::Model::Attribute::Category::DATA && attribute->index < model->instanceIndex ) ?
203 globals.contains(attribute) :
204 ( instanceValues != values.end() && instanceValues->second.contains(attribute) );
205 if ( !known ) {
206 throw std::runtime_error("InstanceDataReader: expression '" + expressionString + "' refers to attribute '" + attribute->name + "' without a value");
207 }
208 }
209
210 auto value = expression.execute(status, data);
211 if ( !value.has_value() ) {
212 throw std::runtime_error("InstanceDataReader: failed to evaluate expression '" + expressionString + "'");
213 }
214 return convert(value.value(), type);
215}
