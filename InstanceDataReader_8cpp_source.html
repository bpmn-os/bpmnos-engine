12InstanceDataReader::InstanceDataReader(const BPMNOS::Model::Model* model, const std::string& instanceFileOrString, const std::vector<std::string>& columns)
13 : model(model)
14{
15 enum { INSTANCE, NODE, INITIALIZATION, OTHER };
16 if ( columns.size() < OTHER ) {
17 throw std::logic_error("InstanceDataReader: at least three columns are required");
18 }
19
20 BPMNOS::CSVReader reader(instanceFileOrString, ";");
21 auto table = reader.read();
22 if ( table.empty() ) {
23 throw std::runtime_error("InstanceDataReader: table '" + instanceFileOrString + "' is empty");
24 }
25
26 // the table names the first three or more of the columns given
27 auto& header = table.front();
28 if ( header.size() < OTHER || header.size() > columns.size() ) {
29 throw std::runtime_error("InstanceDataReader: expected between " + std::to_string((size_t)OTHER) + " and " + std::to_string(columns.size()) + " columns, got " + std::to_string(header.size()));
30 }
31 for ( size_t i = 0; i < header.size(); i++ ) {
32 if ( !std::holds_alternative<std::string>(header[i]) || std::get<std::string>(header[i]) != columns[i] ) {
33 throw std::runtime_error("InstanceDataReader: column " + std::to_string(i + 1) + " must be named '" + columns[i] + "'");
34 }
35 }
36
37 for ( auto& cells : table | std::views::drop(1) ) {
38 if ( cells.empty() ) {
39 continue;
40 }
41 if ( cells.size() != header.size() ) {
42 throw std::runtime_error("InstanceDataReader: inconsistent number of cells");
43 }
44 for ( size_t i = 0; i < OTHER; i++ ) {
45 if ( !std::holds_alternative<std::string>(cells[i]) ) {
46 throw std::runtime_error("InstanceDataReader: illegal cell in column '" + columns[i] + "'");
47 }
48 }
49 auto& instanceIdentifier = std::get<std::string>(cells[INSTANCE]);
50 auto& nodeId = std::get<std::string>(cells[NODE]);
51
52 Row row{0, nullptr, std::get<std::string>(cells[INITIALIZATION]), std::vector<std::string>(columns.size() - OTHER)};
53 for ( size_t i = OTHER; i < cells.size(); i++ ) {
54 row.cells[i - OTHER] = std::holds_alternative<std::string>(cells[i]) ?
55 std::get<std::string>(cells[i]) :
56 std::to_string((double)std::get<BPMNOS::number>(cells[i]));
57 }
58
59 if ( !row.initialization.empty() && !row.initialization.contains(":=") ) {
60 // a size declaration, which holds for every instance
61 if ( !instanceIdentifier.empty() ) {
62 throw std::runtime_error("InstanceDataReader: size declaration '" + row.initialization + "' must not name an instance");
63 }
64 for ( size_t i = OTHER; i < cells.size(); i++ ) {
65 if ( !row.cells[i - OTHER].empty() ) {
66 throw std::runtime_error("InstanceDataReader: size declaration '" + row.initialization + "' must not have a value in column '" + columns[i] + "'");
67 }
68 }
69 row.declaresSizes = true;
70 row.node = nodeId.empty() ? nullptr : findNode(nodeId);
71 rows.push_back(std::move(row));
72 continue;
73 }
74 if ( instanceIdentifier.empty() && nodeId.empty() ) {
75 // the value of a global attribute
76 for ( size_t i = OTHER; i < cells.size(); i++ ) {
77 if ( !row.cells[i - OTHER].empty() ) {
78 throw std::runtime_error("InstanceDataReader: a global attribute must not have a value in column '" + columns[i] + "'");
79 }
80 }
81 if ( !row.initialization.empty() ) {
82 rows.push_back(std::move(row));
83 }
84 continue;
85 }
86 if ( instanceIdentifier.empty() ) {
87 throw std::runtime_error("InstanceDataReader: instance required for node '" + nodeId + "'");
88 }
89
90 row.instanceId = (size_t)BPMNOS::to_number(instanceIdentifier, STRING);
91 row.node = findNode(nodeId);
92 if ( !processes.contains(row.instanceId) ) {
93 // the first row of an instance names its process
94 auto process = row.node->represents<BPMN::Process>();
95 if ( !process ) {
96 throw std::runtime_error("InstanceDataReader: the first row of instance '" + instanceIdentifier + "' must name a process, not '" + nodeId + "'");
97 }
98 processes[row.instanceId] = process;
99 }
100 rows.push_back(std::move(row));
101 }

116std::pair<std::string, std::string> InstanceDataReader::splitInitialization(const std::string& initialization) {
117 auto position = initialization.find(":=");
118 if ( position == std::string::npos ) {
119 throw std::runtime_error("InstanceDataReader: initialization must have the form 'attribute := expression', got '" + initialization + "'");
120 }
121 auto trim = [&initialization](std::string text) {
122 auto start = text.find_first_not_of(" \t");
123 if ( start == std::string::npos ) {
124 throw std::runtime_error("InstanceDataReader: incomplete initialization '" + initialization + "'");
125 }
126 auto end = text.find_last_not_of(" \t");
127 return text.substr(start, end - start + 1);
128 };
129 return { trim(initialization.substr(0, position)), trim(initialization.substr(position + 2)) };
130}

132std::pair<const BPMNOS::Model::Attribute*, std::string> InstanceDataReader::lookupAttribute(const BPMN::Node* node, const std::string& initialization) const {
133 // the engine creates event subprocesses and compensation activities itself, with the values the model
134 // assigns to their attributes, so the data may give them none
135 auto activity = node->represents<BPMN::Activity>();
136 if ( node->represents<BPMN::EventSubProcess>() || ( activity && activity->isForCompensation ) ) {
137 throw std::runtime_error("InstanceDataReader: no values may be given for event subprocess or compensation activity '" + node->id + "'");
138 }
139
140 auto [attributeName, expression] = splitInitialization(initialization);
141 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
142 if ( !extensionElements->attributeRegistry.contains(attributeName) ) {
143 // the attributes a guidance declares take only the values the model assigns to them
144 for ( auto guidance : { &extensionElements->entryGuidance, &extensionElements->exitGuidance, &extensionElements->choiceGuidance, &extensionElements->messageDeliveryGuidance } ) {
145 if ( guidance->has_value() && guidance->value()->attributeRegistry.contains(attributeName) ) {
146 throw std::runtime_error("InstanceDataReader: attribute '" + attributeName + "' of node '" + node->id + "' belongs to a guidance and takes no value from the data");
147 }
148 }
149 throw std::runtime_error("InstanceDataReader: node '" + node->id + "' has no attribute '" + attributeName + "'");
150 }
151
152 auto attribute = extensionElements->attributeRegistry[attributeName];
153 if ( attribute->expression || attribute->initialObject.has_value() ) {
154 throw std::runtime_error("InstanceDataReader: attribute '" + attributeName + "' is assigned by the model and must not be given");
155 }
156 return {attribute, expression};
157}

196void InstanceDataReader::setObject(const Row& row) {
197 for ( auto& cell : row.cells ) {
198 if ( !cell.empty() ) {
199 throw std::runtime_error("InstanceDataReader: the value of an object is known from the start and takes no further value, in '" + row.initialization + "'");
200 }
201 }

203 // the literal has been registered as a constant object when the line was read, and stands as its index

206 index = std::stoul(literal);

216 }

219 }

274void InstanceDataReader::evaluateGlobal(const std::string& initialization, const LIMEX::Handle<double>& handle) {
275 auto [attributeName, expressionString] = splitInitialization(initialization);
276
277 const BPMNOS::Model::Attribute* attribute = nullptr;
278 for ( auto& globalAttribute : model->attributes ) {
279 if ( globalAttribute->name == attributeName ) {
280 attribute = globalAttribute.get();
281 break;
282 }
283 }
284 if ( !attribute ) {
285 throw std::runtime_error("InstanceDataReader: unknown global attribute '" + attributeName + "'");
286 }
287 if ( attribute->expression ) {
288 throw std::runtime_error("InstanceDataReader: global attribute '" + attributeName + "' is assigned by the model and must not be given");
289 }
290
291 BPMNOS::Model::Expression expression(handle, BPMNOS::InputEncoder::fragment(expressionString), model->attributeRegistry);
292 // the global attributes are the first data attributes
293 BPMNOS::Data globalValues(model->attributes.size());
294 for ( auto& [globalAttribute, value] : globals ) {
295 globalValues.attributes[globalAttribute->index] = value;
296 }
297 for ( auto referencedAttribute : expression.variables ) {
298 if ( !globals.contains(referencedAttribute) ) {
299 throw std::runtime_error("InstanceDataReader: global attribute '" + attributeName + "' refers to global attribute '" + referencedAttribute->name + "' without a value");
300 }
301 }
302 // the global objects given by earlier rows or by the model
303 for ( auto& object : model->objects ) {
304 globalValues.objects.push_back( knownObject(std::nullopt, object.get()) );
305 }
306 for ( auto referencedAttribute : expression.paths ) {
307 // a path to an input has no attribute, an input holding its object from the start
308 if ( referencedAttribute && !globalValues.objects[referencedAttribute->index] ) {
309 throw std::runtime_error("InstanceDataReader: global attribute '" + attributeName + "' refers to global object '" + referencedAttribute->name + "' without a value");
310 }
311 }
312
313 auto value = expression.execute(BPMNOS::Status{}, globalValues);
314 if ( !value.has_value() ) {
315 throw std::runtime_error("InstanceDataReader: failed to evaluate global attribute '" + attributeName + "'");
316 }
317 globals[attribute] = convert(value.value(), attribute->type);
318}

320BPMNOS::number InstanceDataReader::evaluate(size_t instanceId, const BPMN::Node* node, const std::string& expressionString, ValueType type, const LIMEX::Handle<double>& handle) const {
321 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
322 BPMNOS::Model::Expression expression(handle, BPMNOS::InputEncoder::fragment(expressionString), extensionElements->attributeRegistry);
323
324 BPMNOS::Status status(extensionElements->attributeRegistry.statusAttributes.size());
325 BPMNOS::Data data(extensionElements->attributeRegistry.dataAttributes.size());
326 // the global attributes are the first data attributes
327 for ( auto& [attribute, value] : globals ) {
328 data.attributes[attribute->index] = value;
329 }
330 auto instanceValues = values.find(instanceId);
331 if ( instanceValues != values.end() ) {
332 for ( auto& [attribute, value] : instanceValues->second ) {
333 if ( attribute->category == BPMNOS::Model::Attribute::Category::STATUS ) {
334 status.attributes[attribute->index] = value;
335 }
336 else if ( attribute->category == BPMNOS::Model::Attribute::Category::DATA ) {
337 data.attributes[attribute->index] = value;
338 }
339 }
340 }
341
342 for ( auto attribute : expression.variables ) {
343 bool known = attribute->isGlobal ?
344 globals.contains(attribute) :
345 ( instanceValues != values.end() && instanceValues->second.contains(attribute) );
346 if ( !known ) {
347 throw std::runtime_error("InstanceDataReader: expression '" + expressionString + "' refers to attribute '" + attribute->name + "' without a value");
348 }
349 }
350
351 // the objects given by earlier rows or by the model, the global objects being the first data objects
352 auto& registry = extensionElements->attributeRegistry;
353 status.objects.resize(registry.statusObjects.size());
354 for ( auto object : registry.statusObjects ) {
355 status.objects[object->index] = knownObject(instanceId, object);
356 }
357 data.objects.resize(registry.dataObjects.size());
358 for ( auto object : registry.dataObjects ) {
359 bool global = ( object->index < model->objects.size() );
360 data.objects[object->index] = knownObject(global ? std::nullopt : std::optional<size_t>(instanceId), object);
361 }
362 for ( auto attribute : expression.paths ) {
363 // a path to an input has no attribute, an input holding its object from the start
364 if ( attribute && !registry.getObject(attribute, status, data) ) {
365 throw std::runtime_error("InstanceDataReader: expression '" + expressionString + "' refers to object '" + attribute->name + "' without a value");
366 }
367 }
368
369 auto value = expression.execute(status, data);
370 if ( !value.has_value() ) {
371 throw std::runtime_error("InstanceDataReader: failed to evaluate expression '" + expressionString + "'");
372 }
373 return convert(value.value(), type);
374}
