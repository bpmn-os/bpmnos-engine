49void StochasticDataProvider::initializeStochasticHandle() {

62void StochasticDataProvider::readInstances() {
63 CSVReader::Table table = reader.read();
64 if (table.empty()) {
65 throw std::runtime_error("StochasticDataProvider: table '" + reader.instanceFileOrString + "' is empty");
66 }
67
68 if (table.size() < 2) {
69 throw std::runtime_error("StochasticDataProvider: table must have at least a header and one data row");
70 }
71
72 columnCount = table[0].size();
73 if (columnCount != 3 && columnCount != 4 && columnCount != 6) {
74 throw std::runtime_error("StochasticDataProvider: expected 3, 4, or 6 columns, got " + std::to_string(columnCount));
75 }
76
77 // Column indices
78 enum { INSTANCE_ID, NODE_ID, INITIALIZATION, DISCLOSURE, READY, COMPLETION };
79
80 for (auto& row : table | std::views::drop(1)) {
81 if (row.empty()) {
82 continue;
83 }
84 if (row.size() != columnCount) {
85 throw std::runtime_error("StochasticDataProvider: inconsistent number of cells");
86 }
87
88 // Get instance ID
89 if (!std::holds_alternative<std::string>(row.at(INSTANCE_ID))) {
90 throw std::runtime_error("StochasticDataProvider: illegal instance id");
91 }
92 std::string instanceIdentifier = std::get<std::string>(row.at(INSTANCE_ID));
93
94 // Get node ID
95 if (!std::holds_alternative<std::string>(row.at(NODE_ID))) {
96 throw std::runtime_error("StochasticDataProvider: illegal node id");
97 }
98 std::string nodeId = std::get<std::string>(row.at(NODE_ID));
99
100 // Get initialization expression
101 if (!std::holds_alternative<std::string>(row.at(INITIALIZATION))) {
102 throw std::runtime_error("StochasticDataProvider: illegal initialization");
103 }
104 std::string initializationString = std::get<std::string>(row.at(INITIALIZATION));
105
106 // Get disclosure (if 4+ columns)
107 std::string disclosureExpression;
108 if (columnCount >= 4) {
109 if (std::holds_alternative<std::string>(row.at(DISCLOSURE))) {
110 disclosureExpression = std::get<std::string>(row.at(DISCLOSURE));
111 }
112 else if (std::holds_alternative<BPMNOS::number>(row.at(DISCLOSURE))) {
113 disclosureExpression = std::to_string((double)std::get<BPMNOS::number>(row.at(DISCLOSURE)));
114 }
115 }
116
117 // Get ready (if 6 columns)
118 std::string readyExpression;
119 if (columnCount == 6) {
120 if (!std::holds_alternative<std::string>(row.at(READY))) {
121 throw std::runtime_error("StochasticDataProvider: illegal ready");
122 }
123 readyExpression = std::get<std::string>(row.at(READY));
124 }
125
126 // Get completion (if 6 columns)
127 std::string completionExpression;
128 if (columnCount == 6) {
129 if (!std::holds_alternative<std::string>(row.at(COMPLETION))) {
130 throw std::runtime_error("StochasticDataProvider: illegal completion");
131 }
132 completionExpression = std::get<std::string>(row.at(COMPLETION));
133 }
134
135 // Handle global attributes
136 if (instanceIdentifier.empty() && nodeId.empty()) {
137 if (!disclosureExpression.empty()) {
138 throw std::runtime_error("StochasticDataProvider: global attributes must not have disclosure");
139 }
140 if (!readyExpression.empty()) {
141 throw std::runtime_error("StochasticDataProvider: global attributes must not have ready");
142 }
143 if (!completionExpression.empty()) {
144 throw std::runtime_error("StochasticDataProvider: global attributes must not have completion");
145 }
146 if (initializationString.empty()) {
147 continue;
148 }
149 evaluateGlobal(initializationString);
150 }
151 else if (instanceIdentifier.empty()) {
152 throw std::runtime_error("StochasticDataProvider: instance id required when node id is provided");
153 }
154 else {
155 auto instanceId = (size_t)BPMNOS::to_number(instanceIdentifier, STRING);
156 BPMN::Node* node = findNode(nodeId);
157
158 // First occurrence of instance must have node = process
159 if (!instances.contains(instanceId)) {
160 if (!node->represents<BPMN::Process>()) {
161 throw std::runtime_error("StochasticDataProvider: '" + nodeId + "' is not a process (first row for instance '" + instanceIdentifier + "' must reference a process)");
162 }
163 auto process = dynamic_cast<BPMN::Process*>(node);
164 instances[instanceId] = StochasticInstanceData{process, instanceId,
165 std::numeric_limits<BPMNOS::number>::max(), {}};
166 disclosureTimes[instanceId][process] = 0;
167 }
168
169 // Handle COMPLETION expression (valid for all Task types)
170 if (!completionExpression.empty()) {
171 if (!node->represents<BPMN::Task>()) {
172 throw std::runtime_error("StochasticDataProvider: completion expressions are not allowed for node '" + nodeId + "'");
173 }
174
175 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
176 auto expression = std::make_shared<Expression>(stochasticHandle, InputEncoder::fragment(completionExpression),
177 extensionElements->attributeRegistry);
178
179 // Completion expressions must only modify STATUS attributes
180 if (expression->target.has_value() &&
181 expression->target.value()->category != BPMNOS::Model::Attribute::Category::STATUS) {
182 throw std::runtime_error("StochasticDataProvider: completion expression for '" + nodeId + "' attempts to modify non-status attribute '" + expression->target.value()->name + "'");
183 }
184
185 // For SendTask, ReceiveTask, DecisionTask: timestamp must not be modified (completion time is event-driven)
186 if (expression->target.has_value() &&
187 expression->target.value()->name == Keyword::Timestamp &&
188 (node->represents<BPMN::SendTask>() ||
189 node->represents<BPMN::ReceiveTask>() ||
190 node->represents<DecisionTask>())) {
191 throw std::runtime_error("StochasticDataProvider: completion expression must not modify timestamp for '" + nodeId + "'");
192 }
193
194 completionExpressions[instanceId][node].push_back(std::move(expression));
195 }
196
197 // Handle READY expression (valid for all Activity types)
198 if (!readyExpression.empty()) {
199 if (!node->represents<BPMN::Activity>()) {
200 throw std::runtime_error("StochasticDataProvider: '" + nodeId + "' is not an activity (ready expressions are only allowed for activities)");
201 }
202
203 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
204 auto expression = std::make_shared<Expression>(stochasticHandle, InputEncoder::fragment(readyExpression), extensionElements->attributeRegistry);
205
206 // Ready expressions must only modify STATUS attributes
207 if (expression->target.has_value() &&
208 expression->target.value()->category != BPMNOS::Model::Attribute::Category::STATUS) {
209 throw std::runtime_error("StochasticDataProvider: ready expression for '" + nodeId + "' attempts to modify non-status attribute '" + expression->target.value()->name + "'");
210 }
211
212
213 readyExpressions[instanceId][node].push_back(std::move(expression));
214 }
215
216 // Handle INITIALIZATION
217 if (initializationString.empty()) {
218 if (!disclosureExpression.empty()) {
219 throw std::runtime_error("StochasticDataProvider: disclosure expression requires initialization expression in the same row");
220 }
221 }
222 else {
223 auto [attribute, expressionString] = lookupAttribute(node, initializationString);
224 auto extensionElements = node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
225
226 // Create shared expressions for per-scenario evaluation
227 auto initializationExpression = std::make_shared<Expression>(stochasticHandle, InputEncoder::fragment(expressionString), extensionElements->attributeRegistry);
228
229 // If no DISCLOSURE expression, default to disclosure at time 0
230 std::string disclosureString = disclosureExpression.empty() ? "0" : disclosureExpression;
231 auto disclosureExpressionPtr = std::make_shared<Expression>(stochasticHandle, InputEncoder::fragment(disclosureString), extensionElements->attributeRegistry);
232
233 deferredAttributes[instanceId].push_back({attribute, node, initializationExpression, disclosureExpressionPtr});
234 }
235 }
236 }
237
238 // Finalize instances
239 for (auto& [id, instance] : instances) {
240 ensureDefaultValue(instance, Keyword::Instance, id);
241 ensureDefaultValue(instance, Keyword::Timestamp);
242
243 // Set placeholder instantiation time (actual value will be per-scenario)
244 auto timestampAttribute = attributes[instance.process][Keyword::Timestamp];
245 instance.instantiation = instance.data.at(timestampAttribute);
246 }

265BPMNOS::number StochasticDataProvider::getEffectiveDisclosure(size_t instanceId, const BPMN::Node* node, BPMNOS::number ownDisclosure) {
266 BPMNOS::number effectiveDisclosure = ownDisclosure;
267
268 if (auto childNode = node->represents<BPMN::ChildNode>()) {
269 auto parentNode = childNode->parent;
270 if (!disclosureTimes[instanceId].contains(parentNode)) {
271 throw std::runtime_error("StochasticDataProvider: disclosure for '" + node->id + "' given before parent '" + parentNode->id + "'");
272 }
273 effectiveDisclosure = std::max(effectiveDisclosure, disclosureTimes[instanceId][parentNode]);
274 }
275
276 if (!disclosureTimes[instanceId].contains(node)) {
277 disclosureTimes[instanceId][node] = effectiveDisclosure;
278 }
279 else {
280 disclosureTimes[instanceId][node] = std::max(disclosureTimes[instanceId][node], effectiveDisclosure);
281 }
282
283 return effectiveDisclosure;
284}

286std::unique_ptr<Scenario> StochasticDataProvider::createScenario(unsigned int scenarioId) {
287 auto scenario = std::make_unique<StochasticScenario>(model.get(), globalValueMap, seed + scenarioId);
288
289 // Set references for expression evaluation
290 scenario->randomFactory = &randomFactory;
291 scenario->stochasticHandle = &stochasticHandle;
292
293 for (auto& [id, instance] : instances) {
294 scenario->addInstance(instance.process, id);
295 for (auto& [attribute, value] : instance.data) {
296 scenario->setValue(id, attribute, value);
297 }
298 }
299
300 // Set node disclosure times (from immediate disclosures)
301 for (auto& [instanceId, nodes] : disclosureTimes) {
302 for (auto& [node, disclosureTime] : nodes) {
303 scenario->setDisclosure(instanceId, node, disclosureTime);
304 }
305 }
306
307 // Add deferred disclosures (shared expressions, evaluated per-scenario with different RNG)
308 for (auto& [instanceId, deferreds] : deferredAttributes) {
309 for (auto& deferred : deferreds) {
310 scenario->addDeferredDisclosure(
311 instanceId,
312 {deferred.attribute, deferred.node, deferred.initializationExpression, deferred.disclosureExpression}
313 );
314 }
315 }
316
317 // Share completion expressions
318 for (auto& [instanceId, tasks] : completionExpressions) {
319 for (auto& [task, expressions] : tasks) {
320 for (auto& expression : expressions) {
321 scenario->addCompletionExpression(instanceId, task, expression);
322 }
323 }
324 }
325
326 // Share ready expressions
327 for (auto& [instanceId, nodes] : readyExpressions) {
328 for (auto& [node, expressions] : nodes) {
329 for (auto& expression : expressions) {
330 scenario->addReadyExpression(instanceId, node, expression);
331 }
332 }
333 }
334
335 // Evaluate deferred disclosures with scenario-specific RNG
336 scenario->evaluateDeferredDisclosures();
337
338 // Reveal data disclosed at time 0
339 scenario->revealData(0);
340
341 return scenario;
