73void StochasticDataProvider::readValue(InstanceDataReader& reader, const InstanceDataReader::Row& row, [[maybe_unused]] const LIMEX::Handle<double>& rowHandle) {
74 enum { DISCLOSURE, READY, COMPLETION };
75 auto& disclosure = row.cells[DISCLOSURE];
76 auto& ready = row.cells[READY];
77 auto& completion = row.cells[COMPLETION];
78
79 if ( !ready.empty() ) {
80 // the engine creates event subprocesses and compensation activities itself
81 auto activity = row.node->represents<BPMN::Activity>();
82 if ( !activity || row.node->represents<BPMN::EventSubProcess>() || activity->isForCompensation ) {
83 throw std::runtime_error("StochasticDataProvider: no ready expression may be given for '" + row.node->id + "'");
84 }
85 readyExpressions[row.instanceId][row.node].push_back(compileStatusExpression(row, ready, "ready"));
86 }
87
88 if ( !completion.empty() ) {
89 // send, receive and decision tasks complete as the model determines
90 if (
91 !row.node->represents<BPMN::Task>() ||
92 row.node->represents<BPMN::SendTask>() ||
93 row.node->represents<BPMN::ReceiveTask>() ||
94 row.node->represents<BPMNOS::Model::DecisionTask>()
95 ) {
96 throw std::runtime_error("StochasticDataProvider: no completion expression may be given for '" + row.node->id + "'");
97 }
98 completionExpressions[row.instanceId][row.node].push_back(compileStatusExpression(row, completion, "completion"));
99 }
100
101 if ( row.initialization.empty() ) {
102 if ( !disclosure.empty() ) {
103 throw std::runtime_error("StochasticDataProvider: disclosure of '" + row.node->id + "' requires an initialization in the same row");
104 }
105 return;
106 }
107
108 auto [attribute, expression] = reader.lookupAttribute(row.node, row.initialization);
109 auto& attributeRegistry = row.node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributeRegistry;
110 initializations.push_back({
111 row.instanceId,
112 row.node,
113 attribute,
114 std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(expression), attributeRegistry),
115 std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(disclosure.empty() ? "0" : disclosure), attributeRegistry)
116 });
117}

133void StochasticDataProvider::sample(Scenario& scenario, const Scenario* original, BPMNOS::number spawnTime) const {
134 for ( auto& [instanceId, instance] : instances ) {
135 scenario.values[instanceId] = instance.values;
136 scenario.instantiationTimes[instanceId] = instance.instantiationTime;
137 scenario.disclosureTimes[instanceId][instance.process] = 0;
138 }
139
140 // the initializations are sampled in the order of the rows, so that each may refer to those before it
141 for ( size_t k = 0; k < initializations.size(); k++ ) {
142 auto& initialization = initializations[k];
143 auto& values = scenario.values.at(initialization.instanceId);
144 auto process = instances.at(initialization.instanceId).process;
145 bool isTimestamp = ( initialization.attribute == process->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp].get() );
146
147 BPMNOS::number value;
148 BPMNOS::number disclosureTime;
149 if ( original && original->initializationDisclosureTimes[k] < spawnTime ) {
150 // the initialization was disclosed before the spawn time, so the fork keeps it
151 value = original->values.at(initialization.instanceId).at(initialization.attribute);
152 disclosureTime = original->initializationDisclosureTimes[k];
153 }
154 else {
155 auto& attributeRegistry = initialization.node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributeRegistry;
156 BPMNOS::Values status(attributeRegistry.statusAttributes.size());
157 BPMNOS::Values data(attributeRegistry.dataAttributes.size());
158 for ( auto& [attribute, attributeValue] : values ) {
159 if ( attribute->category == BPMNOS::Model::Attribute::Category::STATUS && attribute->index < status.size() && attributeRegistry.contains(attribute) ) {
160 status[attribute->index] = attributeValue;
161 }
162 else if ( attribute->category == BPMNOS::Model::Attribute::Category::DATA && attribute->index < data.size() && attributeRegistry.contains(attribute) ) {
163 data[attribute->index] = attributeValue;
164 }
165 }
166
167 randomDistributionFactory.setCurrentRng(&scenario.getRandomNumberGenerator(initialization.instanceId, initialization.node));
168 auto evaluate = [&]() {
169 return convert(initialization.value->execute(status, data, globals).value_or(0), initialization.attribute->type);
170 };
171 value = evaluate();
172 if ( isTimestamp ) {
173 // a fork samples a timestamp before the spawn time again, and sets it to the spawn time at last
174 for ( int tries = 1; value < spawnTime && tries < maxResamplingTries; tries++ ) {
175 value = evaluate();
176 }
177 value = std::max(value, spawnTime);
178 }
179 if ( initialization.attribute->category == BPMNOS::Model::Attribute::Category::STATUS ) {
180 status[initialization.attribute->index] = value;
181 }
182 else if ( initialization.attribute->category == BPMNOS::Model::Attribute::Category::DATA ) {
183 data[initialization.attribute->index] = value;
184 }
185 auto evaluateDisclosure = [&]() {
186 return BPMNOS::number(std::ceil(initialization.disclosure->execute(status, data, globals).value_or(0)));
187 };
188 disclosureTime = evaluateDisclosure();
189 // a fork samples a disclosure time before the spawn time again, and sets it to the spawn time at last
190 for ( int tries = 1; disclosureTime < spawnTime && tries < maxResamplingTries; tries++ ) {
191 disclosureTime = evaluateDisclosure();
192 }
193 disclosureTime = std::max(disclosureTime, spawnTime);
194 randomDistributionFactory.setCurrentRng(nullptr);
195 }
196
197 values[initialization.attribute] = value;
198 scenario.initializationDisclosureTimes.push_back(disclosureTime);
199 if ( isTimestamp ) {
200 // instances are instantiated at integral times
201 scenario.instantiationTimes[initialization.instanceId] = BPMNOS::number(std::ceil((double)value));
202 }
203
204 // the disclosure time of a node is at least that of the scope containing it
205 auto& nodeDisclosureTimes = scenario.disclosureTimes.at(initialization.instanceId);
206 if ( auto childNode = initialization.node->represents<BPMN::ChildNode>() ) {
207 if ( auto scope = nodeDisclosureTimes.find(childNode->parent); scope != nodeDisclosureTimes.end() ) {
208 disclosureTime = std::max(disclosureTime, scope->second);
209 }
210 }
211 auto [it, inserted] = nodeDisclosureTimes.try_emplace(initialization.node, disclosureTime);
212 if ( !inserted ) {
213 it->second = std::max(it->second, disclosureTime);
214 }
215 }

247void StochasticDataProvider::computeStatus(Scenario& scenario, const Expressions& expressions, size_t instanceId, const BPMN::Node* node, BPMNOS::Values& status, const BPMNOS::SharedValues& data, const BPMNOS::Values& globals, BPMNOS::number earliest) const {
248 auto instanceExpressions = expressions.find(instanceId);
249 if ( instanceExpressions == expressions.end() ) {
250 return;
251 }
252 auto nodeExpressions = instanceExpressions->second.find(node);
253 if ( nodeExpressions == instanceExpressions->second.end() ) {
254 return;
255 }
256 randomDistributionFactory.setCurrentRng(&scenario.getRandomNumberGenerator(instanceId, node));
257 const auto initialStatus = status;
258 auto evaluate = [&]() {
259 status = initialStatus;
260 for ( auto& expression : nodeExpressions->second ) {
261 if ( auto value = expression->execute(status, data, globals) ) {
262 auto target = expression->target.value();
263 status[target->index] = convert(value.value(), target->type);
264 }
265 }
266 };
267 evaluate();
268 // a status computed anew for a token awaiting its event in an installed system state is sampled again
269 // while its timestamp precedes the time of that state, and its timestamp is then set to that time, since
270 // the event has not happened before; a status computed otherwise has the lowest number as earliest time
271 constexpr auto Timestamp = BPMNOS::Model::ExtensionElements::Index::Timestamp;
272 auto tooEarly = [&]() {
273 return status[Timestamp].has_value() && status[Timestamp].value() < earliest;
274 };
275 // sampling again can change the timestamp only if an expression assigns it
276 bool timestampSampled = std::ranges::any_of(nodeExpressions->second, [](auto& expression) {
277 auto target = expression->target.value();
278 return target->category == BPMNOS::Model::Attribute::Category::STATUS && target->index == Timestamp;
279 });
280 for ( int tries = 1; timestampSampled && tooEarly() && tries < maxResamplingTries; tries++ ) {
281 evaluate();
282 }
283 if ( tooEarly() ) {
284 status[Timestamp] = earliest;
285 }
286 randomDistributionFactory.setCurrentRng(nullptr);
