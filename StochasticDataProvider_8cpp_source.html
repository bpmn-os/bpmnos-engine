68std::unique_ptr<BPMNOS::Model::Expression> StochasticDataProvider::compileReadyExpression(const InstanceDataReader::Row& row, const std::string& expression) const {
69 auto extensionElements = row.node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
70 auto compiled = std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(expression), extensionElements->attributeRegistry);
71 if ( !compiled->target.has_value() ) {
72 throw std::runtime_error("StochasticDataProvider: the ready expression of '" + row.node->id + "' must assign an attribute");
73 }
74 auto target = compiled->target.value();
75 // a ready event gives the whole status and the data the scope owns, never inherited data; an object, an element
76 // or a field may be assigned as a scalar may
77 if ( target->category != BPMNOS::Model::Attribute::Category::STATUS && !holds(extensionElements->data, target) && !holds(extensionElements->dataObjects, target) ) {
78 throw std::runtime_error("StochasticDataProvider: the ready expression of '" + row.node->id + "' must assign a status attribute or a data attribute the scope owns");
79 }
80 if ( assignsTimestamp(*compiled) ) {
81 // the timestamp determines when the token becomes ready, so it is sampled when the token arrives, before any
82 // other ready expression and before the attributes of the scope exist
83 if ( getReadyExpressions(row.instanceId, row.node) ) {
84 throw std::runtime_error("StochasticDataProvider: the ready expression of '" + row.node->id + "' assigning the timestamp must be the first ready expression of the node");
85 }
86 for ( auto input : compiled->inputs ) {
87 if ( holds(extensionElements->attributes, input) || holds(extensionElements->statusObjects, input) || holds(extensionElements->data, input) || holds(extensionElements->dataObjects, input) ) {
88 throw std::runtime_error("StochasticDataProvider: the ready expression of '" + row.node->id + "' assigning the timestamp must read only what the scope inherits, not '" + input->name + "'");
89 }
90 }
91 }
92 return compiled;
93}

123void StochasticDataProvider::requireInitialised(const InstanceDataReader& reader, size_t instanceId, const BPMNOS::Model::Expression& expression, const std::string& text) const {
124 // a global value is given by the instance data before any instance is read, an object possibly by the model
125 for ( auto attribute : expression.variables ) {
126 bool known = attribute->isGlobal ? reader.globals.contains(attribute) : initialisedBefore(instanceId, attribute);
127 if ( !known ) {
128 throw std::runtime_error("StochasticDataProvider: expression '" + text + "' refers to attribute '" + attribute->name + "' without a value");
129 }
130 }
131 for ( auto object : expression.paths ) {
132 // a path to an input has no attribute, an input holding its object from the start
133 if ( !object ) {
134 continue;
135 }
136 bool known = std::holds_alternative<size_t>(object->initialization) || ( object->isGlobal ? reader.globalObjects.contains(object) : initialisedBefore(instanceId, object) );
137 if ( !known ) {
138 throw std::runtime_error("StochasticDataProvider: expression '" + text + "' refers to object '" + object->name + "' without a value");
139 }
140 }

143void StochasticDataProvider::readValue(InstanceDataReader& reader, const InstanceDataReader::Row& row, [[maybe_unused]] const LIMEX::Handle<double>& rowHandle) {
144 enum { DISCLOSURE, READY, COMPLETION };
145 auto& disclosure = row.cells[DISCLOSURE];
146 auto& ready = row.cells[READY];
147 auto& completion = row.cells[COMPLETION];
148
149 auto activity = row.node->represents<BPMN::Activity>();
150 if ( !ready.empty() ) {
151 // the engine creates event subprocesses itself
152 if ( !activity || row.node->represents<BPMN::EventSubProcess>() ) {
153 throw std::runtime_error("StochasticDataProvider: no ready expression may be given for '" + row.node->id + "'");
154 }
155 auto compiled = compileReadyExpression(row, ready);
156 if ( activity->isForCompensation && assignsTimestamp(*compiled) ) {
157 // a compensation activity becomes ready when compensation is triggered
158 throw std::runtime_error("StochasticDataProvider: the ready expression of compensation activity '" + row.node->id + "' must not assign the timestamp");
159 }
160 readyExpressions[row.instanceId][row.node].push_back(std::move(compiled));
161 }
162
163 if ( activity && activity->isForCompensation && !disclosure.empty() ) {
164 // a compensation activity becomes ready when compensation is triggered, the disclosure of its scope having passed
165 throw std::runtime_error("StochasticDataProvider: no disclosure may be given for compensation activity '" + row.node->id + "'");
166 }
167
168 if ( !completion.empty() ) {
169 // send, receive and decision tasks complete as the model determines
170 if (
171 !row.node->represents<BPMN::Task>() ||
172 row.node->represents<BPMN::SendTask>() ||
173 row.node->represents<BPMN::ReceiveTask>() ||
174 row.node->represents<BPMNOS::Model::DecisionTask>()
175 ) {
176 throw std::runtime_error("StochasticDataProvider: no completion expression may be given for '" + row.node->id + "'");
177 }
178 completionExpressions[row.instanceId][row.node].push_back(compileCompletionExpression(row, completion));
179 }
180
181 if ( row.initialization.empty() ) {
182 if ( !disclosure.empty() ) {
183 throw std::runtime_error("StochasticDataProvider: disclosure of '" + row.node->id + "' requires an initialization in the same row");
184 }
185 return;
186 }
187
188 auto& attributeRegistry = row.node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributeRegistry;
189 auto disclosureText = disclosure.empty() ? std::string("0") : disclosure;
190 auto disclosureExpression = std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(disclosureText), attributeRegistry);
191 if ( reader.assignsObject(row) ) {
192 auto [object, initializer] = reader.lookupObject(row);
193 std::vector<std::unique_ptr<BPMNOS::Model::Expression>> members;
194 for ( auto& member : initializer.members() ) {
195 members.push_back( std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(member.expression), attributeRegistry) );
196 requireInitialised(reader, row.instanceId, *members.back(), member.expression);
197 }
198 initializations.push_back({
199 row.instanceId,
200 row.node,
201 object,
202 nullptr,
203 ObjectInitialization{ std::move(initializer), std::move(members), reader.getSchema(object) },
204 std::move(disclosureExpression)
205 });
206 }
207 else {
208 auto [attribute, expression] = reader.lookupAttribute(row.node, row.initialization);
209 auto value = std::make_unique<BPMNOS::Model::Expression>(handle, BPMNOS::InputEncoder::fragment(expression), attributeRegistry);
210 requireInitialised(reader, row.instanceId, *value, expression);
211 initializations.push_back({
212 row.instanceId,
213 row.node,
214 attribute,
215 std::move(value),
216 std::nullopt,
217 std::move(disclosureExpression)
218 });
219 }
220
221 // the disclosure may read the attribute or object the row initialises
222 requireInitialised(reader, row.instanceId, *initializations.back().disclosure, disclosureText);
223}

239void StochasticDataProvider::sample(Scenario& scenario, const Scenario* original, BPMNOS::number spawnTime) const {
240 for ( auto& instance : instances ) {
241 scenario.values[instance.id] = instance.values;
242 scenario.objects[instance.id] = instance.objects;
243 scenario.instantiationTimes[instance.id] = instance.instantiationTime;
244 scenario.disclosureTimes[instance.id][instance.process] = 0;
245 }
246
247 // the initializations are sampled in the order of the rows, so that each may refer to those before it
248 for ( size_t k = 0; k < initializations.size(); k++ ) {
249 auto& initialization = initializations[k];
250 auto& values = scenario.values.at(initialization.instanceId);
251 auto& objects = scenario.objects.at(initialization.instanceId);
252 auto process = getInstance(initialization.instanceId).process;
253 bool isTimestamp = ( !initialization.object.has_value() && initialization.attribute == process->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp].get() );
254
255 BPMNOS::number value = 0;
256 std::shared_ptr<const BPMNOS::Object> object;
257 BPMNOS::number disclosureTime;
258 if ( original && original->initializationDisclosureTimes[k] < spawnTime ) {
259 // the initialization was disclosed before the spawn time, so the fork keeps it
260 if ( initialization.object.has_value() ) {
261 object = original->objects.at(initialization.instanceId).at(initialization.attribute);
262 }
263 else {
264 value = original->values.at(initialization.instanceId).at(initialization.attribute);
265 }
266 disclosureTime = original->initializationDisclosureTimes[k];
267 }
268 else {
269 auto& attributeRegistry = initialization.node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributeRegistry;
270 BPMNOS::Status status(attributeRegistry.statusAttributes.size());
271 BPMNOS::Data data(attributeRegistry.dataAttributes.size());
272 // the global attributes are the first data attributes
273 std::copy(globals.attributes.begin(), globals.attributes.end(), data.attributes.begin());
274 for ( auto& [attribute, attributeValue] : values ) {
275 if ( attribute->category == BPMNOS::Model::Attribute::Category::STATUS && attribute->index < status.attributes.size() && attributeRegistry.contains(attribute) ) {
276 status.attributes[attribute->index] = attributeValue;
277 }
278 else if ( attribute->category == BPMNOS::Model::Attribute::Category::DATA && attribute->index < data.attributes.size() && attributeRegistry.contains(attribute) ) {
279 data.attributes[attribute->index] = attributeValue;
280 }
281 }
282 // the objects given by earlier rows, every other object holding its default, the global objects being the
283 // first data objects
284 auto objectOf = [&](const BPMNOS::Model::Attribute* attribute) {
285 auto it = objects.find(attribute);
286 return it != objects.end() ? it->second : getDefaultObject(attribute);
287 };
288 status.objects.resize(attributeRegistry.statusObjects.size());
289 for ( auto attribute : attributeRegistry.statusObjects ) {
290 status.objects[attribute->index] = objectOf(attribute);
291 }
292 data.objects.resize(attributeRegistry.dataObjects.size());
293 for ( auto attribute : attributeRegistry.dataObjects ) {
294 data.objects[attribute->index] = ( attribute->index < globals.objects.size() ) ? globals.objects[attribute->index] : objectOf(attribute);
295 }
296
297 // an expression of the row that yields no value is refused, as when the instance data are read
298 auto valueOf = [&](const BPMNOS::Model::Expression& expression) {
299 auto result = expression.execute(status, data);
300 if ( !result.has_value() ) {
301 throw std::runtime_error("StochasticDataProvider: the initialization of '" + initialization.attribute->name + "' of instance '" + stringRegistry[initialization.instanceId] + "' or its disclosure yields no value");
302 }
303 return result.value();
304 };
305
306 randomDistributionFactory.setCurrentRng(&scenario.getRandomNumberGenerator(initialization.instanceId, initialization.node));
307 if ( initialization.object.has_value() ) {
308 // the members read the values before the row, the object being given only when all its members are evaluated
309 std::vector<double> memberValues;
310 for ( auto& member : initialization.object->members ) {
311 memberValues.push_back( valueOf(*member) );
312 }
313 object = InstanceDataReader::instantiate(initialization.object->schema, initialization.attribute, initialization.object->initializer, memberValues);
314 attributeRegistry.setObject(initialization.attribute, status, data, object, *this);
315 }
316 else {
317 auto evaluate = [&]() {
318 return BPMNOS::cast(valueOf(*initialization.value), std::get<BPMNOS::ValueType>(initialization.attribute->type));
319 };
320 value = evaluate();
321 if ( isTimestamp ) {
322 // a fork samples a timestamp before the spawn time again, and sets it to the spawn time at last
323 for ( int tries = 1; value < spawnTime && tries < maxResamplingTries; tries++ ) {
324 value = evaluate();
325 }
326 value = std::max(value, spawnTime);
327 }
328 if ( initialization.attribute->category == BPMNOS::Model::Attribute::Category::STATUS ) {
329 status.attributes[initialization.attribute->index] = value;
330 }
331 else if ( initialization.attribute->category == BPMNOS::Model::Attribute::Category::DATA ) {
332 data.attributes[initialization.attribute->index] = value;
333 }
334 }
335 auto evaluateDisclosure = [&]() {
336 return BPMNOS::number(std::ceil(valueOf(*initialization.disclosure)));
337 };
338 disclosureTime = evaluateDisclosure();
339 // a fork samples a disclosure time before the spawn time again, and sets it to the spawn time at last
340 for ( int tries = 1; disclosureTime < spawnTime && tries < maxResamplingTries; tries++ ) {
341 disclosureTime = evaluateDisclosure();
342 }
343 disclosureTime = std::max(disclosureTime, spawnTime);
344 randomDistributionFactory.setCurrentRng(nullptr);
345 }
346
347 if ( initialization.object.has_value() ) {
348 objects[initialization.attribute] = std::move(object);
349 }
350 else {
351 values[initialization.attribute] = value;
352 }
353 scenario.initializationDisclosureTimes.push_back(disclosureTime);
354 if ( isTimestamp ) {
355 // instances are instantiated at integral times
356 scenario.instantiationTimes[initialization.instanceId] = BPMNOS::number(std::ceil((double)value));
357 }
358
359 // the disclosure time of a node is at least that of the scope containing it
360 auto& nodeDisclosureTimes = scenario.disclosureTimes.at(initialization.instanceId);
361 if ( auto childNode = initialization.node->represents<BPMN::ChildNode>() ) {
362 if ( auto scope = nodeDisclosureTimes.find(childNode->parent); scope != nodeDisclosureTimes.end() ) {
363 disclosureTime = std::max(disclosureTime, scope->second);
364 }
365 }
366 auto [it, inserted] = nodeDisclosureTimes.try_emplace(initialization.node, disclosureTime);
367 if ( !inserted ) {
368 it->second = std::max(it->second, disclosureTime);
369 }
370 }
371}

389std::optional<BPMNOS::number> StochasticDataProvider::getReadyTimestamp(StaticDataProvider::Scenario& scenario, const Token* token, BPMNOS::number earliest) const {
390 auto instanceId = (size_t)token->owner->root->instance.value();
391 auto expressions = getReadyExpressions(instanceId, token->node);
392 if ( !expressions || !assignsTimestamp(*expressions->front()) ) {
393 return std::nullopt;
394 }
395 auto& expression = *expressions->front();
396 auto target = expression.target.value();
397 randomDistributionFactory.setCurrentRng(&static_cast<Scenario&>(scenario).getRandomNumberGenerator(instanceId, token->node));
398 auto evaluate = [&]() -> std::optional<BPMNOS::number> {
399 if ( auto value = expression.execute(token->status, *token->data) ) {
400 return BPMNOS::cast(value.value(), std::get<BPMNOS::ValueType>(target->type));
401 }
402 return std::nullopt;
403 };
404 auto timestamp = evaluate();
405 // a timestamp sampled for a token awaiting its ready event in an installed system state is sampled again while
406 // it precedes the time of that state, and is then set to that time, since the event has not happened before
407 for ( int tries = 1; timestamp.has_value() && timestamp.value() < earliest && tries < maxResamplingTries; tries++ ) {
408 timestamp = evaluate();
409 }
410 if ( timestamp.has_value() && timestamp.value() < earliest ) {
411 timestamp = earliest;
412 }
413 randomDistributionFactory.setCurrentRng(nullptr);
414 return timestamp;
415}

445void StochasticDataProvider::computeStatus(Scenario& scenario, size_t instanceId, const BPMN::Node* node, BPMNOS::Status& status, BPMNOS::SharedData& data, BPMNOS::number earliest) const {
446 auto instanceExpressions = completionExpressions.find(instanceId);
447 if ( instanceExpressions == completionExpressions.end() ) {
448 return;
449 }
450 auto nodeExpressions = instanceExpressions->second.find(node);
451 if ( nodeExpressions == instanceExpressions->second.end() ) {
452 return;
453 }
454 randomDistributionFactory.setCurrentRng(&scenario.getRandomNumberGenerator(instanceId, node));
455 const auto initialStatus = status;
456 auto evaluate = [&]() {
457 status = initialStatus;
458 for ( auto& expression : nodeExpressions->second ) {
459 if ( expression->writesToObject() ) {
460 // the status object is copied on write, so that the status restored before a new sample keeps the object
461 // it was computed from
462 expression->writeToObject(status, data, *this);
463 }
464 else if ( auto value = expression->execute(status, data) ) {
465 auto target = expression->target.value();
466 status.attributes[target->index] = BPMNOS::cast(value.value(), std::get<BPMNOS::ValueType>(target->type));
467 }
468 }
469 };
470 evaluate();
471 // a status computed anew for a token awaiting its event in an installed system state is sampled again
472 // while its timestamp precedes the time of that state, and its timestamp is then set to that time, since
473 // the event has not happened before; a status computed otherwise has the lowest number as earliest time
474 constexpr auto Timestamp = BPMNOS::Model::ExtensionElements::Index::Timestamp;
475 auto tooEarly = [&]() {
476 return status.attributes[Timestamp].has_value() && status.attributes[Timestamp].value() < earliest;
477 };
478 // sampling again can change the timestamp only if an expression assigns it
479 bool timestampSampled = std::ranges::any_of(nodeExpressions->second, [](auto& expression) {
480 auto target = expression->target.value();
481 return target->category == BPMNOS::Model::Attribute::Category::STATUS && target->index == Timestamp;
482 });
483 for ( int tries = 1; timestampSampled && tooEarly() && tries < maxResamplingTries; tries++ ) {
484 evaluate();
485 }
486 if ( tooEarly() ) {
487 status.attributes[Timestamp] = earliest;
488 }
489 randomDistributionFactory.setCurrentRng(nullptr);
