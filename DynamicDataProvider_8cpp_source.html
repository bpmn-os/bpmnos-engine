18void DynamicDataProvider::readValue(InstanceDataReader& reader, const InstanceDataReader::Row& row, const LIMEX::Handle<double>& handle) {
19 enum { DISCLOSURE };
20 auto& disclosure = row.cells[DISCLOSURE];
21 auto& nodeDisclosureTimes = disclosureTimes[row.instanceId];
22
23 // the disclosure time of a node is at least that of the scope containing it
24 BPMNOS::number disclosureTime = 0;
25 if ( auto childNode = row.node->represents<BPMN::ChildNode>() ) {
26 auto scope = nodeDisclosureTimes.find(childNode->parent);
27 if ( scope == nodeDisclosureTimes.end() ) {
28 throw std::runtime_error("DynamicDataProvider: disclosure of '" + row.node->id + "' given before that of '" + childNode->parent->id + "'");
29 }
30 disclosureTime = scope->second;
31 }
32
33 if ( row.initialization.empty() ) {
34 if ( !disclosure.empty() ) {
35 throw std::runtime_error("DynamicDataProvider: disclosure of '" + row.node->id + "' requires an initialization in the same row");
36 }
37 }
38 else {
39 StaticDataProvider::readValue(reader, row, handle);
40 if ( !disclosure.empty() ) {
41 // the disclosure may refer to the value just initialized
42 auto ownDisclosureTime = BPMNOS::number(std::ceil((double)reader.evaluate(row.instanceId, row.node, disclosure, DECIMAL, handle)));
43 disclosureTime = std::max(disclosureTime, ownDisclosureTime);
44 }
45 }
46
47 auto [it, inserted] = nodeDisclosureTimes.try_emplace(row.node, disclosureTime);
48 if ( !inserted ) {
49 it->second = std::max(it->second, disclosureTime);
50 }
51}
