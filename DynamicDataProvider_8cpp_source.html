36 CSVReader::Table table = reader.read();
37 if ( table.empty() ) {
38 throw std::runtime_error("DynamicDataProvider: table '" + reader.instanceFileOrString + "' is empty");
39 }
40
41 if ( table.size() < 2 ) {
42 throw std::runtime_error("DynamicDataProvider: table must have at least a header and one data row");
43 }
44
45 size_t columnCount = table[0].size();
46 if ( columnCount != 4 ) {
47 throw std::runtime_error("DynamicDataProvider: expected 4 columns (INSTANCE_ID, NODE_ID, INITIALIZATION, DISCLOSURE), got " + std::to_string(columnCount));
48 }

51 enum {INSTANCE_ID, NODE_ID, INITIALIZATION, DISCLOSURE};

53 for (auto &row : table | std::views::drop(1)) { // assume a single header line
54 if ( row.empty() ) {
55 continue;
56 }
57 if ( row.size() != 4 ) {
58 throw std::runtime_error("DynamicDataProvider: illegal number of cells");
59 }
60
61 // Get instance ID (may be empty for globals)
62 if ( !std::holds_alternative<std::string>(row.at(INSTANCE_ID)) ) {
63 throw std::runtime_error("DynamicDataProvider: illegal instance id");
64 }
65 std::string instanceIdentifier = std::get<std::string>(row.at(INSTANCE_ID));
66
67 // Get node ID (may be empty for globals)
68 if ( !std::holds_alternative<std::string>(row.at(NODE_ID)) ) {
69 throw std::runtime_error("DynamicDataProvider: illegal node id");
70 }
71 std::string nodeId = std::get<std::string>(row.at(NODE_ID));
72
73 // Get initialization expression
74 if ( !std::holds_alternative<std::string>(row.at(INITIALIZATION)) ) {
75 throw std::runtime_error("DynamicDataProvider: illegal initialization");

77 std::string initializationString = std::get<std::string>(row.at(INITIALIZATION));
78
79 // Get disclosure expression (may be string or number)

82 std::get<std::string>(row.at(DISCLOSURE)) :
83 std::to_string((double)std::get<BPMNOS::number>(row.at(DISCLOSURE)))
84 ;

88 if ( !disclosureString.empty() ) {

95 }

104
105 // First occurrence of instance must have node = process

107 // Check that node is a process
108 if ( !node->represents<BPMN::Process>() ) {

110 }

114 disclosure[instanceId][process] = 0;
115 }
116
117 // If no initialization, just create instance (already done above)
118 if ( initializationString.empty() ) {
119 if ( !disclosureString.empty() ) {
120 throw std::runtime_error("DynamicDataProvider: DISCLOSURE requires INITIALIZATION in the same row");
121 }
122 continue;
123 }
124

126 auto [attribute, expressionString] = lookupAttribute(node, initializationString);
127

129 BPMNOS::number value = evaluateExpression(instanceId, node, expressionString, attribute->type);
130 parseTimeEvaluatedValues[instanceId][attribute] = value;
131
132 // Parse disclosure time (can reference the just-initialized attribute)
133 BPMNOS::number ownDisclosure = 0;
134 if ( !disclosureString.empty() ) {

136 }
137

139 BPMNOS::number disclosureTime = getEffectiveDisclosure(instanceId, node, ownDisclosure);
140
141 if (disclosureTime == 0) {
142 // Immediate disclosure: store value directly
143 instance.data[attribute] = value;
144 }
145 else {
146 // Deferred disclosure: store pre-computed value for later reveal
147 deferredInitializations[instanceId].push_back({attribute, disclosureTime, value});
148 }
149 }

158 if (deferredInitializations.contains(id)) {
159 for (auto& pending : deferredInitializations[id]) {
160 if (pending.attribute == timestampAttribute) {
161 instance.data[timestampAttribute] = pending.value;
162 break;
163 }
164 }
165 }
166
167 ensureDefaultValue(instance, Keyword::Timestamp);
168 // set time of instantiation
169 instance.instantiation = instance.data.at(timestampAttribute);
170
171 // Effective instantiation time is max(instantiation, processDisclosure)
172 BPMNOS::number effectiveInstantiation = instance.instantiation;
173 if ( disclosure.contains(id) && disclosure.at(id).contains(instance.process) ) {
174 effectiveInstantiation = std::max(effectiveInstantiation, disclosure.at(id).at(instance.process));
175 }
176
177 if ( earliestInstantiation > effectiveInstantiation ) {
178 earliestInstantiation = effectiveInstantiation;
179 }

181 latestInstantiation = effectiveInstantiation;
182 }

186BPMNOS::number DynamicDataProvider::getEffectiveDisclosure(size_t instanceId, const BPMN::Node* node, BPMNOS::number ownDisclosure) {
187 BPMNOS::number effectiveDisclosure = ownDisclosure;
188
189 // Check parent scope's disclosure time
190 if ( auto childNode = node->represents<BPMN::ChildNode>() ) {
191 auto parentNode = childNode->parent;
192 if ( !disclosure[instanceId].contains(parentNode) ) {
193 throw std::runtime_error("DynamicDataProvider: disclosure for '" + node->id + "' given before parent '" + parentNode->id + "'");
194 }
195 effectiveDisclosure = std::max(effectiveDisclosure, disclosure[instanceId][parentNode]);
196 }
197
198 // Update this scope's disclosure time (track maximum seen for this scope)
199 if ( !disclosure[instanceId].contains(node) ) {
200 disclosure[instanceId][node] = effectiveDisclosure;
201 }
202 else {
203 disclosure[instanceId][node] = std::max(disclosure[instanceId][node], effectiveDisclosure);
204 }
205
206 return effectiveDisclosure;
207}

210 auto scenario = std::make_unique<DynamicScenario>(model.get(), earliestInstantiation, latestInstantiation, globalValueMap);
211 for ( auto& [id, instance] : instances ) {
212 auto& timestampAttribute = attributes[instance.process][Keyword::Timestamp];
213 auto instantiationTime = instance.data[timestampAttribute];
214 scenario->addInstance(instance.process, id, instantiationTime);
215 for ( auto& [attribute, value] : instance.data ) {
216 scenario->setValue(id, attribute, value);
217 }
218 }
219 // Set node disclosure times (effective disclosure = max of own and parent scope)
220 for ( auto& [instanceId, nodes] : disclosure ) {
221 for ( auto& [node, disclosureTime] : nodes ) {
222 scenario->setDisclosure(instanceId, node, disclosureTime);
223 }
224 }
225 // Add deferred initializations (pre-computed values)
226 for (auto& [instanceId, pendings] : deferredInitializations) {
227 for (auto& pending : pendings) {
228 scenario->addPendingDisclosure(instanceId, {pending.attribute, pending.disclosureTime, pending.value});

230 }
