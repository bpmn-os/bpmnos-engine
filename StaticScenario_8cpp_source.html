76std::optional<BPMNOS::number> StaticScenario::getValue(const Scenario::InstanceData* instance, const BPMNOS::Model::Attribute* attribute, [[maybe_unused]] const BPMNOS::number currentTime) const {
77 if ( attribute->expression && attribute->expression->type == Expression::Type::ASSIGN ) {
78 // Value is computed from an expression
79 std::vector<double> variableValues;
80 for ( auto input : attribute->expression->variables ) {
81 if ( !input->isImmutable ) {
82 return std::nullopt;
83 }
84 auto value = getValue(instance, input, currentTime);
85 if ( !value.has_value() ) {
86 return std::nullopt;
87 }
88 variableValues.push_back( (double)value.value() );
89 }
90
91 std::vector<std::vector<double>> collectionValues;
92 for ( auto input : attribute->expression->collections ) {
93 if ( !input->isImmutable ) {
94 return std::nullopt;
95 }
96 collectionValues.push_back( {} );
97 auto collection = getValue(instance, input, currentTime);
98 if ( !collection.has_value() ) {
99 return std::nullopt;
100 }
101 for ( auto value : collectionRegistry[(size_t)collection.value()] ) {
102 collectionValues.back().push_back( value );
103 }
104 }
105
106 return number(attribute->expression->compiled.evaluate(variableValues, collectionValues));
107 }
108 else {
109 // Return stored value directly
110 if ( instance->values.contains(attribute) ) {
111 return instance->values.at(attribute);
112 }
113 }
114 return std::nullopt;

154) const {
155 // status keyed by full instance id; declared instance/values keyed by root id
156 auto key = std::make_pair((size_t)instanceId, activity);

163
164 // Start with parent's status

167 // Add node's own attributes (same approach as getStatus)
