27 assert(request->token->node);

31 // determine explicitly addressed candidate messages

33 if( auto message = message_ptr.lock();
34 message &&
35 std::ranges::contains(senderCandidates, message->origin) &&
36 message->matches(recipientHeader) &&
37 ( message->recipient.has_value() || recipientHeader[BPMNOS::Model::MessageDefinition::Index::Sender].has_value() )
38 ) {
39 candidates.emplace_back( message->weak_from_this() );
40 }

43 }

51 if ( auto token = token_ptr.lock();
52 token &&
53 std::ranges::contains(recipientCandidates, token->node)
54 ) {
55 auto request = request_ptr.lock();
56 assert( request );
57 auto& recipientHeader = request->recipientHeader;
58 if (
59 message->matches(recipientHeader) &&
60 ( message->recipient.has_value() || recipientHeader[BPMNOS::Model::MessageDefinition::Index::Sender].has_value() )
61 ) {
62 candidates.emplace_back( message->weak_from_this() );
63 }
64 }
65 }
66 }
67 }

69 // A foreign state was installed (e.g. on resume). Rebuild from it: incremental Message/MessageDeliveryRequest
70 // notices are not replayed for state that already existed at install time, so a directly-addressed delivery
71 // pending at install (e.g. an order waiting at NoticeJobCompletionTask for its completion) would otherwise be

73 // them — mirroring MessageDeliveries::notice(SystemState).
74 auto systemState = static_cast<const SystemState*>(observable);
75 messages.clear();

78 notice( message.get() );
79 }
80 for ( auto& [token_ptr, request_ptr] : systemState->pendingMessageDeliveryDecisions ) {

84 }
