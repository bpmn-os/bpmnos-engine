35void StaticDataProvider::readInstances(const std::string& instanceFileOrString, const std::vector<std::string>& columns, const LIMEX::Handle<double>& handle) {
36 InstanceDataReader reader(this->model.get(), instanceFileOrString, columns);
37 for ( auto& row : reader.rows ) {
38 if ( !row.node ) {
39 reader.evaluateGlobal(row.initialization, handle);
40 }
41 else {
42 readValue(reader, row, handle);
43 }
44 }
45
46 // the globals given are completed by those the model assigns
47 globals = BPMNOS::Values(this->model->attributes.size());
48 for ( auto& [attribute, value] : reader.globals ) {
49 globals[attribute->index] = value;
50 }
51 for ( auto& attribute : this->model->attributes ) {
52 if ( attribute->expression && !globals[attribute->index].has_value() ) {
53 auto value = attribute->expression->execute(BPMNOS::Values{}, BPMNOS::Values{}, globals);
54 if ( !value.has_value() ) {
55 throw std::runtime_error("StaticDataProvider: failed to evaluate global attribute '" + attribute->id + "'");
56 }
57 globals[attribute->index] = value.value();
58 }
59 }
60
61 for ( auto& [instanceId, process] : reader.processes ) {
62 auto values = reader.values.contains(instanceId) ? reader.values.at(instanceId) : std::unordered_map<const BPMNOS::Model::Attribute*, BPMNOS::number>{};
63 reader.addDefaultValues(instanceId, values);
64 auto timestampAttribute = process->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp].get();
65 // instances are instantiated at integral times
66 auto instantiationTime = BPMNOS::number(std::ceil((double)values.at(timestampAttribute)));
67 instances[instanceId] = Instance{process, std::move(values), instantiationTime};
68 }
69}

102void StaticDataProvider::notice(const Observable* observable, Execution::Scenario& executionScenario, EventQueue& queue) const {
103 auto& scenario = static_cast<Scenario&>(executionScenario);
104
105 if ( observable->getObservableType() == Observable::Type::SystemState ) {
106 // the state of a run not yet begun holds no instance, an installed state every instance known up to its
107 // time; the instances not yet held are instantiated when they become known, and the events the tokens
108 // await are determined anew
109 auto systemState = static_cast<const SystemState*>(observable);
110 scenario.time = systemState->getTime();
111 bool runBegun = ( systemState->getTime() != std::numeric_limits<BPMNOS::number>::lowest() );
112 scenario.scheduledEvents.clear();
113 for ( auto& [instanceId, instance] : instances ) {
114 if ( !runBegun || getKnownTime(scenario, instanceId) > systemState->getTime() ) {
115 auto event = std::make_shared<InstantiationEvent>(instance.process, getStatus(scenario, instanceId, instance.process), getData(scenario, instanceId, instance.process));
116 schedule(scenario, queue, getKnownTime(scenario, instanceId), systemState->getTime(), std::move(event));
117 }
118 }
119 for ( auto& [token_ptr] : systemState->tokensAwaitingReadyEvent ) {
120 if ( auto token = token_ptr.lock() ) {
121 if ( token->node ) {
122 readyActivity(scenario, queue, token.get(), systemState->getTime());
123 }
124 else {
125 readyProcess(scenario, queue, token.get());
126 }
127 }
128 }
129 for ( auto element : systemState->tokensAwaitingCompletionEvent ) {
130 if ( auto token = std::get<1>(element).lock() ) {
131 completeTask(scenario, queue, token.get(), systemState->getTime());
132 }
133 }
134 return;
135 }
136
137 if ( observable->getObservableType() == Observable::Type::Event ) {
138 if ( auto clockTickEvent = static_cast<const Event*>(observable)->is<ClockTickEvent>() ) {
139 scenario.time = clockTickEvent->time;
140 // the events due by the time the clock advances to are released in the order of their times
141 auto end = scenario.scheduledEvents.upper_bound(clockTickEvent->time);
142 for ( auto it = scenario.scheduledEvents.begin(); it != end; ++it ) {
143 queue.push_back(it->second);
144 }
145 scenario.scheduledEvents.erase(scenario.scheduledEvents.begin(), end);
146 }
147 return;
148 }
149
150 if ( observable->getObservableType() != Observable::Type::Token ) {
151 return;
152 }
153 auto token = static_cast<const Token*>(observable);
154
155 if ( !token->node ) {
156 if ( token->state == Token::State::CREATED ) {
157 readyProcess(scenario, queue, token);
158 }
159 return;
160 }
161
162 // the instances of a multi-instance activity receive no ready event of their own, being created from the
163 // ready event of the main token, which is never a key of tokenAtMultiInstanceActivity
164 if (
165 token->node->represents<BPMN::Activity>() &&
166 ( token->state == Token::State::ARRIVED || token->state == Token::State::CREATED ) &&
167 !token->owner->systemState->tokenAtMultiInstanceActivity.contains(const_cast<Token*>(token))
168 ) {
169 readyActivity(scenario, queue, token, std::numeric_limits<BPMNOS::number>::lowest());
170 return;
171 }
172
173 if (
174 token->node->represents<BPMN::Task>() &&
175 token->state == Token::State::BUSY &&
176 // send, receive and decision tasks complete through other events
177 !token->node->represents<BPMN::SendTask>() &&
178 !token->node->represents<BPMN::ReceiveTask>() &&
179 !token->node->represents<BPMNOS::Model::DecisionTask>()
180 ) {
181 completeTask(scenario, queue, token, std::numeric_limits<BPMNOS::number>::lowest());
182 }
183}

260std::optional<BPMNOS::number> StaticDataProvider::getValue(const Scenario& scenario, size_t instanceId, const BPMNOS::Model::Attribute* attribute) const {
261 if ( attribute->expression && attribute->expression->type == BPMNOS::Model::Expression::Type::ASSIGN ) {
262 // the value is computed from values of the instance which must not change during the run
263 std::vector<double> variableValues;
264 for ( auto input : attribute->expression->variables ) {
265 auto value = input->isImmutable ? getValue(scenario, instanceId, input) : std::nullopt;
266 if ( !value.has_value() ) {
267 return std::nullopt;
268 }
269 variableValues.push_back((double)value.value());
270 }
271 std::vector<std::vector<double>> collectionValues;
272 for ( auto input : attribute->expression->collections ) {
273 auto collection = input->isImmutable ? getValue(scenario, instanceId, input) : std::nullopt;
274 if ( !collection.has_value() ) {
275 return std::nullopt;
276 }
277 collectionValues.push_back(collectionRegistry[(size_t)collection.value()]);
278 }
279 return BPMNOS::number(attribute->expression->compiled.evaluate(variableValues, collectionValues));
280 }
281 auto& values = getInstanceValues(scenario, instanceId);
282 if ( auto it = values.find(attribute); it != values.end() ) {
283 return it->second;
284 }
285 return std::nullopt;
286}
