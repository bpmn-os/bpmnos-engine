48void StaticDataProvider::readInstances() {
49 CSVReader::Table table = reader->read();
50 if (table.empty()) {
51 throw std::runtime_error("StaticDataProvider: table '" + reader->instanceFileOrString + "' is empty");
52 }
53
54 if (table.size() < 2) {
55 throw std::runtime_error("StaticDataProvider: table must have at least a header and one data row");
56 }
57
58 size_t columnCount = table[0].size();
59 if (columnCount != 3) {
60 throw std::runtime_error("StaticDataProvider: expected 3 columns (INSTANCE_ID, NODE_ID, INITIALIZATION), got " + std::to_string(columnCount));
61 }
62
63 enum { INSTANCE_ID, NODE_ID, INITIALIZATION };
64
65 for (auto& row : table | std::views::drop(1)) {
66 if (row.empty()) {
67 continue;
68 }
69 if (row.size() != 3) {
70 throw std::runtime_error("StaticDataProvider: inconsistent number of cells");
71 }
72
73 // Get instance ID (may be empty for globals)
74 if (!std::holds_alternative<std::string>(row.at(INSTANCE_ID))) {
75 throw std::runtime_error("StaticDataProvider: illegal instance id");
76 }
77 std::string instanceIdentifier = std::get<std::string>(row.at(INSTANCE_ID));
78
79 // Get node ID (may be empty for globals)
80 if (!std::holds_alternative<std::string>(row.at(NODE_ID))) {
81 throw std::runtime_error("StaticDataProvider: illegal node id");
82 }
83 std::string nodeId = std::get<std::string>(row.at(NODE_ID));
84
85 // Get initialization expression
86 if (!std::holds_alternative<std::string>(row.at(INITIALIZATION))) {
87 throw std::runtime_error("StaticDataProvider: illegal initialization");
88 }
89 std::string initializationString = std::get<std::string>(row.at(INITIALIZATION));
90
91 if (instanceIdentifier.empty() && nodeId.empty()) {
92 // Global attribute
93 if (initializationString.empty()) {
94 continue;
95 }
96 evaluateGlobal(initializationString);
97 }
98 else if (instanceIdentifier.empty()) {
99 throw std::runtime_error("StaticDataProvider: instance id required when node id is provided");
100 }
101 else {
102 auto instanceId = (size_t)BPMNOS::to_number(instanceIdentifier, STRING);
103
104 // Find the node
105 BPMN::Node* node = findNode(nodeId);
106
107 // First occurrence of instance must have node = process
108 if (!instances.contains(instanceId)) {
109 // Check that node is a process
110 if (!node->represents<BPMN::Process>()) {
111 throw std::runtime_error("StaticDataProvider: first row for instance '" + instanceIdentifier + "' must reference a process node, got '" + nodeId + "'");
112 }
113 auto process = dynamic_cast<BPMN::Process*>(node);
114 instances[instanceId] = StaticInstanceData({process, instanceId, std::numeric_limits<BPMNOS::number>::max(), {}});
115 }
116
117 // If no initialization, just create instance (already done above)
118 if (initializationString.empty()) {
119 continue;
120 }
121
122 auto& instance = instances[instanceId];
123 auto [attribute, expressionString] = lookupAttribute(node, initializationString);
124
125 BPMNOS::number value = evaluateExpression(instanceId, node, expressionString, attribute->type);
126 instance.data[attribute] = value;
127 parseTimeEvaluatedValues[instanceId][attribute] = value;
128 }
129 }
130
131 for (auto& [id, instance] : instances) {
132 // ensure that default attributes are available
133 ensureDefaultValue(instance, Keyword::Instance, id);
134 ensureDefaultValue(instance, Keyword::Timestamp);
135 // set time of instantiation (ceiled to align with integer time steps)
136 instance.instantiation = BPMNOS::number(std::ceil((double)instance.data.at( attributes[instance.process][Keyword::Timestamp] )));
137
138 if ( earliestInstantiation > instance.instantiation ) {
139 earliestInstantiation = instance.instantiation;
140 }
141 if ( latestInstantiation < instance.instantiation ) {
142 latestInstantiation = instance.instantiation;
143 }
144 }
