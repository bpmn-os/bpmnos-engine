36void StaticDataProvider::readInstances(const std::string& instanceFileOrString, const std::vector<std::string>& columns, const LIMEX::Handle<double>& handle) {
37 InstanceDataReader reader(this->model.get(), instanceFileOrString, columns);
38 // the sizes are declared for every instance, so they are known before any value is read
39 for ( auto& row : reader.rows ) {
40 if ( row.declaresSizes ) {
41 reader.declareSizes(row);
42 }
43 }
44 for ( auto& row : reader.rows ) {
45 if ( row.declaresSizes ) {
46 continue;
47 }
48 if ( !row.node ) {
49 // a global value is given once, for every run
50 if ( reader.assignsObject(row) ) {
51 reader.setObject(row, handle);
52 }
53 else {
54 reader.evaluateGlobal(row.initialization, handle);
55 }
56 }
57 else {
58 readValue(reader, row, handle);
59 }
60 }
61
62 // the global attributes are given their values in declaration order, each the value of its initial
63 // expression if the model assigns it, otherwise the value given, so that an expression reads the global
64 // attributes declared before it
65 globals = BPMNOS::Data(this->model->attributes.size());
66 for ( auto& attribute : this->model->attributes ) {
67 if ( attribute->expression ) {
68 // the global attributes are the first data attributes
69 auto value = attribute->expression->execute(BPMNOS::Status{}, globals);
70 if ( !value.has_value() ) {
71 throw std::runtime_error("StaticDataProvider: failed to evaluate global attribute '" + attribute->id + "'");
72 }
73 globals.attributes[attribute->index] = value.value();
74 }
75 else if ( auto it = reader.globals.find(attribute.get()); it != reader.globals.end() ) {
76 globals.attributes[attribute->index] = it->second;
77 }
78 }
79
80 // the objects: the default of each, from the model's initial value or with undefined values in the sizes
81 // declared, an open dimension having no elements, and the global objects
82 auto conform = [&reader](const BPMNOS::Model::Attribute* object, size_t index, const std::string& owner) {
83 try {
84 return reader.getSchema(object).conform( objectRegistry[index] );
85 }
86 catch ( const std::exception& error ) {
87 throw std::runtime_error("StaticDataProvider: illegal value of object '" + object->id + "' " + owner + ".\n" + error.what());
88 }
89 };
90 for ( auto object : objectsOf() ) {
91 auto schema = reader.getSchema(object);
92 if ( object->initialObject.has_value() ) {
93 defaultObjects[object] = conform(object, object->initialObject.value(), "in the model");
94 }
95 else {
96 // an open dimension has no elements until an assignment gives it its length
97 defaultObjects[object] = schema.undefinedObject();
98 }
99 }
100 // the default content with the sizes declared at the node throwing it or, for a signal, by its name, every node of
101 // a signal sharing one
102 for ( auto& [name, schema] : reader.signalSchemas ) {
103 defaultContents[name] = schema.undefinedObject();
104 }
105 for ( auto& process : this->model->processes ) {
106 for ( auto node : process->find_all([](const BPMN::Node* node) { return node->extensionElements && node->extensionElements->represents<const BPMNOS::Model::ExtensionElements>() && node->extensionElements->as<const BPMNOS::Model::ExtensionElements>()->content; }) ) {
107 auto extensionElements = node->extensionElements->as<const BPMNOS::Model::ExtensionElements>();
108 auto content = extensionElements->content.get();
109 if ( auto signalDefinition = extensionElements->signalDefinition.get() ) {
110 defaultObjects[content] = defaultContents.at(signalDefinition->name);
111 }
112 else if ( reader.schemas.contains(content) ) {
113 defaultObjects[content] = reader.getSchema(content).undefinedObject();
114 }
115 }
116 }
117 for ( auto& object : this->model->objects ) {
118 if ( auto it = reader.globalObjects.find(object.get()); it != reader.globalObjects.end() ) {
119 globals.objects.push_back( it->second );
120 }
121 else {
122 globals.objects.push_back( getDefaultObject(object.get()) );
123 }
124 }
125
126 // the instances are kept in the order of their first rows
127 for ( auto& row : reader.rows ) {
128 if ( !row.node || row.declaresSizes || instancePositions.contains(row.instanceId) ) {
129 continue;
130 }
131 auto instanceId = row.instanceId;
132 auto process = reader.processes.at(instanceId);
133 auto values = reader.values.contains(instanceId) ? reader.values.at(instanceId) : std::unordered_map<const BPMNOS::Model::Attribute*, BPMNOS::number>{};
134 reader.addDefaultValues(instanceId, values);
135 auto timestampAttribute = process->extensionElements->as<BPMNOS::Model::ExtensionElements>()->attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp].get();
136 // instances are instantiated at integral times
137 auto instantiationTime = BPMNOS::number(std::ceil((double)values.at(timestampAttribute)));
138 // the objects the instance data gives, every other object of the instance taking its default
139 auto objects = reader.objects.contains(instanceId) ? reader.objects.at(instanceId) : std::unordered_map<const BPMNOS::Model::Attribute*, std::shared_ptr<const BPMNOS::Object>>{};
140 instancePositions[instanceId] = instances.size();
141 instances.push_back(Instance{instanceId, process, std::move(values), instantiationTime, std::move(objects)});
142 }

179void StaticDataProvider::notice(const Observable* observable, Execution::Scenario& executionScenario, EventQueue& queue) const {
180 auto& scenario = static_cast<Scenario&>(executionScenario);
181
182 if ( observable->getObservableType() == Observable::Type::SystemState ) {
183 // the state of a run not yet begun holds no instance, an installed state every instance known up to its
184 // time; the instances not yet held are instantiated when they become known, and the events the tokens
185 // await are determined anew
186 auto systemState = static_cast<const SystemState*>(observable);
187 scenario.time = systemState->getTime();
188 bool runBegun = ( systemState->getTime() != std::numeric_limits<BPMNOS::number>::lowest() );
189 scenario.scheduledEvents.clear();
190 for ( auto& instance : instances ) {
191 if ( !runBegun || getKnownTime(scenario, instance.id) > systemState->getTime() ) {
192 auto event = instantiate(scenario, instance);
193 schedule(scenario, queue, getKnownTime(scenario, instance.id), systemState->getTime(), [event](Scenario&) { return event; });
194 }
195 }
196 for ( auto& [token_ptr] : systemState->tokensAwaitingReadyEvent ) {
197 if ( auto token = token_ptr.lock() ) {
198 if ( token->node->represents<BPMN::FlowNode>() ) {
199 readyActivity(scenario, queue, token.get(), systemState->getTime());
200 }
201 else {
202 readyProcess(scenario, queue, token.get());
203 }
204 }
205 }
206 for ( auto element : systemState->tokensAwaitingCompletionEvent ) {
207 if ( auto token = std::get<1>(element).lock() ) {
208 completeTask(scenario, queue, token.get(), systemState->getTime());
209 }
210 }
211 return;
212 }
213
214 if ( observable->getObservableType() == Observable::Type::Event ) {
215 if ( auto clockTickEvent = static_cast<const Event*>(observable)->is<ClockTickEvent>() ) {
216 scenario.time = clockTickEvent->time;
217 // the events due by the time the clock advances to are released in the order of their times
218 auto end = scenario.scheduledEvents.upper_bound(clockTickEvent->time);
219 for ( auto it = scenario.scheduledEvents.begin(); it != end; ++it ) {
220 if ( auto event = it->second(scenario) ) {
221 queue.push_back(std::move(event));
222 }
223 }
224 scenario.scheduledEvents.erase(scenario.scheduledEvents.begin(), end);
225 }
226 return;
227 }
228
229 if ( observable->getObservableType() != Observable::Type::Token ) {
230 return;
231 }
232 auto token = static_cast<const Token*>(observable);
233
234 if ( token->node->represents<BPMN::Process>() ) {
235 if ( token->state == Token::State::CREATED ) {
236 readyProcess(scenario, queue, token);
237 }
238 return;
239 }
240
241 // the instances of a multi-instance activity receive no ready event of their own, being created from the
242 // ready event of the main token, which is never a key of tokenAtMultiInstanceActivity
243 if (
244 token->node->represents<BPMN::Activity>() &&
245 ( token->state == Token::State::ARRIVED || token->state == Token::State::CREATED ) &&
246 !token->owner->systemState->tokenAtMultiInstanceActivity.contains(const_cast<Token*>(token))
247 ) {
248 readyActivity(scenario, queue, token, std::numeric_limits<BPMNOS::number>::lowest());
249 return;
250 }
251
252 if (
253 token->node->represents<BPMN::Task>() &&
254 token->state == Token::State::BUSY &&
255 // send, receive and decision tasks complete through other events
256 !token->node->represents<BPMN::SendTask>() &&
257 !token->node->represents<BPMN::ReceiveTask>() &&
258 !token->node->represents<BPMNOS::Model::DecisionTask>()
259 ) {
260 completeTask(scenario, queue, token, std::numeric_limits<BPMNOS::number>::lowest());
261 }
262}

315std::shared_ptr<Event> StaticDataProvider::instantiate(const Scenario& scenario, const Instance& instance) const {
316 // the instantiation gives the instance its identity and timestamp, its other values being given when its
317 // process becomes ready; the objects are laid out with their defaults, so that every object of the instance has
318 // its slot
319 auto extensionElements = instance.process->extensionElements->as<BPMNOS::Model::ExtensionElements>();
320 BPMNOS::Status status( extensionElements->attributes.size() );
321 auto timestampAttribute = extensionElements->attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp].get();
322 status.attributes[BPMNOS::Model::ExtensionElements::Index::Timestamp] = getInstanceValues(scenario, instance.id).at(timestampAttribute);
323 status.objects.reserve(extensionElements->statusObjects.size());
324 for ( auto& object : extensionElements->statusObjects ) {
325 status.objects.push_back( getDefaultObject(object.get()) );
326 }
327 BPMNOS::Data data( extensionElements->data.size() );
328 data.attributes[BPMNOS::Model::ExtensionElements::Position::Instance] = BPMNOS::number(instance.id);
329 data.objects.reserve(extensionElements->dataObjects.size());
330 for ( auto& object : extensionElements->dataObjects ) {
331 data.objects.push_back( getDefaultObject(object.get()) );
332 }
333 return std::make_shared<InstantiationEvent>(instance.process, std::move(status), std::move(data));
334}

337 auto instanceId = (size_t)token->getInstanceId();
338 // the event is built when it becomes due, so that the initial values read the globals of that time
339 std::weak_ptr<const Token> token_ptr = token->weak_from_this();

350void StaticDataProvider::readyActivity(Scenario& scenario, EventQueue& queue, const Token* token, BPMNOS::number earliest) const {

352 // the timestamp a ready value gives is determined when the token arrives, since the due time depends on it

354 // the event is built when it becomes due, so that the initial values read the data and globals of that time

385std::vector<const BPMNOS::Model::Attribute*> StaticDataProvider::objectsOf(const BPMN::Process* process) const {
386 std::vector<const BPMNOS::Model::Attribute*> objects;
387 auto collect = [&objects](const BPMN::Node* node) {
388 // not every node owns extension elements
389 if ( !node->extensionElements ) {
390 return;
391 }
392 if ( auto extensionElements = node->extensionElements->represents<const BPMNOS::Model::ExtensionElements>() ) {
393 for ( auto list : { &extensionElements->statusObjects, &extensionElements->dataObjects } ) {
394 for ( auto& object : *list ) {
395 objects.push_back(object.get());
396 }
397 }
398 }
399 };
400 for ( auto& candidate : this->model->processes ) {
401 if ( process && candidate.get() != process ) {
402 continue;
403 }
404 collect(candidate.get());
405 for ( auto node : candidate->find_all([](const BPMN::Node*) { return true; }) ) {
406 collect(node);
407 }
408 }
409 if ( !process ) {
410 for ( auto& object : this->model->objects ) {
411 objects.push_back(object.get());
412 }
413 }
414 return objects;
