10/// Returns the hash of an object: of its layout, which is short, and of its values, an undefined value being
11/// distinguished from every number.
12size_t hashOf(const Object& object) {
13 size_t hash = std::hash<std::string>{}( object.layout->stringify() );
14 auto combine = [&hash](size_t value) {
15 hash ^= value + 0x9e3779b97f4a7c15ULL + (hash << 6) + (hash >> 2);
16 };
17 for ( auto& value : object.values ) {
18 combine( value.has_value() ? std::hash<double>{}( (double)value.value() ) : 0x5bd1e995ULL );
19 }
20 return hash;
21}

31const std::shared_ptr<const Object>& ObjectRegistry::operator[](size_t i) const {

40std::optional<size_t> ObjectRegistry::find(const Object& object, size_t hash) const {
41 auto [first, last] = index.equal_range(hash);
42 for ( auto it = first; it != last; ++it ) {
43 auto& candidate = *blocks[it->second / blockSize].load(std::memory_order_acquire)[it->second % blockSize];
44 if ( *candidate.layout == *object.layout && candidate.values == object.values ) {
45 return it->second;
46 }
47 }
48 return std::nullopt;

51size_t ObjectRegistry::operator()(Object object) {
52 auto hash = hashOf(object);
53
54 std::shared_lock read_lock(registryMutex);
55 if ( auto found = find(object, hash) ) {

62 // registered by another thread in the meantime
63 return found.value();
64 }
65
66 size_t i = published.load(std::memory_order_relaxed);
67 if ( i / blockSize >= maxBlocks ) {
68 throw std::runtime_error("ObjectRegistry: too many objects");
69 }
70 auto block = blocks[i / blockSize].load(std::memory_order_relaxed);
71 if ( !block ) {
72 block = new std::shared_ptr<const Object>[blockSize];
73 blocks[i / blockSize].store(block, std::memory_order_release);
74 }
75 block[i % blockSize] = std::make_shared<const Object>(std::move(object));
76 index.emplace(hash, i);
77 published.store(i + 1, std::memory_order_release);
