44 /**
45 * @brief Runs a scenario from the beginning.
46 *
47 * Creates a fresh system state at the given start time and executes until no tokens remain,
48 * no new instantiations are pending, or the end time is exceeded. The start time must not be later
49 * than the scenario's earliest instantiation time, since an instance is instantiated at the instant
50 * its instantiation time is reached and a later start would leave every earlier instance uncreated.
51 *
52 * @param scenario The scenario to execute
53 * @param startTime Time the run begins at
54 * @param endTime Last time to process (engine stops when time >= endTime)
55 */
56 void run(const BPMNOS::Model::Scenario* scenario, BPMNOS::number startTime = 0, BPMNOS::number endTime = std::numeric_limits<BPMNOS::number>::max());
57
58 /**
59 * @brief Initializes the engine with a fresh system state and advances time to the run's first instant.
60 *
61 * Does not process any further event; call run, resume, or step afterwards. The opening clock tick is
62 * announced like any other, so the record stream states the time the run begins at and deferred data is
63 * disclosed for that instant.
64 *
65 * @param scenario The scenario to execute
66 * @param startTime Time the run begins at; must not be later than the scenario's earliest instantiation
67 * time, since an instance is instantiated at the instant its instantiation time is reached and a
68 * later start would leave every earlier instance uncreated

70 void initialize(const BPMNOS::Model::Scenario* scenario, BPMNOS::number startTime = 0);

72 /**
