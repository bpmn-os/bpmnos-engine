16/**
17 * @brief Registry of constant objects, by which a literal is represented as a number.
18 *
19 * Every literal, `[ ... ]` and `{ name := ... }` alike, is registered here as a constant object, and the index
20 * under which it is registered is the number standing for it wherever a value is a number: in an expression, in
21 * a lookup table, and in an attribute holding it. Equal literals are registered once, so that two of them have
22 * the same index; two objects are equal if they have the same layout and the same values. An object is found
23 * among those registered by a hash of its layout and its values, computed without converting a value, and an
24 * object with an equal hash is compared value by value, so that a collision never merges different objects.
25 *
26 * A registered object never moves and never changes: the entries are held in blocks of fixed size, each
27 * allocated once, and the number of entries published so far is held in an atomic counter, which a
28 * registration raises only after writing its entry. A reference to a registered object therefore stays valid
29 * however many objects are registered later, and reading an entry takes no lock; only a registration takes the
30 * mutex, so that concurrent registrations of the same object yield one index.
31 */
32struct ObjectRegistry {
33 ObjectRegistry() = default;
34 ~ObjectRegistry();
35
36 /// Returns the registered object with the given index, without taking a lock.
37 const std::shared_ptr<const Object>& operator[](size_t i) const;
38 /// Registers the object, unless an equal one is registered already, and returns its index.
39 size_t operator()(Object object);
40 size_t size() const;
41
42 ObjectRegistry(const ObjectRegistry&) = delete;
43 ObjectRegistry& operator=(const ObjectRegistry&) = delete;
44private:
45 static constexpr size_t blockSize = 4096; ///< The number of entries per block
46 static constexpr size_t maxBlocks = 16384; ///< The number of blocks, bounding the registry at 2^26 entries
47 /// The blocks of entries, each allocated by the registration filling its first entry and never moved.
48 std::array< std::atomic< std::shared_ptr<const Object>* >, maxBlocks > blocks{};
49 /// The number of published entries, raised by a registration after its entry is written.
50 std::atomic<size_t> published{0};
51 /// The indices of the registered objects by the hash of their layout and values.
52 std::unordered_multimap<size_t, size_t> index;
53 /// Returns the index of a registered object equal to the given one with the given hash, if there is one.
54 std::optional<size_t> find(const Object& object, size_t hash) const;
55 mutable std::shared_mutex registryMutex; ///< Guards the index and serialises registrations
56};
