14/**
15 * @brief The shape of an attribute, as stated by its `type`.
16 *
17 * A type is a base followed by its dimensions, `type := base dimension*`. The base is a scalar type, one of
18 * `boolean`, `integer`, `decimal` and `string`, or an object `{ name: type, ... }` whose fields
19 * have types of their own. A dimension is written `[n]` if its size is fixed and `[]` if it is open, and the
20 * dimensions are read from left to right in the order of the indices: `boolean[3][4]` has 3 elements, each a
21 * `boolean[4]`, accessed as `x[i][j]` with `i <= 3` and `j <= 4`. A schema without dimensions whose base is a
22 * scalar type describes a scalar attribute; every other schema describes an object.
23 */
24struct Schema {
25 std::optional<ValueType> scalar; ///< The scalar type of the base, or nullopt if the base is an object
26 std::vector< std::pair<std::string, Schema> > fields; ///< The fields of an object base, in declaration order
27 std::vector< std::optional<size_t> > dimensions; ///< The sizes of the dimensions in index order, nullopt if open
28
29 /// @brief Parses the text of a type, throwing an error describing the first violation of the grammar.
30 static Schema parse(const std::string& text);
31 /// @brief Returns true if the schema describes a scalar attribute.
32 bool isScalar() const { return scalar.has_value() && dimensions.empty(); }
33 /// @brief Returns the text of the type in its canonical form.
34 std::string stringify() const;
35 /// @brief Returns true if the schemas agree: they have the same scalar type or the same set of field names, each
36 /// field agreeing in turn, regardless of the order of the fields, and the same dimensions with the same sizes.
37 bool agrees(const Schema& other) const;
38
39 /**
40 * @brief Fixes the sizes a size declaration states, given as the text following the name of the attribute.
41 *
42 * The text is a path of dimensions and fields, e.g. `[6].flags[3]`, in which every number in brackets is the
43 * size of the next dimension of the base or field it follows and `[]` leaves that dimension as it is. A size
44 * contradicting a size fixed before, more dimensions than the type has, and an unknown field are errors.
45 */
46 void declareSizes(const std::string& steps);
47
48 /// @brief Returns true if every dimension, also of the fields, has a fixed size.
49 bool isFixed() const;
50
51 /**
52 * @brief Returns an object of this schema with every value undefined, an open dimension having no elements.
53 */
54 std::shared_ptr<const BPMNOS::Object> undefinedObject() const;
55
56 /**
57 * @brief Returns an object of this schema holding the values of the given constant object.
58 *
59 * The constant object must have as many dimensions as the schema at the base and at every field, its
60 * scalar values must be convertible to the types of the schema, and its fields must be fields of the schema.
61 * A dimension of fixed size is padded with undefined values and must not be exceeded, an open dimension takes
62 * the length of the constant object, and a field it lacks is undefined. If the constant object has exactly the
63 * layout of the schema, it is returned itself, so that it is shared.
64 */
65 std::shared_ptr<const BPMNOS::Object> conform(const std::shared_ptr<const BPMNOS::Object>& constant) const;
66
67 /**
68 * @brief Returns the schema an object of the given layout imposes on a value assigned to it.
69 *
70 * A value assigned to the whole object keeps the sizes of the fixed dimensions and takes those of the open
71 * ones. A value assigned to an element keeps every size, except a size zero of an open dimension, which the
72 * value determines.
73 */
74 static Schema of(const BPMNOS::Object::Layout& layout, bool element);
75
76 /**
77 * @brief Returns an object of this schema holding the values of the given JSON text.
78 *
79 * An array is a dimension, whose elements have equal lengths at an open dimension and at most the declared
80 * length at a fixed one, an object is a value with fields, whose keys must be fields of the schema, and a number,
81 * truth value or string is a value of the type the schema states, `null` being undefined. A fixed dimension is
82 * padded with undefined values, an open dimension takes the length of the JSON array, and a field the JSON lacks
83 * is undefined.
84 */
85 std::shared_ptr<const BPMNOS::Object> fromJSON(const std::string& text) const;
86};
