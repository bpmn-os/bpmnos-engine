17class SchemaParser {
18public:
19 explicit SchemaParser(const std::string& text) : text(text), position(0) {}

21 Schema parseType() {
22 Schema schema;
23 skipWhitespace();
24 if ( peek() == '{' ) {
25 parseObject(schema);
26 }
27 else {
28 schema.scalar = parseScalar();
29 }
30 skipWhitespace();
31 while ( peek() == '[' ) {
32 ++position;
33 skipWhitespace();
34 if ( peek() == ']' ) {
35 schema.dimensions.push_back(std::nullopt);
36 }
37 else {
38 schema.dimensions.push_back(parseSize());
39 }
40 skipWhitespace();
41 expect(']');
42 skipWhitespace();
43 }
44 return schema;
45 }

47 void expectEnd() {
48 skipWhitespace();
49 if ( position != text.size() ) {

51 }

90 BPMNOS::ValueType parseScalar() {
91 std::string name = parseName();
92 if ( name == "boolean" ) return BPMNOS::ValueType::BOOLEAN;
93 if ( name == "integer" ) return BPMNOS::ValueType::INTEGER;
94 if ( name == "decimal" ) return BPMNOS::ValueType::DECIMAL;
95 if ( name == "string" ) return BPMNOS::ValueType::STRING;

99 size_t parseSize() {
100 size_t start = position;
101 while ( position < text.size() && std::isdigit((unsigned char)text[position]) ) {
102 ++position;
103 }
104 if ( position == start ) {
105 fail("expected a size at position " + std::to_string(position + 1));
106 }
107 size_t size = std::stoul(text.substr(start, position - start));
108 if ( size == 0 ) {
109 fail("a size must be positive");
110 }

114 void parseObject(Schema& schema) {
115 expect('{');
116 skipWhitespace();
117 if ( peek() == '}' ) {
118 fail("an object requires at least one field");
119 }
120 while ( true ) {
121 skipWhitespace();
122 std::string name = parseName();
123 for ( auto& [field, _] : schema.fields ) {
124 if ( field == name ) {
125 fail("duplicate field '" + name + "'");
126 }
127 }
128 skipWhitespace();
129 expect(':');
130 schema.fields.emplace_back(name, parseType());
131 skipWhitespace();
132 if ( peek() == ',' ) {
133 ++position;
134 continue;
135 }
136 expect('}');
137 return;
138 }
139 }

174void Schema::declareSizes(const std::string& steps) {
175 Schema* node = this;
176 size_t dimension = 0;
177 size_t position = 0;
178 auto fail = [&steps](const std::string& reason) {
179 throw std::runtime_error("Schema: illegal size declaration '" + steps + "': " + reason);
180 };
181 auto skipWhitespace = [&]() {
182 while ( position < steps.size() && std::isspace((unsigned char)steps[position]) ) {
183 ++position;
184 }
185 };
186 while ( true ) {
187 skipWhitespace();
188 if ( position == steps.size() ) {
189 return;
190 }
191 if ( steps[position] == '[' ) {
192 ++position;
193 skipWhitespace();
194 size_t start = position;
195 while ( position < steps.size() && std::isdigit((unsigned char)steps[position]) ) {
196 ++position;
197 }
198 std::optional<size_t> size;
199 if ( position > start ) {
200 size = std::stoul(steps.substr(start, position - start));
201 if ( size.value() == 0 ) {
202 fail("a size must be positive");
203 }
204 }
205 skipWhitespace();
206 if ( position == steps.size() || steps[position] != ']' ) {
207 fail("expected ']'");
208 }
209 ++position;
210 if ( dimension >= node->dimensions.size() ) {
211 fail("more dimensions than the type has");
212 }
213 auto& declared = node->dimensions[dimension];
214 if ( size.has_value() ) {
215 if ( declared.has_value() && declared.value() != size.value() ) {
216 fail("size " + std::to_string(size.value()) + " contradicts size " + std::to_string(declared.value()));
217 }
218 declared = size;
219 }
220 ++dimension;
221 }
222 else if ( steps[position] == '.' ) {
223 ++position;
224 size_t start = position;
225 while ( position < steps.size() && ( std::isalnum((unsigned char)steps[position]) || steps[position] == '_' ) ) {
226 ++position;
227 }
228 std::string name = steps.substr(start, position - start);
229 auto field = std::find_if(node->fields.begin(), node->fields.end(), [&name](auto& candidate) { return candidate.first == name; });
230 if ( field == node->fields.end() ) {
231 fail("unknown field '" + name + "'");
232 }
233 node = &field->second;
234 dimension = 0;
235 }
236 else {
237 fail("unexpected '" + steps.substr(position) + "'");
238 }
239 }

258/// Returns the layout of a schema whose dimensions all have sizes, a dimension being fixed if it has a size in
259/// the schema declared, of which the schema is a resolution.
260BPMNOS::Object::Layout layoutOf(const Schema& schema, const Schema& declared) {
261 BPMNOS::Object::Layout layout;
262 layout.scalar = schema.scalar;
263 for ( size_t k = 0; k < schema.fields.size(); k++ ) {
264 layout.fields.push_back( BPMNOS::Object::Layout::Field{ schema.fields[k].first, 0, layoutOf(schema.fields[k].second, declared.fields[k].second) } );
265 }
266 for ( size_t d = 0; d < schema.dimensions.size(); d++ ) {
267 assert( schema.dimensions[d].has_value() );
268 layout.dimensions.push_back( schema.dimensions[d].value() );
269 layout.fixed.push_back( declared.dimensions[d].has_value() );
270 }
271 layout.arrange();
272 return layout;
273}

275/// Fixes the open dimensions of a schema by the layout of a constant object, checking that it fits.
276void resolve(Schema& schema, const BPMNOS::Object::Layout& constant, const std::string& path) {
277 if ( constant.dimensions.size() != schema.dimensions.size() ) {
278 throw std::runtime_error("Schema: '" + path + "' has " + std::to_string(constant.dimensions.size()) + " dimensions instead of " + std::to_string(schema.dimensions.size()));
279 }
280 for ( size_t i = 0; i < schema.dimensions.size(); i++ ) {
281 if ( !schema.dimensions[i].has_value() ) {
282 schema.dimensions[i] = constant.dimensions[i];
283 }
284 else if ( constant.dimensions[i] > schema.dimensions[i].value() ) {
285 throw std::runtime_error("Schema: '" + path + "' has " + std::to_string(constant.dimensions[i]) + " elements where " + std::to_string(schema.dimensions[i].value()) + " are declared");
286 }
287 }
288 if ( schema.scalar.has_value() ) {
289 if ( !constant.scalar.has_value() ) {
290 throw std::runtime_error("Schema: '" + path + "' has fields where a value is expected");
291 }
292 auto from = constant.scalar.value();
293 auto to = schema.scalar.value();
294 bool numeric = ( from == BPMNOS::ValueType::DECIMAL || from == BPMNOS::ValueType::BOOLEAN );
295 bool fits = ( from == to ) || ( numeric && ( to == BPMNOS::ValueType::DECIMAL || to == BPMNOS::ValueType::INTEGER || to == BPMNOS::ValueType::BOOLEAN ) );
296 if ( !fits ) {
297 throw std::runtime_error("Schema: '" + path + "' cannot be converted to the type of the attribute");
298 }
299 return;
300 }
301 if ( constant.scalar.has_value() ) {
302 throw std::runtime_error("Schema: '" + path + "' is a value where fields are expected");
303 }
304 for ( auto& field : constant.fields ) {
305 auto it = std::find_if(schema.fields.begin(), schema.fields.end(), [&field](auto& candidate) { return candidate.first == field.name; });
306 if ( it == schema.fields.end() ) {
307 throw std::runtime_error("Schema: '" + path + "' has the unknown field '" + field.name + "'");
308 }
309 resolve(it->second, field.layout, path + "." + field.name);
310 }

341/// Copies the values of a constant object of the given layout into an object of the target layout.
342void copyValues(const BPMNOS::Object::Layout& target, const BPMNOS::Object::Layout& source, const std::vector<BPMNOS::Value>& from, size_t fromOffset, std::vector<BPMNOS::Value>& to, size_t toOffset) {
343 std::function<void(size_t, size_t, size_t)> copyDimension = [&](size_t dimension, size_t sourceOffset, size_t targetOffset) {
344 if ( dimension == source.dimensions.size() ) {
345 if ( target.scalar.has_value() ) {
346 to[targetOffset] = convert(from[sourceOffset], target.scalar.value());
347 return;
348 }
349 for ( auto& field : source.fields ) {
350 auto it = std::find_if(target.fields.begin(), target.fields.end(), [&field](auto& candidate) { return candidate.name == field.name; });
351 copyValues(it->layout, field.layout, from, sourceOffset + field.offset, to, targetOffset + it->offset);
352 }
353 return;
354 }
355 size_t sourceSpan = source.stride;
356 size_t targetSpan = target.stride;
357 for ( size_t d = dimension + 1; d < source.dimensions.size(); d++ ) {
358 sourceSpan *= source.dimensions[d];
359 targetSpan *= target.dimensions[d];
360 }
361 for ( size_t i = 0; i < source.dimensions[dimension]; i++ ) {
362 copyDimension(dimension + 1, sourceOffset + i * sourceSpan, targetOffset + i * targetSpan);
363 }
364 };
365 copyDimension(0, fromOffset, toOffset);
366}

381std::shared_ptr<const BPMNOS::Object> Schema::conform(const std::shared_ptr<const BPMNOS::Object>& constant) const {
382 Schema resolved = *this;
383 resolve(resolved, *constant->layout, BPMNOS::to_string(*constant));
384 close(resolved);
385 auto layout = layoutOf(resolved, *this);
386 if ( layout == *constant->layout ) {
387 // the object holds exactly what the attribute declares, with the same dimensions fixed, and is shared
388 return constant;
389 }

391 auto object = std::make_shared<BPMNOS::Object>();
392 object->layout = std::make_shared<const BPMNOS::Object::Layout>( std::move(layout) );
393 object->values.resize( object->layout->size() );
394 copyValues(*object->layout, *constant->layout, constant->values, 0, object->values, 0);

398Schema Schema::of(const BPMNOS::Object::Layout& layout, bool element) {
399 Schema schema;
400 schema.scalar = layout.scalar;
401 for ( auto& field : layout.fields ) {
402 schema.fields.emplace_back( field.name, of(field.layout, element) );
403 }
404 for ( size_t d = 0; d < layout.dimensions.size(); d++ ) {
405 bool fixed = d < layout.fixed.size() && layout.fixed[d];
406 // an element keeps every length but one still undetermined, whereas an object replaced keeps only the
407 // lengths that are fixed
408 bool keep = element ? ( fixed || layout.dimensions[d] > 0 ) : fixed;
409 schema.dimensions.push_back( keep ? std::optional<size_t>(layout.dimensions[d]) : std::nullopt );
410 }
411 return schema;
412}

418/// Fixes the open dimensions of a schema by the JSON values standing where it applies, every array at an open
419/// dimension having the same length.
420void resolveJson(Schema& schema, const std::vector<const Json*>& values, const std::string& path) {
421 std::vector<const Json*> level = values;
422 for ( auto& dimension : schema.dimensions ) {
423 std::optional<size_t> length;
424 std::vector<const Json*> next;
425 for ( auto value : level ) {
426 if ( value->is_null() ) {
427 continue;
428 }
429 if ( !value->is_array() ) {
430 throw std::runtime_error("Schema: '" + path + "' holds '" + value->dump() + "' where an array is expected");
431 }
432 if ( dimension.has_value() ) {
433 if ( value->size() > dimension.value() ) {
434 throw std::runtime_error("Schema: '" + path + "' has " + std::to_string(value->size()) + " elements where " + std::to_string(dimension.value()) + " are declared");
435 }
436 }
437 else if ( length.has_value() && value->size() != length.value() ) {
438 throw std::runtime_error("Schema: '" + path + "' holds arrays of different lengths");
439 }
440 length = value->size();
441 for ( auto& element : *value ) {
442 next.push_back(&element);
443 }
444 }
445 if ( !dimension.has_value() ) {
446 dimension = length.value_or(0);
447 }
448 level = std::move(next);
449 }
450 if ( schema.scalar.has_value() ) {
451 return;
452 }
453 for ( auto value : level ) {
454 if ( value->is_null() ) {
455 continue;
456 }
457 if ( !value->is_object() ) {
458 throw std::runtime_error("Schema: '" + path + "' holds '" + value->dump() + "' where fields are expected");
459 }
460 for ( auto& [key, _] : value->items() ) {
461 if ( std::ranges::none_of(schema.fields, [&key](auto& field) { return field.first == key; }) ) {
462 throw std::runtime_error("Schema: '" + path + "' has the unknown field '" + key + "'");
463 }
464 }
465 }
466 for ( auto& [name, field] : schema.fields ) {
467 std::vector<const Json*> nested;
468 for ( auto value : level ) {
469 if ( !value->is_null() ) {
470 if ( auto it = value->find(name); it != value->end() ) {
471 nested.push_back(&*it);
472 }
473 }
474 }
475 resolveJson(field, nested, path + "." + name);
476 }

479/// Returns the value a JSON value states for a scalar of the given type.
480BPMNOS::Value jsonValue(const Json& value, BPMNOS::ValueType type, const std::string& path) {
481 if ( value.is_null() ) {
482 return std::nullopt;
483 }
484 switch ( type ) {
485 case BPMNOS::ValueType::BOOLEAN:
486 if ( value.is_boolean() ) {
487 return BPMNOS::number( value.get<bool>() ? 1 : 0 );
488 }
489 break;
490 case BPMNOS::ValueType::INTEGER:
491 if ( value.is_number() ) {
492 return BPMNOS::number( (long)value.get<double>() );
493 }
494 break;
495 case BPMNOS::ValueType::DECIMAL:
496 if ( value.is_number() ) {

509/// Writes the values a JSON value states into an object of the given layout.
510void fillJson(const BPMNOS::Object::Layout& layout, const Json& value, std::vector<BPMNOS::Value>& values, size_t offset, size_t dimension, const std::string& path) {
511 if ( value.is_null() ) {
512 return;
513 }
514 if ( dimension < layout.dimensions.size() ) {
515 size_t span = layout.stride;
516 for ( size_t d = dimension + 1; d < layout.dimensions.size(); d++ ) {
517 span *= layout.dimensions[d];
518 }
519 size_t i = 0;
520 for ( auto& element : value ) {
521 fillJson(layout, element, values, offset + i * span, dimension + 1, path);
522 ++i;
523 }
524 return;
525 }
526 if ( layout.scalar.has_value() ) {
527 values[offset] = jsonValue(value, layout.scalar.value(), path);
528 return;
529 }
530 for ( auto& field : layout.fields ) {
531 if ( auto it = value.find(field.name); it != value.end() ) {
532 fillJson(field.layout, *it, values, offset + field.offset, 0, path + "." + field.name);
533 }
534 }
