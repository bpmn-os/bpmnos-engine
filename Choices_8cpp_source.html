38void Choices::notice(const Observable* observable) {
39 if ( observable->getObservableType() == Observable::Type::ChoiceRequest ) {
40 assert( dynamic_cast<const DecisionRequest*>(observable) );
41 auto request = static_cast<const DecisionRequest*>(observable);
42 queueSearch( request->token->weak_from_this(), request->weak_from_this() );
43 }
44 else if ( observable->getObservableType() == Observable::Type::SystemState ) {
45 clear(); // start from a clean cache, then rebuild from the installed state
46 systemState = static_cast<const SystemState*>(observable);
47 // search the choice requests a freshly installed (e.g. resumed) state lists as pending
48 for ( auto& [ token_ptr, request_ptr ] : systemState->pendingChoiceDecisions ) {
49 queueSearch( token_ptr, request_ptr );
50 }
51 }
52 else {
53 if ( observable->getObservableType() == Observable::Type::DataUpdate ) {
54 // a request whose search yielded no alternative holds no decision a data update could invalidate, and is
55 // searched again when the update touches an attribute its choices depend on, for the instance of its token or,
56 // for a global attribute, for every instance
57 auto update = static_cast<const DataUpdate*>(observable);
58 for ( auto it = requestsWithoutAlternatives.begin(); it != requestsWithoutAlternatives.end(); ) {
59 auto token = std::get<0>(*it).lock();
60 assert( token );
61 auto& dependencies = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->choiceDependencies;
62 bool concerned = std::ranges::any_of(update->attributes, [&](const BPMNOS::Model::Attribute* attribute) {
63 return dependencies.contains(attribute) && ( attribute->isGlobal || token->owner->root->instance.value() == update->instanceId );
64 });
65 if ( concerned ) {
66 queueSearch( std::get<0>(*it), std::get<1>(*it) );
67 it = requestsWithoutAlternatives.erase(it);
68 }
69 else {
70 ++it;
71 }
72 }
73 }
74 CachedCandidates::notice(observable);
75 }

84void Choices::evaluateCandidates() {
85 // a decision invalidated by a data update has its request searched anew rather than being evaluated alone
86 evaluateDecisions(
87 [this]( std::weak_ptr<const Token> token_ptr, std::weak_ptr<const DecisionRequest> request_ptr, [[maybe_unused]] std::shared_ptr<Decision> decision ) -> std::shared_ptr<Event> {
88 queueSearch( token_ptr, request_ptr );
89 return nullptr;
90 }
91 );
92 // the requests are searched in the order in which they arose, a request of a deferred token once its deferral ends
93 for ( auto it = requestsToSearch.begin(); it != requestsToSearch.end(); ) {
94 if ( deferrals ) {
95 if ( auto token = std::get<0>(*it).lock(); token && deferrals->isDeferred(token.get()) ) {
96 ++it;
97 continue;
98 }
99 }
100 auto request = std::get<1>(*it).lock();
101 assert( request );
102 search(request);
103 it = requestsToSearch.erase(it);
104 }

145std::shared_ptr<Decision> Choices::determineBestChoices(std::shared_ptr<const DecisionRequest> request) {
146 auto token = request->token;
147 assert( token->node->represents<BPMN::FlowNode>() );
148 assert( token->node->represents<BPMNOS::Model::DecisionTask>() );
149 assert( token->node->extensionElements->represents<BPMNOS::Model::ExtensionElements>() );
150 auto extensionElements = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
151 assert( extensionElements->choices.size() );
152
153 token_ptr = token->weak_from_this();
154 request_ptr = request->weak_from_this();
155
156 if ( extensionElements->choices.size() > 1 ) {
157 return bestEnumeratedChoices(request);
158 }
159
160 auto& choice = extensionElements->choices[0];
161
162 if ( !choice->enumeration.empty() ) {
163 bounded = false;
164 enumeration = choice->getEnumeration(token->status, *token->data);
165 count = enumeration.size();
166 }
167 else {
168 if ( !choice->lowerBound || !choice->upperBound ) {
169 throw std::runtime_error("Choices: choice of attribute '" + choice->attribute->name + "' requires an enumeration or bounds");
170 }
171 auto [LB, UB] = choice->getBounds(token->status, *token->data);
172 // the step of a decimal attribute without a step is the precision of numbers
173 step = BPMNOS_NUMBER_PRECISION;
174 if ( choice->multipleOf ) {
175 auto discretizer = choice->multipleOf->execute(token->status, *token->data);
176 if ( !discretizer.has_value() ) {
177 throw std::runtime_error("Choices: cannot determine step for '" + choice->multipleOf->expression + "'");
178 }
179 step = std::abs(discretizer.value());
180 if ( !(step > 0.0) ) {
181 throw std::runtime_error("Choices: step for '" + choice->multipleOf->expression + "' is not positive");
182 }
183 }
184 // the multiples are counted from zero, as by Choice::getEnumeration
185 bounded = true;
186 firstMultiple = std::floor((double)LB / step);
187 if ( BPMNOS::number(firstMultiple * step) < LB ) {
188 firstMultiple += 1.0;
189 }
190 double lastMultiple = std::floor((double)UB / step);
191 if ( BPMNOS::number(lastMultiple * step) > UB ) {
192 lastMultiple -= 1.0;
193 }
194 if ( lastMultiple < firstMultiple ) {
195 count = 0;
196 }
197 else if ( lastMultiple - firstMultiple + 1.0 >= (double)std::numeric_limits<size_t>::max() ) {
198 throw std::runtime_error("Choices: choice of attribute '" + choice->attribute->name + "' allows too many values");
199 }
200 else {
201 count = (size_t)(lastMultiple - firstMultiple + 1.0);
202 }
203 }
204
205 if ( count == 0 ) {
206 return nullptr;
207 }
208
209 if ( bisect.has_value() ? count >= bisect.value() : !choice->multipleOf && choice->enumeration.empty() ) {
210 return discreteBisection();
211 }
212 return bestEnumeratedChoice();
213}

215BPMNOS::number Choices::value(size_t index) const {
216 assert( index < count );
217 return bounded ? BPMNOS::number((firstMultiple + (double)index) * step) : enumeration[index];

220std::shared_ptr<Decision> Choices::bestEnumeratedChoices(std::shared_ptr<const DecisionRequest> request) {
221 auto token = request->token;
222 auto decisionTask = token->node->as<BPMNOS::Model::DecisionTask>();
223
224 auto alternativeChoices = decisionTask->enumerateAlternatives(token->status, *token->data);
225 std::shared_ptr<Decision> bestDecision = nullptr;
226 for ( auto& choices : alternativeChoices ) {
227 auto decision = std::make_shared<ChoiceDecision>(request.get(), std::move(choices), evaluator.get());
228 decision->evaluate();
229 auto reward = decision->reward();
230 if ( reward.has_value() && ( !bestDecision || reward.value() > bestDecision->reward().value() ) ) {
231 bestDecision = decision;
232 }
233 // every alternative is a candidate (infeasible last)
234 addAlternative(decision);
235 }
236 return bestDecision;
237}

260std::tuple<size_t, Choices::Candidate, size_t> Choices::findFeasible(size_t first, size_t last) {
261 const size_t npos = std::numeric_limits<size_t>::max();
262 if ( first > last ) return { npos, { npos, nullptr }, npos };
263
264 // BFS on intervals to find any feasible
265 std::queue<std::pair<size_t, size_t>> intervals;
266 intervals.push({first, last});
267
268 while ( !intervals.empty() ) {
269 auto [l, r] = intervals.front();
270 intervals.pop();
271
272 size_t mid = l + (r - l) / 2;
273 auto candidate = evaluate(mid);
274 if ( candidate.isFeasible() ) {
275 // Found feasible - return with nearest known infeasible bounds
276 size_t nearestLeft = (mid > first) ? l : npos;
277 size_t nearestRight = (mid < last) ? r : npos;
278 return { nearestLeft, candidate, nearestRight };
279 }
280
281 // mid is infeasible - continue searching both halves
282 if ( mid > l ) intervals.push({l, mid - 1});
283 if ( mid < r ) intervals.push({mid + 1, r});
284 }
285
286 return { npos, { npos, nullptr }, npos };
287}

289void Choices::findBetweenFeasibleAndFeasible(Candidate left, Candidate right) {
290 if ( right.index <= left.index + 1 ) return;
291
292 size_t mid = left.index + (right.index - left.index) / 2;
293 auto candidate = evaluate(mid);
294
295 if ( !candidate.isFeasible() ) {
296 // Mid is infeasible - search both sides
297 findBetweenFeasibleAndInfeasible(left, mid);
298 findBetweenInfeasibleAndFeasible(mid, right);
299 }
300 else {
301 // Mid is feasible
302 if ( candidate.reward() > best.reward() ) {
303 best = candidate;
304 }
305 // Check if mid is worse than both boundaries -> peak is at a boundary
306 if ( candidate.reward() <= left.reward() && candidate.reward() <= right.reward() ) {
307 return; // Peak is at one of the boundaries, already checked
308 }
309 // Continue toward better side
310 if ( left.reward() >= right.reward() ) {
311 findBetweenFeasibleAndFeasible(left, candidate);
312 }
313 else {
314 findBetweenFeasibleAndFeasible(candidate, right);
315 }
316 }
317}

319void Choices::findBetweenFeasibleAndInfeasible(Candidate feasible, size_t infeasibleIndex) {
320 while ( infeasibleIndex > feasible.index + 1 ) {
321 size_t mid = feasible.index + (infeasibleIndex - feasible.index + 1) / 2;
322 auto candidate = evaluate(mid);
323
324 if ( !candidate.isFeasible() ) {
325 // Still infeasible - narrow right boundary
326 infeasibleIndex = mid;
327 }
328 else if ( candidate.reward() <= best.reward() ) {
329 // Feasible but worse - done (past peak)
330 break;
331 }
332 else {
333 // Feasible and better - new best, continue toward infeasible
334 best = candidate;
335 feasible = candidate;
336 }
337 }

361std::shared_ptr<Decision> Choices::discreteBisection() {
362 const size_t npos = std::numeric_limits<size_t>::max();
363 best = { npos, nullptr };
364
365 // Check both boundaries
366 auto left = evaluate(0);
367 auto right = (count > 1) ? evaluate(count - 1) : Candidate{ npos, nullptr };
368
369
370 if ( left.isFeasible() ) {
371 best = left;
372 }
373 if ( right.isFeasible() && (!best.isFeasible() || right.reward() > best.reward()) ) {
374 best = right;
375 }
376
377 if ( left.isFeasible() && right.isFeasible() ) {
378 // Both feasible - search between them
379 findBetweenFeasibleAndFeasible(left, right);
380 }
381 else if ( left.isFeasible() && !right.isFeasible() ) {
382 // Left feasible, right infeasible - search toward right
383 findBetweenFeasibleAndInfeasible(left, count - 1);
384 }
385 else if ( !left.isFeasible() && right.isFeasible() ) {
386 // Left infeasible, right feasible - search toward left
387 findBetweenInfeasibleAndFeasible(0, right);
388 }
389 else {
390 // Both infeasible - find any feasible first
391 if ( count <= 2 ) {
392 return nullptr;
393 }
394 auto [leftInfeasible, candidate, rightInfeasible] = findFeasible(1, count - 2);
395 if ( !candidate.isFeasible() ) {
396 return nullptr;
397 }
398 best = candidate;
399
400 // Search both directions from feasible point
401 if ( leftInfeasible != npos ) {
402 findBetweenInfeasibleAndFeasible(leftInfeasible, best);
403 }
404 if ( rightInfeasible != npos ) {
405 findBetweenFeasibleAndInfeasible(best, rightInfeasible);
406 }
407 }
408
409 return best.decision;
