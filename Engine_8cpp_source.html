71void Engine::initialize(std::unique_ptr<Scenario> scenario, BPMNOS::number startTime) {
72 acceptScenario(scenario.get());
73 if ( startTime > scenario->dataProvider->getEarliestInstantiationTime(*scenario) ) {
74 throw std::logic_error("Engine: start time is later than the earliest instantiation time");
75 }
76
77 // create initial system state before the first instant of the run, so that the opening clock tick
78 // advances time to it and time is reached the same way at the first instant as at every later one
79 systemState = std::make_unique<SystemState>(this, scenario.get(), std::numeric_limits<BPMNOS::number>::lowest());
80 environment.setScenario(std::move(scenario));
81 commands.clear();
82 conditionalEventObserver.connect( systemState.get() );
83 // announce the installed state so subscribers (cached candidate sources) reset for the new run
84 notify( systemState.get() );
85
86 // open the run by advancing time to its first instant; the tick is announced like any other, so the
87 // log states the time the run began at, and deferred data is disclosed for that instant
88 ClockTickEvent clockTickEvent(systemState.get(), startTime);
89 notify(&clockTickEvent);
90 clockTickEvent.processBy(this);
91}

129void Engine::resume(std::shared_ptr<Event> event) {
130 if ( !systemState ) {
131 throw std::logic_error("Engine: resume requires an existing system state (call run or initializeSystemState first)");
132 }
133 if ( event->expired() ) {
134 // a controller must check Event::expired() before forcing an event; guard against a stale one
135 throw std::logic_error("Engine: event to resume is expired");
136 }
137 if ( event->is<TerminationEvent>() ) {
138 // resuming only to terminate is forbidden
139 throw std::logic_error("Engine: TerminationEvent is not allowed for resume");
140 }
141 // the run continues with the given decision
142 notify(event.get());

147bool Engine::advance() {
148 if ( !systemState ) {
149 throw std::logic_error("Engine: advance requires an existing system state (call initialize, run or initializeSystemState first)");
150 }
151 // every enqueued command is executed by whoever enqueued it, so none is outstanding here
152 assert(commands.empty());
153
154 // a round asks the environment for an event due at the current time, then the dispatchers of the
155 // controller, and, if neither supplies one, asks the environment to advance
156 auto event = environment.dispatchEvent(systemState.get());
157 if ( !event ) {
158 event = fetchEvent(systemState.get());
159 }
160 if ( !event ) {
161 event = environment.advance(systemState.get());
162 }
163 if ( !event ) {
164 // the run waits for the wall clock or for events from outside the run
165 wait();
166 return true;
167 }
168
169 if ( event->expired() ) {
170 // the engine assumes only non-expired events are dispatched; a controller that is not safe by
171 // design must check Event::expired() before dispatching. Guard against a stale event here.
172 throw std::logic_error("Engine: event fetched is expired");
173 }
174 notify(event.get());
175 event->processBy(this);
176
177 if ( event->is<TerminationEvent>() ) {
178 // the run was told to stop
179 return false;
180 }
181
182 return true;
183}

185void Engine::broadcastSignal(Signal signal) {
186 // the signal is announced before it is delivered anywhere, so that what is observed is the signal
187 // rather than what it causes
188 notify(&signal);
189
190 auto& waitingTokens = systemState->tokensAwaitingSignal[signal.name];
191 for ( auto& [token_ptr] : waitingTokens ) {
192 auto token = token_ptr.lock();
193 assert( token );
194 // receive signal content
195 token->setSignalContent(signal.content);
196
197 // advance receiving token
198 commands.emplace_back(std::bind(&Token::advanceToCompleted,token.get()), token.get());
199 }
200 waitingTokens.clear();
201
202 // instantiate the process the signal triggers, if any
203 auto& processesTriggeredBySignal = model->processesTriggeredBySignal;
204 if ( auto it = processesTriggeredBySignal.find(signal.name); it != processesTriggeredBySignal.end() ) {
205 // the instantiation is enqueued rather than performed here, so that a process throwing the signal
206 // instantiating it does not recurse through the stack of the broadcast
207 commands.emplace_back( std::bind(&Engine::triggerInstance, this, it->second, signal.content) );
208 }

211void Engine::triggerInstance(const BPMN::Process* process, BPMNOS::VariedValueMap content) {
212 auto extensionElements = process->extensionElements->as<BPMNOS::Model::ExtensionElements>();
213 auto& attributeRegistry = extensionElements->attributeRegistry;
214
215 // the identifier is generated, an instance created by a trigger being declared nowhere
216 auto counter = ++systemState->instantiationCounter[process];
217 auto instanceId = process->id + StateMachine::delimiters[1] + std::to_string(counter);
218
219 BPMNOS::Values data( extensionElements->data.size() );
220 data[Model::ExtensionElements::Position::Instance] = BPMNOS::to_number(instanceId,STRING);
221
222 BPMNOS::Values status( extensionElements->attributes.size() );
223 status[Model::ExtensionElements::Index::Timestamp] = systemState->getTime();
224
225 // the instance is created before the content of the trigger is applied, so that the content is written
226 // through the data of the token at the process, which holds the global attributes and those of the instance
227 auto token = systemState->stateMachine->createInstance(process, std::move(data), std::move(status));
228
229 // the content of the trigger is applied to the initial status of the instance, which is where it is
230 // needed and after which it is of no further concern
231 assert( process->startNodes.size() == 1 );
232 auto startNode = process->startNodes.front();
233
234 // a signal start event holds its definition as its extension elements, whereas a message start event
235 // holds extension elements carrying a message definition
236 const BPMNOS::Model::ContentMap* contentMap;
237 const std::vector<const BPMNOS::Model::Attribute*>* dataUpdateAttributes;
238 if ( auto signalDefinition = startNode->extensionElements->represents<BPMNOS::Model::SignalDefinition>() ) {
239 contentMap = &signalDefinition->contentMap;
240 dataUpdateAttributes = &signalDefinition->dataUpdate.attributes;
241 }
242 else {
243 auto startNodeExtensionElements = startNode->extensionElements->as<BPMNOS::Model::ExtensionElements>();
244 contentMap = &startNodeExtensionElements->getMessageDefinition()->contentMap;
245 dataUpdateAttributes = &startNodeExtensionElements->dataUpdate.attributes;
246 }
247
248 BPMNOS::number objectiveChange = 0;
249 for ( auto& [key,definition] : *contentMap ) {
250 auto attribute = definition->attribute;
251 auto it = content.find(key);
252 if ( it == content.end() ) {
253 // key in content of start event, but not in content of the trigger
254 objectiveChange += attributeRegistry.setValue(attribute, token->status, *token->data, std::nullopt );
255 }
256 else if ( std::holds_alternative< std::optional<BPMNOS::number> >(it->second) ) {
257 objectiveChange += attributeRegistry.setValue(attribute, token->status, *token->data, std::get< std::optional<BPMNOS::number> >(it->second) );
258 }
259 else {
260 // use default value of emitter
261 ValueVariant value = std::get< std::string >(it->second);
262 objectiveChange += attributeRegistry.setValue(attribute, token->status, *token->data, BPMNOS::to_number(value,attribute->type) );
263 }
264 }
265
266 addToObjective(objectiveChange);
267
268 if ( !dataUpdateAttributes->empty() ) {
269 // the observers decide by the index of each attribute whether it concerns this instance or all of them
270 notify( DataUpdate( token->getInstanceId(), *dataUpdateAttributes ) );
271 }
272
273 // the trigger is the condition for the start, so the instance is started at once and not upon a ready
274 // event: the token at the process advances through READY
275 token->advanceToReady();
276}

278void Engine::triggerInstanceByMessage(const BPMN::Process* process, std::weak_ptr<Message> message_ptr) {
279 auto message = message_ptr.lock();

299void Engine::process(const InstantiationEvent* event) {
300 auto process = event->process;
301 auto& status = const_cast<InstantiationEvent*>(event)->status;
302 auto& data = const_cast<InstantiationEvent*>(event)->data;
303 if ( !data[Model::ExtensionElements::Position::Instance].has_value() ) {
304 throw std::runtime_error("Engine: instance of process '" + process->id + "' has no id");
305 }
306 if ( !status[Model::ExtensionElements::Index::Timestamp].has_value() ) {
307 throw std::runtime_error("Engine: instance of process '" + process->id + "' has no timestamp");
308 }
309 systemState->instantiationCounter[process]++;
310 auto token = systemState->stateMachine->createInstance(process, std::move(data), std::move(status));
311 // the token at the process awaits the ready event starting the instance
312 token->advanceFromCreated();
313
314 processCommands();
315}

338void Engine::process(const ReadyEvent* event) {
339 auto token_ptr = event->token.lock();
340 assert( token_ptr );
341 Token* token = const_cast<Token*>(token_ptr.get());
342 systemState->tokensAwaitingReadyEvent.remove(token);
343
344 auto& status = const_cast<ReadyEvent*>(event)->statusAttributes;
345 token->status = std::move(status);
346 token->status[BPMNOS::Model::ExtensionElements::Index::Timestamp] = systemState->currentTime;
347
348 if ( token->node->represents<BPMN::Process>() ) {
349 // the token at a process starts the instance: the data owned by the state machine of the instance is
350 // replaced element by element, so that the references to it remain valid; the data of the instance
351 // follows the global attributes, beginning with the instance at the instance index
352 auto stateMachine = token->owned.get();
353 auto& data = const_cast<ReadyEvent*>(event)->dataAttributes;
354 auto instanceIndex = model->instanceIndex;
355 assert( data.size() == stateMachine->data.size() - instanceIndex );
356 for ( size_t i = 0; i < data.size(); i++ ) {
357 stateMachine->data[instanceIndex + i].get() = data[i];
358 }
359 }
360 else if ( auto scope = token->node->represents<BPMN::Scope>() ) {
361 auto& data = const_cast<ReadyEvent*>(event)->dataAttributes;
362 const_cast<StateMachine*>(token->owner)->createChild(token, scope, std::move(data));
363 }
364 commands.emplace_back(std::bind(&Token::advanceToReady,token), token);
365
366 token_ptr.reset();
367 processCommands();
368}

495void Engine::process([[maybe_unused]] const ClockTickEvent* event) {
496//std::cerr << "ClockTickEvent " << std::endl;
497 systemState->increaseTimeTo(event->time);
498
499 // trigger tokens awaiting timer
500 while ( !systemState->tokensAwaitingTimer.empty() ) {
501 auto it = systemState->tokensAwaitingTimer.begin();
502 auto [time, token_ptr] = *it;
503 if ( time > systemState->getTime() ) {
504 break;
505 }
506 auto token = token_ptr.lock();
507 assert( token );
508 notify(TimerEvent(token.get()));
509 commands.emplace_back(std::bind(&Token::advanceToCompleted,token.get()), token.get());
510 systemState->tokensAwaitingTimer.remove(token.get());
511 }
512
513 processCommands();
514}
