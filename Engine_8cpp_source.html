58void Engine::initialize(const BPMNOS::Model::Scenario* scenario, BPMNOS::number startTime) {
59 if ( startTime > scenario->getEarliestInstantiationTime() ) {
60 throw std::logic_error("Engine: start time is later than the earliest instantiation time");
61 }
62
63 // create initial system state before the first instant of the run, so that the opening clock tick
64 // advances time to it and time is reached the same way at the first instant as at every later one
65 systemState = std::make_unique<SystemState>(this, scenario, std::numeric_limits<BPMNOS::number>::lowest());
66 lastInstantiationTime = std::numeric_limits<BPMNOS::number>::lowest();
67 commands.clear();
68 conditionalEventObserver.connect( systemState.get() );
69 // announce the installed state so subscribers (cached candidate sources) reset for the new run
70 notify( systemState.get() );
71
72 // open the run by advancing time to its first instant; the tick is announced like any other, so the
73 // log states the time the run began at, and deferred data is disclosed for that instant
74 ClockTickEvent clockTickEvent(systemState.get(), startTime);
75 notify(&clockTickEvent);
76 clockTickEvent.processBy(this);
77}

89void Engine::initializeSystemState(const BPMNOS::Model::Scenario* scenario, const SystemState* foreignState) {
90 // install a deep copy of the foreign state as this engine's own state; the copy already holds every
91 // instance due up to its current time, so the instantiation watermark starts at that time
92 systemState = std::make_unique<SystemState>(this, scenario, foreignState);
93 lastInstantiationTime = systemState->getTime();
94 // installing a new state resets the run state and binds the conditional-event observer to it
95 commands.clear();
96 conditionalEventObserver.connect( systemState.get() );
97 // announce the initialized state so subscribers can build their data structures
98 notify( systemState.get() );
99}

114void Engine::resume(std::shared_ptr<Event> event, BPMNOS::number endTime) {
115 if ( !systemState ) {
116 throw std::logic_error("Engine: resume requires an existing system state (call run or initializeSystemState first)");
117 }
118 if ( event->expired() ) {
119 // a controller must check Event::expired() before forcing an event; guard against a stale one
120 throw std::logic_error("Engine: event to resume is expired");
121 }
122 if ( event->is<TerminationEvent>() ) {
123 // resuming only to terminate is forbidden
124 throw std::logic_error("Engine: TerminationEvent is not allowed for resume");
125 }
126 // the run continues with the given decision
127 notify(event.get());
128 event->processBy(this);
129 run(endTime);
130}

132bool Engine::advance(BPMNOS::number endTime) {
133 if ( !systemState ) {
134 throw std::logic_error("Engine: advance requires an existing system state (call initialize, run or initializeSystemState first)");
135 }
136 // every enqueued command is executed by whoever enqueued it, so none is outstanding here
137 assert(commands.empty());
138
139 // fetch and process a single event
140 auto event = fetchEvent(systemState.get());
141 if ( !event ) {
142 // No event available, terminate
143 return false;
144 }
145
146 if ( event->expired() ) {
147 // the engine assumes only non-expired events are dispatched; a controller that is not safe by
148 // design must check Event::expired() before dispatching. Guard against a stale event here.
149 throw std::logic_error("Engine: event fetched is expired");
150 }
151 // stop before processing clock tick that would exceed endTime
152 if ( auto clockTickEvent = event->is<ClockTickEvent>(); clockTickEvent && clockTickEvent->time > endTime ) {
153 return false;
154 }
155 notify(event.get());
156 event->processBy(this);
157
158 if ( event->is<ClockTickEvent>() ) {
159//std::cerr << "ClockTick" << std::endl;
160 // continue only while the state has something left to advance
161 return systemState->isAlive();
162 }
163
164 if ( event->is<TerminationEvent>() ) {
165 // the run was told to stop
166 return false;
167 }
168
169 return true;
170}

172void Engine::addInstances() {
173 for (auto& [process,status,data] : systemState->getInstantiations() ) {
174 if ( !process->isExecutable ) {
175 throw std::runtime_error("Engine: process '" + process->id + "' is not executable");
176 }
177 if ( !data[Model::ExtensionElements::Index::Instance].has_value() ) {
178 throw std::runtime_error("Engine: instance of process '" + process->id + "' has no id");
179 }
180 if ( !status[Model::ExtensionElements::Index::Timestamp].has_value() ) {
181 throw std::runtime_error("Engine: instance of process '" + process->id + "' has no timestamp");
182 }
183 systemState->instantiationCounter++;
184 systemState->instances.push_back(std::make_shared<StateMachine>(systemState.get(),process,std::move(data)));
185 // run instance and advance token
186 systemState->instances.back()->run(std::move(status));
187 }
188 lastInstantiationTime = systemState->getTime();
189}

196void Engine::process(const ReadyEvent* event) {
197 auto token_ptr = event->token.lock();
198 assert( token_ptr );
199 Token* token = const_cast<Token*>(token_ptr.get());
200 systemState->tokensAwaitingReadyEvent.remove(token);
201
202 auto& status = const_cast<ReadyEvent*>(event)->statusAttributes;
203 token->status = std::move(status);
204 token->status[BPMNOS::Model::ExtensionElements::Index::Timestamp] = systemState->currentTime;
205
206 if ( auto scope = token->node->represents<BPMN::Scope>() ) {
207 auto& data = const_cast<ReadyEvent*>(event)->dataAttributes;
208 const_cast<StateMachine*>(token->owner)->createChild(token, scope, std::move(data));
209 }
210 commands.emplace_back(std::bind(&Token::advanceToReady,token), token);
211
212 token_ptr.reset();

238void Engine::process(const ChoiceEvent* event) {
239//std::cerr << "ChoiceEvent " << event.token->node->id << std::endl;
240 auto token_ptr = event->token.lock();
241 assert( token_ptr );
242 Token* token = const_cast<Token*>(token_ptr.get());
243 token->decisionRequest.reset();
244 token->status[BPMNOS::Model::ExtensionElements::Index::Timestamp] = systemState->currentTime;
245 assert( token->node );

255 }

282void Engine::process(const MessageDeliveryEvent* event) {
283 auto token_ptr = event->token.lock();
284 assert( token_ptr );
285 Token* token = const_cast<Token*>(token_ptr.get());
286 token->decisionRequest.reset();
287 token->status[BPMNOS::Model::ExtensionElements::Index::Timestamp] = systemState->currentTime;
288 assert( token->node );
289
290 auto message_ptr = event->message.lock();
291 assert( message_ptr );
292 Message* message = const_cast<Message*>(message_ptr.get());
293 // update token status
294 auto oldObjective = token->globals[BPMNOS::Model::ExtensionElements::Index::Objective];
295 message->apply(token->node,token->getAttributeRegistry(),token->status,*token->data,token->globals);
296 if ( token->globals[BPMNOS::Model::ExtensionElements::Index::Objective] != oldObjective ) {
297 // dataUpdate indicating that objective has changed
298 notify( DataUpdate( { token->getAttributeRegistry().globalAttributes[BPMNOS::Model::ExtensionElements::Index::Objective] } ) );
299 }
300
301 message->state = Message::State::DELIVERED;
302 notify(message);
303
304 erase_ptr<Message>(systemState->messages,message);
305
306 if ( message->waitingToken ) {
307 // send task is completed
308 systemState->messageAwaitingDelivery.erase( message->waitingToken );
309 commands.emplace_back(std::bind(&Token::advanceToCompleted,message->waitingToken), message->waitingToken);
310 }
311 commands.emplace_back(std::bind(&Token::advanceToCompleted,token), token);
312
313 message_ptr.reset();
314 token_ptr.reset();
315 processCommands();
316}

350void Engine::process([[maybe_unused]] const ClockTickEvent* event) {
351//std::cerr << "ClockTickEvent " << std::endl;
352 systemState->increaseTimeTo(event->time);
353
354 // add new instances that have become due at the new time; instantiation is caused by time advancing,
355 // and time advances here and nowhere else
356 if (lastInstantiationTime < systemState->getTime()) {
357 addInstances();
358 }
359
360 // trigger tokens awaiting timer
361 while ( !systemState->tokensAwaitingTimer.empty() ) {
362 auto it = systemState->tokensAwaitingTimer.begin();
363 auto [time, token_ptr] = *it;
364 if ( time > systemState->getTime() ) {
365 break;
366 }
367 auto token = token_ptr.lock();
368 assert( token );
369 notify(TimerEvent(token.get()));
370 commands.emplace_back(std::bind(&Token::advanceToCompleted,token.get()), token.get());
371 systemState->tokensAwaitingTimer.remove(token.get());
372 }
373
374 processCommands();
375}
