40 // Settle the realized past before resampling: forking spawns at spawnTime = currentTime + 1, so every
41 // disclosure due up to currentTime (= spawnTime - 1) was already revealed to the live run via

84void StochasticScenario::evaluateDeferredDisclosures(BPMNOS::number spawnTime) {
85 if (!stochasticHandle || !randomFactory) {
86 return;
87 }
88
89 for (auto& [instanceId, disclosures] : deferredDisclosures) {
90 auto& instance = instances.at(instanceId);
91
92 for (auto& disclosure : disclosures) {
93 // Skip if already revealed (in pastDisclosures)
94 if (pastDisclosures.contains(instanceId) &&
95 pastDisclosures.at(instanceId).contains(disclosure.attribute)
96 ) {
97 const auto& pastDisclosure = pastDisclosures.at(instanceId).at(disclosure.attribute);
98 if (pastDisclosure.disclosureTime >= spawnTime) {
99 throw std::logic_error(
100 "StochasticScenario: past disclosure for attribute '" + disclosure.attribute->name +
101 "' (" + BPMNOS::to_string(instanceId, BPMNOS::STRING) +
102 ", " + disclosure.node->id +
103 ") has disclosureTime " + std::to_string((double)pastDisclosure.disclosureTime) +
104 " >= spawnTime " + std::to_string((double)spawnTime) +
105 " (disclosed value " + std::to_string((double)pastDisclosure.value) + ")"
106 );
107 }
108 continue;
109 }
110
111 // Check if already evaluated with disclosureTime < spawnTime (precondition violation)
112 if (pendingDisclosures.contains(instanceId) &&
113 pendingDisclosures.at(instanceId).contains(disclosure.attribute) &&
114 pendingDisclosures.at(instanceId).at(disclosure.attribute).disclosureTime < spawnTime
115 ) {
116 const auto& pendingDisclosure = pendingDisclosures.at(instanceId).at(disclosure.attribute);
117 throw std::logic_error(
118 "StochasticScenario: pending disclosure for attribute '" + disclosure.attribute->name +
119 "' (" + BPMNOS::to_string(instanceId, BPMNOS::STRING) +
120 ", " + disclosure.node->id +
121 ") has disclosureTime " + std::to_string((double)pendingDisclosure.disclosureTime) +
122 " < spawnTime " + std::to_string((double)spawnTime) +
123 " (pending value " + std::to_string((double)pendingDisclosure.value) + ")"
124 );
125 }
126
127 // Set RNG context for this (instance, node)
128 auto& rng = getRng(instanceId, disclosure.node);
129 randomFactory->setCurrentRng(&rng);
130
131 // Build status/data vectors from instance values
132 auto extensionElements = disclosure.node->extensionElements->as<const ExtensionElements>();
133 Values status, data;
134 for (auto& attr : extensionElements->attributes) {
135 status.push_back(instance.values.contains(attr.get()) ? instance.values.at(attr.get()) : std::nullopt);
136 }
137 for (auto& attr : extensionElements->data) {
138 data.push_back(instance.values.contains(attr.get()) ? instance.values.at(attr.get()) : std::nullopt);
139 }
140
141 // Evaluate initialization expression
142 BPMNOS::number value = disclosure.initializationExpression->execute(status, data, globals).value_or(0);
143
144 // Apply type conversion
145 switch (disclosure.attribute->type) {
146 case ValueType::INTEGER:
147 value = BPMNOS::number((int)value);
148 break;
149 case ValueType::BOOLEAN:
150 value = BPMNOS::number(value != 0 ? 1 : 0);
151 break;
152 default:
153 break;
154 }
155
156 // Resample timestamp if before spawnTime
157 if (disclosure.attribute->id == Keyword::Timestamp) {
158 for (int tries = 1; value < spawnTime && tries < maxResamplingTries; ++tries) {
159 value = disclosure.initializationExpression->execute(status, data, globals).value_or(0);
160 value = BPMNOS::number((int)value); // Timestamp is INTEGER type
161 }
162
163 // Fall back to spawnTime if still in the past
164 if (value < spawnTime) {
165 value = spawnTime;
166 }
167
168 instance.instantiationTime = BPMNOS::number(std::ceil((double)value));
169 }
170
171 // Store value for dependent attributes
172 instance.values[disclosure.attribute] = value;
173
174 // Evaluate disclosure expression with resampling if needed
175 BPMNOS::number disclosureTime = BPMNOS::number(std::ceil((double)disclosure.disclosureExpression->execute(status, data, globals).value_or(0)));
176
177 // Resample if disclosure time is before spawnTime
178 for (int tries = 1; disclosureTime < spawnTime && tries < maxResamplingTries; ++tries) {
179 disclosureTime = BPMNOS::number(std::ceil((double)disclosure.disclosureExpression->execute(status, data, globals).value_or(0)));
180 }
181
182 // Fall back to spawnTime if still in the past
183 if (disclosureTime < spawnTime) {
184 disclosureTime = spawnTime;
185 }
186
187 // Compute effective disclosure time (max with parent scope)
188 BPMNOS::number effectiveDisclosureTime = disclosureTime;
189 if (auto childNode = disclosure.node->represents<BPMN::ChildNode>()) {
190 auto parentNode = childNode->parent;
191 if (disclosureTimes.contains(instanceId) && disclosureTimes.at(instanceId).contains(parentNode)) {
192 effectiveDisclosureTime = std::max(effectiveDisclosureTime, disclosureTimes.at(instanceId).at(parentNode));
193 }
194 }
195
196 // Update node disclosure time
197 if (!disclosureTimes[instanceId].contains(disclosure.node)) {
198 disclosureTimes[instanceId][disclosure.node] = effectiveDisclosureTime;
199 }
200 else {
201 disclosureTimes[instanceId][disclosure.node] = std::max(disclosureTimes[instanceId][disclosure.node], effectiveDisclosureTime);
202 }
203
204 // Store in pending disclosures
205 pendingDisclosures[instanceId][disclosure.attribute] = {disclosure.attribute, disclosureTime, value};
206 }
207 }
208
209 // Clear RNG context
210 randomFactory->setCurrentRng(nullptr);
211
212 // Compute instantiation bounds now that all disclosures are evaluated
213 computeInstantiationBounds();
214}

241void StochasticScenario::initializeActivityData(BPMNOS::number rootId, const BPMN::Node* node, const Values& status, const SharedValues& data, const Values& globals) const {
242 // store key uses the full instance id (from data) so concurrent executions of one node don't collide;
243 // expressions/RNG/declared-instances are keyed by the root id (specified pre-runtime)
244 auto instanceId = (size_t)data[BPMNOS::Model::ExtensionElements::Index::Instance].get().value();
245 auto key = std::make_pair(instanceId, node);
246 auto& instance = instances.at((size_t)rootId);
247 auto extensionElements = node->extensionElements->as<const ExtensionElements>();
248
249 // Build full status: parent status + node's own attributes
250 Values fullStatus = status;
251 for (auto& attribute : extensionElements->attributes) {
252 fullStatus.push_back(getValue(&instance, attribute.get(), 0));
253 }
254
255 // Apply ready expressions if any
256 if (readyExpressions.contains((size_t)rootId) && readyExpressions.at((size_t)rootId).contains(node)) {
257 // Set RNG context for random functions
258 auto& randomGenerator = getRng((size_t)rootId, node);
259 if (randomFactory) {
260 randomFactory->setCurrentRng(&randomGenerator);
261 }
262
263 // Evaluate ready expressions and apply to full status
264 for (auto& expression : readyExpressions.at((size_t)rootId).at(node)) {
265 auto value = expression->execute(fullStatus, data, globals);
266 if (value.has_value() && expression->target) {
267 // Apply type conversion
268 BPMNOS::number convertedValue;
269 switch (expression->target.value()->type) {
270 case ValueType::INTEGER:
271 convertedValue = BPMNOS::number((int)value.value());
272 break;
273 case ValueType::BOOLEAN:
274 convertedValue = BPMNOS::number(value.value() != 0 ? 1 : 0);
275 break;
276 default:
277 convertedValue = value.value();
278 break;
279 }
280 // Update full status at attribute's index
281 fullStatus[expression->target.value()->index] = convertedValue;
282 }
283 }
284
285 // Clear RNG context
286 if (randomFactory) {
287 randomFactory->setCurrentRng(nullptr);
288 }
289 }
290
291 // Store the full computed status
292 activityArrivalStatus[key] = std::move(fullStatus);

295std::mt19937& StochasticScenario::getRng(size_t instanceId, const BPMN::Node* node) const {
296 auto key = std::make_pair(instanceId, node);
297 if (!rngs.contains(key)) {
298 // Create RNG seeded from current seed combined with instanceId and node hash
299 unsigned int currentSeed = scenarioSeeds.back().second;
300 size_t nodeHash = std::hash<std::string>{}(node->id);
301 size_t combinedSeed = static_cast<size_t>(currentSeed) ^ (instanceId * 31) ^ (nodeHash * 17);
302 rngs[key] = std::mt19937(combinedSeed);
303 }
304 return rngs[key];

374std::optional<BPMNOS::number> StochasticScenario::getValue( const Scenario::InstanceData* instance, const BPMNOS::Model::Attribute* attribute, [[maybe_unused]] const BPMNOS::number currentTime) const {
375 if (attribute->expression && attribute->expression->type == Expression::Type::ASSIGN) {
376 std::vector<double> variableValues;
377 for (auto input : attribute->expression->variables) {
378 if (!input->isImmutable) {
379 return std::nullopt;
380 }
381 auto value = getValue(instance, input, currentTime);
382 if (!value.has_value()) {
383 return std::nullopt;
384 }
385 variableValues.push_back((double)value.value());
386 }
387
388 std::vector<std::vector<double>> collectionValues;
389 for (auto input : attribute->expression->collections) {
390 if (!input->isImmutable) {
391 return std::nullopt;
392 }
393 collectionValues.push_back({});
394 auto collection = getValue(instance, input, currentTime);
395 if (!collection.has_value()) {
396 return std::nullopt;
397 }
398 for (auto value : collectionRegistry[(size_t)collection.value()]) {
399 collectionValues.back().push_back(value);
400 }
401 }
402
403 return number(attribute->expression->compiled.evaluate(variableValues, collectionValues));
404 }
405 else {
406 if (instance->values.contains(attribute)) {
407 return instance->values.at(attribute);
408 }
409 }
410 return std::nullopt;
411}
