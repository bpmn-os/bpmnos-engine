44std::string Layout::stringify() const {
45 // the sizes of the arrays are written after the innermost elements, outermost first
46 std::string sizes;
47 const Layout* current = this;
48 while ( auto array = std::get_if<ArrayLayout>(current) ) {
49 sizes += "[" + std::to_string(array->size) + "]";
50 current = array->layout.get();
51 }
52 if ( std::holds_alternative<std::monostate>(*current) ) {
53 return current->schema->stringify() + sizes;
54 }
55 auto& fields = std::get<ObjectLayout>(*current).fields;
56 auto& types = std::get<ObjectType>(*current->schema).fields;
57 std::string result = "{ ";
58 for ( size_t k = 0; k < fields.size(); k++ ) {
59 result += ( k ? ", " : "" ) + types[k].name + ": " + fields[k].layout->stringify();
60 }
61 return result + " }" + sizes;
62}

64bool Layout::operator==(const Layout& other) const {
65 if ( this == &other ) {
66 return true;
67 }
68 if ( index() != other.index() || size != other.size || !( *schema == *other.schema ) ) {
69 return false;
70 }
71 if ( auto array = std::get_if<ArrayLayout>(this) ) {
72 auto& otherArray = std::get<ArrayLayout>(other);
73 return array->size == otherArray.size && *array->layout == *otherArray.layout;
74 }
75 if ( auto object = std::get_if<ObjectLayout>(this) ) {
76 return std::ranges::equal(object->fields, std::get<ObjectLayout>(other).fields, [](auto& field, auto& otherField) {
77 return field.offset == otherField.offset && *field.layout == *otherField.layout;
78 });
79 }
80 return true;

85/// Renders the values of a layout beginning at the given offset.
86std::string render(const Layout& layout, const std::vector<Value>& values, size_t offset) {
87 if ( auto array = std::get_if<ArrayLayout>(&layout) ) {
88 if ( array->size == 0 ) {
89 return "[ ]";
90 }
91 std::string result;
92 for ( size_t i = 0; i < array->size; i++ ) {
93 result += ", " + render(*array->layout, values, offset + i * array->layout->size);
94 }
95 result.front() = '[';
96 return result + " ]";
97 }
98 if ( std::holds_alternative<std::monostate>(layout) ) {
99 auto& value = values[offset];
100 if ( !value.has_value() ) {
101 return "undefined";
102 }
103 auto type = std::get<ValueType>(*layout.schema);
104 auto text = BPMNOS::to_string(value.value(), type);
105 return type == ValueType::STRING ? "\"" + text + "\"" : text;
106 }
107 auto& fields = std::get<ObjectLayout>(layout).fields;
108 auto& types = std::get<ObjectType>(*layout.schema).fields;
109 std::string result = "{ ";
110 for ( size_t k = 0; k < fields.size(); k++ ) {
111 result += ( k ? ", " : "" ) + types[k].name + " := " + render(*fields[k].layout, values, offset + fields[k].offset);
112 }
113 return result + " }";

116/// Renders the values of a layout beginning at the given offset as JSON.
117nlohmann::ordered_json json(const Layout& layout, const std::vector<Value>& values, size_t offset) {
118 if ( auto array = std::get_if<ArrayLayout>(&layout) ) {
119 auto result = nlohmann::ordered_json::array();
120 for ( size_t i = 0; i < array->size; i++ ) {
121 result.push_back( json(*array->layout, values, offset + i * array->layout->size) );
122 }
123 return result;
124 }
125 if ( std::holds_alternative<std::monostate>(layout) ) {
126 auto& value = values[offset];
127 if ( !value.has_value() ) {
128 return nullptr;
129 }
130 switch ( std::get<ValueType>(*layout.schema) ) {
131 case ValueType::BOOLEAN:
132 return value.value() != number(0);
133 case ValueType::INTEGER:
134 return (long)value.value();
135 case ValueType::DECIMAL:
136 return (double)value.value();
137 case ValueType::STRING:
138 return BPMNOS::to_string(value.value(), ValueType::STRING);
139 }
140 }
141 auto& fields = std::get<ObjectLayout>(layout).fields;
142 auto& types = std::get<ObjectType>(*layout.schema).fields;
143 auto result = nlohmann::ordered_json::object();
144 for ( size_t k = 0; k < fields.size(); k++ ) {
145 result[types[k].name] = json(*fields[k].layout, values, offset + fields[k].offset);
146 }
147 return result;

150/**
151 * Visits the values of an object of the layout `from` that an object of the layout `to`, of the same structure,
152 * also has, calling `visit(source, target)` with the positions of each such value in the two objects.
153 */
154void correspond(const Layout& from, size_t fromOffset, const Layout& to, size_t toOffset, const std::function<void(size_t, size_t)>& visit) {
155 if ( auto array = std::get_if<ArrayLayout>(&from) ) {
156 auto& toArray = std::get<ArrayLayout>(to);
157 size_t count = std::min(array->size, toArray.size);
158 for ( size_t i = 0; i < count; i++ ) {
159 correspond(*array->layout, fromOffset + i * array->layout->size, *toArray.layout, toOffset + i * toArray.layout->size, visit);
160 }
161 return;
162 }
163 if ( std::holds_alternative<std::monostate>(from) ) {
164 visit(fromOffset, toOffset);
165 return;
166 }
167 auto& fields = std::get<ObjectLayout>(from).fields;
168 auto& toFields = std::get<ObjectLayout>(to).fields;
169 for ( size_t k = 0; k < fields.size(); k++ ) {
170 correspond(*fields[k].layout, fromOffset + fields[k].offset, *toFields[k].layout, toOffset + toFields[k].offset, visit);
171 }

183/// Returns the layout in which the array the steps reach from the given one on has the given size, rebuilding the
184/// layouts and, where a fixed array changes its size, the schemas along the steps.
185std::shared_ptr<const Layout> rebuild(const std::shared_ptr<const Layout>& layout, const std::vector< std::optional<std::string> >& steps, size_t step, size_t size) {
186 if ( step == steps.size() ) {
187 auto& array = std::get<ArrayLayout>(*layout);
188 auto& type = std::get<ArrayType>(*layout->schema);
189 // an array fixed in the schema stays fixed at its new size
190 auto schema = type.size.has_value() ? std::make_shared<const Schema>( ArrayType{ type.type, size } ) : layout->schema;
191 return std::make_shared<const Layout>(schema, array.layout, size);
192 }
193 if ( !steps[step].has_value() ) {
194 auto& array = std::get<ArrayLayout>(*layout);
195 auto element = rebuild(array.layout, steps, step + 1, size);
196 auto& type = std::get<ArrayType>(*layout->schema);
197 auto schema = ( element->schema == array.layout->schema ) ? layout->schema : std::make_shared<const Schema>( ArrayType{ element->schema, type.size } );
198 return std::make_shared<const Layout>(schema, element, array.size);
199 }
200 auto& name = steps[step].value();
201 auto type = std::get<ObjectType>(*layout->schema);
202 auto field = (size_t)std::distance(type.fields.begin(), std::ranges::find_if(type.fields, [&name](auto& candidate) { return candidate.name == name; }));
203 if ( field == type.fields.size() ) {
204 throw std::logic_error("Object: unknown field '" + name + "'");
205 }
206 auto fields = fieldLayoutsOf(*layout);
207 auto changed = rebuild(fields[field], steps, step + 1, size);
208 auto schema = layout->schema;
209 if ( changed->schema != fields[field]->schema ) {
210 type.fields[field].type = changed->schema;
211 schema = std::make_shared<const Schema>( std::move(type) );
212 }
213 fields[field] = std::move(changed);
214 return std::make_shared<const Layout>(schema, fields);
215}

217/// Returns the layout in which every array has the largest size it has in the two layouts, and is fixed only where
218/// both fix it at the same size.
219std::shared_ptr<const Layout> widen(const std::shared_ptr<const Layout>& layout, const std::shared_ptr<const Layout>& other) {
220 if ( layout == other || *layout == *other || std::holds_alternative<std::monostate>(*layout) ) {
221 return layout;
222 }
223 if ( auto array = std::get_if<ArrayLayout>(layout.get()) ) {
224 auto& otherArray = std::get<ArrayLayout>(*other);
225 auto element = widen(array->layout, otherArray.layout);
226 auto& type = std::get<ArrayType>(*layout->schema);
227 auto& otherType = std::get<ArrayType>(*other->schema);
228 auto fixed = ( type.size.has_value() && type.size == otherType.size ) ? type.size : std::nullopt;
229 auto schema = ( element->schema == array->layout->schema && fixed == type.size ) ? layout->schema : std::make_shared<const Schema>( ArrayType{ element->schema, fixed } );
230 return std::make_shared<const Layout>(schema, element, std::max(array->size, otherArray.size));
231 }
232 auto fields = fieldLayoutsOf(*layout);
233 auto otherFields = fieldLayoutsOf(*other);
234 auto type = std::get<ObjectType>(*layout->schema);
235 bool changed = false;
236 for ( size_t k = 0; k < fields.size(); k++ ) {
237 auto widened = widen(fields[k], otherFields[k]);
238 if ( widened->schema != fields[k]->schema ) {
239 type.fields[k].type = widened->schema;
240 changed = true;
241 }
242 fields[k] = std::move(widened);
243 }
244 return std::make_shared<const Layout>( changed ? std::make_shared<const Schema>( std::move(type) ) : layout->schema, fields );
245}

267std::shared_ptr<const Object> merge(const std::vector< std::shared_ptr<const Object> >& objects) {
268 assert( !objects.empty() );
269 if ( std::ranges::all_of(objects, [&objects](auto& object) { return object == objects.front(); }) ) {
270 return objects.front();
271 }
272 auto layout = objects.front()->layout;
273 for ( auto& object : objects ) {
274 layout = widen(layout, object->layout);
275 }
276 auto result = std::make_shared<Object>();
277 result->values.resize(layout->size);
278 // a value is merged as a status value is: the first defined value is kept, and a conflict with it makes the
279 // value undefined for good
280 std::vector<bool> conflict(layout->size, false);
281 for ( auto& object : objects ) {
282 correspond(*object->layout, 0, *layout, 0, [&](size_t source, size_t target) {
283 auto& value = object->values[source];
284 auto& merged = result->values[target];
285 if ( conflict[target] ) {
286 return;
287 }
288 if ( !merged.has_value() ) {
289 merged = value;
290 }
291 else if ( value.has_value() && value.value() != merged.value() ) {
292 merged = std::nullopt;
293 conflict[target] = true;
294 }
295 });
296 }
297 result->layout = std::move(layout);
298 return result;
299}

333/// Returns the layout of the object converted to the schema from a source whose layout the given pointer holds, or, if
334/// no source is given, of an object of the schema with every open array having size 0. Every part of the layout equal
335/// to the corresponding part of the layout of the source is that part, so that it is shared.
336std::shared_ptr<const Layout> determineLayout(const std::shared_ptr<const Schema>& schema, const std::shared_ptr<const Layout>* source, const std::string& path) {
337 auto share = [source](std::shared_ptr<const Layout> layout) {
338 return ( source && *layout == **source ) ? *source : layout;
339 };
340 const Schema* sourceSchema = source ? (*source)->schema.get() : nullptr;
341 if ( sourceSchema && countArrays(*sourceSchema) != countArrays(*schema) ) {
342 throw std::runtime_error("Object: '" + path + "' has " + std::to_string(countArrays(*sourceSchema)) + " dimensions instead of " + std::to_string(countArrays(*schema)));
343 }
344 if ( auto array = std::get_if<ArrayType>(schema.get()) ) {
345 // a fixed array has its size, an open array the size of the source
346 auto sourceArray = source ? &std::get<ArrayLayout>(**source) : nullptr;
347 auto element = determineLayout(array->type, sourceArray ? &sourceArray->layout : nullptr, path);
348 return share( std::make_shared<const Layout>(schema, element, array->size.value_or( sourceArray ? sourceArray->size : 0 )) );
349 }
350 if ( auto valueType = std::get_if<ValueType>(schema.get()) ) {
351 if ( sourceSchema ) {
352 auto sourceType = std::get_if<ValueType>(sourceSchema);
353 if ( !sourceType ) {
354 throw std::runtime_error("Object: '" + path + "' has fields where a value is expected");
355 }
356 // arrays and objects are not cast
357 if ( *valueType != *sourceType ) {
358 throw std::runtime_error("Object: '" + path + "' cannot be converted to the type of the attribute");
359 }
360 }
361 return share( std::make_shared<const Layout>(schema) );
362 }
363 auto& fields = std::get<ObjectType>(*schema).fields;
364 const std::vector<Field>* sourceFields = nullptr;
365 if ( sourceSchema ) {
366 if ( std::holds_alternative<ValueType>(*sourceSchema) ) {
367 throw std::runtime_error("Object: '" + path + "' is a value where fields are expected");
368 }
369 sourceFields = &std::get<ObjectType>(*sourceSchema).fields;
370 for ( auto& sourceField : *sourceFields ) {
371 if ( std::ranges::none_of(fields, [&sourceField](auto& field) { return field.name == sourceField.name; }) ) {
372 throw std::runtime_error("Object: '" + path + "' has the unknown field '" + sourceField.name + "'");
373 }
374 }
375 }
376 std::vector< std::shared_ptr<const Layout> > layouts;
377 layouts.reserve(fields.size());
378 for ( auto& field : fields ) {
379 const std::shared_ptr<const Layout>* sourceField = nullptr;
380 if ( sourceFields ) {
381 if ( auto it = std::ranges::find_if(*sourceFields, [&field](auto& candidate) { return candidate.name == field.name; }); it != sourceFields->end() ) {
382 sourceField = &std::get<ObjectLayout>(**source).fields[(size_t)std::distance(sourceFields->begin(), it)].layout;
383 }
384 }
385 layouts.push_back( determineLayout(field.type, sourceField, path + "." + field.name) );
386 }
387 return share( std::make_shared<const Layout>(schema, layouts) );
388}

390/// Copies the values of a source of the layout `from` into an object of the layout `to`, and rejects a source whose
391/// elements beyond the size of a fixed array hold a defined value.
392void copy(const Layout& from, const std::vector<Value>& source, size_t sourceOffset, const Layout& to, std::vector<Value>& target, size_t targetOffset, const std::string& path) {
393 if ( auto array = std::get_if<ArrayLayout>(&from) ) {
394 auto& toArray = std::get<ArrayLayout>(to);
395 size_t span = array->layout->size;
396 for ( size_t i = 0; i < array->size; i++ ) {
397 if ( i < toArray.size ) {
398 copy(*array->layout, source, sourceOffset + i * span, *toArray.layout, target, targetOffset + i * toArray.layout->size, path);
399 }
400 else if ( std::any_of(source.begin() + (long)(sourceOffset + i * span), source.begin() + (long)(sourceOffset + (i + 1) * span), [](auto& value) { return value.has_value(); }) ) {
401 // only undefined values are cut off
402 throw std::runtime_error("Object: '" + path + "' has " + std::to_string(array->size) + " elements where " + std::to_string(toArray.size) + " are declared");
403 }
404 }
405 return;
406 }
407 if ( std::holds_alternative<std::monostate>(from) ) {
408 target[targetOffset] = source[sourceOffset];
409 return;
410 }
411 auto& fields = std::get<ObjectLayout>(from).fields;
412 auto& types = std::get<ObjectType>(*from.schema).fields;
413 auto& toFields = std::get<ObjectLayout>(to).fields;
414 auto& toTypes = std::get<ObjectType>(*to.schema).fields;
415 for ( size_t k = 0; k < fields.size(); k++ ) {
416 auto& name = types[k].name;
417 auto it = std::ranges::find_if(toTypes, [&name](auto& candidate) { return candidate.name == name; });
418 auto& toField = toFields[(size_t)std::distance(toTypes.begin(), it)];
419 copy(*fields[k].layout, source, sourceOffset + fields[k].offset, *toField.layout, target, targetOffset + toField.offset, path + "." + name);
420 }
421}

425/// Returns the layout of an object of the schema holding the JSON values standing where the schema applies, every
426/// array at an open array having the same size.
427std::shared_ptr<const Layout> determineLayout(const std::shared_ptr<const Schema>& schema, const std::vector<const Json*>& values, const std::string& path) {
428 if ( auto array = std::get_if<ArrayType>(schema.get()) ) {
429 std::optional<size_t> size;
430 std::vector<const Json*> elements;
431 for ( auto value : values ) {
432 if ( value->is_null() ) {
433 continue;
434 }
435 if ( !value->is_array() ) {
436 throw std::runtime_error("Object: '" + path + "' holds '" + value->dump() + "' where an array is expected");
437 }
438 if ( array->size.has_value() ) {
439 if ( value->size() > array->size.value() ) {
440 throw std::runtime_error("Object: '" + path + "' has " + std::to_string(value->size()) + " elements where " + std::to_string(array->size.value()) + " are declared");
441 }
442 }
443 else if ( size.has_value() && value->size() != size.value() ) {
444 throw std::runtime_error("Object: '" + path + "' holds arrays of different sizes");
445 }
446 size = value->size();
447 for ( auto& element : *value ) {
448 elements.push_back(&element);
449 }
450 }
451 auto element = determineLayout(array->type, elements, path);
452 return std::make_shared<const Layout>(schema, element, array->size.value_or( size.value_or(0) ));
453 }
454 if ( std::holds_alternative<ValueType>(*schema) ) {
455 return std::make_shared<const Layout>(schema);
456 }
457 auto& fields = std::get<ObjectType>(*schema).fields;
458 for ( auto value : values ) {
459 if ( value->is_null() ) {
460 continue;
461 }
462 if ( !value->is_object() ) {
463 throw std::runtime_error("Object: '" + path + "' holds '" + value->dump() + "' where fields are expected");
464 }
465 for ( auto& [key, _] : value->items() ) {
466 if ( std::ranges::none_of(fields, [&key](auto& field) { return field.name == key; }) ) {
467 throw std::runtime_error("Object: '" + path + "' has the unknown field '" + key + "'");
468 }
469 }
470 }
471 std::vector< std::shared_ptr<const Layout> > layouts;
472 layouts.reserve(fields.size());
473 for ( auto& field : fields ) {
474 std::vector<const Json*> nested;
475 for ( auto value : values ) {
476 if ( !value->is_null() ) {
477 if ( auto it = value->find(field.name); it != value->end() ) {
478 nested.push_back(&*it);
479 }
480 }
481 }
482 layouts.push_back( determineLayout(field.type, nested, path + "." + field.name) );
483 }
484 return std::make_shared<const Layout>(schema, layouts);
485}

487/// Returns the single value a JSON value states for the given value type.
488Value convert(const Json& value, ValueType type, const std::string& path) {
489 if ( value.is_null() ) {
490 return std::nullopt;
491 }
492 switch ( type ) {
493 case ValueType::BOOLEAN:
494 if ( value.is_boolean() ) {
495 return number( value.get<bool>() ? 1 : 0 );
496 }
497 break;
498 case ValueType::INTEGER:
499 if ( value.is_number() ) {
500 return number( (long)value.get<double>() );
501 }
502 break;
503 case ValueType::DECIMAL:
504 if ( value.is_number() ) {
505 return number( value.get<double>() );
506 }
507 break;
508 case ValueType::STRING:
509 if ( value.is_string() ) {
510 return number( stringRegistry( value.get<std::string>() ) );
511 }
512 break;
513 }
514 throw std::runtime_error("Object: '" + path + "' holds '" + value.dump() + "', which is no value of its type");
515}

517/// Writes the values a JSON value states into an object of the given layout, beginning at the given offset.
518void fill(const Layout& layout, const Json& value, std::vector<Value>& values, size_t offset, const std::string& path) {
519 if ( value.is_null() ) {
520 return;
521 }
522 if ( auto array = std::get_if<ArrayLayout>(&layout) ) {
523 size_t i = 0;
524 for ( auto& element : value ) {
525 fill(*array->layout, element, values, offset + i * array->layout->size, path);
526 ++i;
527 }
528 return;
529 }
530 if ( std::holds_alternative<std::monostate>(layout) ) {
531 values[offset] = convert(value, std::get<ValueType>(*layout.schema), path);
532 return;
533 }
534 auto& fields = std::get<ObjectLayout>(layout).fields;
535 auto& types = std::get<ObjectType>(*layout.schema).fields;
536 for ( size_t k = 0; k < fields.size(); k++ ) {
537 if ( auto it = value.find(types[k].name); it != value.end() ) {
538 fill(*fields[k].layout, *it, values, offset + fields[k].offset, path + "." + types[k].name);
539 }
540 }
