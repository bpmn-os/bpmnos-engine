61/// Renders the values of a layout beginning at the given offset, from the given dimension on.
62std::string render(const Object::Layout& layout, const std::vector<Value>& values, size_t offset, size_t dimension) {
63 if ( dimension < layout.dimensions.size() ) {
64 // the elements along this dimension, each holding the remaining dimensions
65 size_t span = layout.stride;
66 for ( size_t d = dimension + 1; d < layout.dimensions.size(); d++ ) {
67 span *= layout.dimensions[d];
68 }
69 if ( layout.dimensions[dimension] == 0 ) {
70 return "[ ]";
71 }
72 std::string result;
73 for ( size_t i = 0; i < layout.dimensions[dimension]; i++ ) {
74 result += ", " + render(layout, values, offset + i * span, dimension + 1);
75 }
76 result.front() = '[';
77 return result + " ]";
78 }
79 if ( layout.scalar.has_value() ) {
80 auto& value = values[offset];
81 if ( !value.has_value() ) {
82 return "undefined";
83 }
84 auto text = BPMNOS::to_string(value.value(), layout.scalar.value());
85 return layout.scalar.value() == ValueType::STRING ? "\"" + text + "\"" : text;
86 }
87 std::string result = "{ ";
88 for ( size_t i = 0; i < layout.fields.size(); i++ ) {
89 auto& field = layout.fields[i];
90 result += ( i ? ", " : "" ) + field.name + " := " + render(field.layout, values, offset + field.offset, 0);
91 }
92 return result + " }";
93}

108/**
109 * Visits the values of an object of the layout `from` that an object of the layout `to`, of the same schema,
110 * also has, calling `visit(source, target)` with the positions of each such value in the two objects.
111 */
112void correspond(const Object::Layout& from, size_t fromOffset, const Object::Layout& to, size_t toOffset, const std::function<void(size_t, size_t)>& visit) {
113 std::function<void(size_t, size_t, size_t)> walk = [&](size_t dimension, size_t source, size_t target) {
114 if ( dimension == from.dimensions.size() ) {
115 if ( from.scalar.has_value() ) {
116 visit(source, target);
117 return;
118 }
119 for ( size_t k = 0; k < from.fields.size(); k++ ) {
120 correspond(from.fields[k].layout, source + from.fields[k].offset, to.fields[k].layout, target + to.fields[k].offset, visit);
121 }
122 return;
123 }
124 size_t count = std::min(from.dimensions[dimension], to.dimensions[dimension]);
125 size_t fromSpan = span(from, dimension);
126 size_t toSpan = span(to, dimension);
127 for ( size_t i = 0; i < count; i++ ) {
128 walk(dimension + 1, source + i * fromSpan, target + i * toSpan);
129 }
130 };
131 walk(0, fromOffset, toOffset);
132}

134/// Widens a layout so that every dimension has at least the length it has in the other layout.
135void widen(Object::Layout& layout, const Object::Layout& other) {
136 for ( size_t d = 0; d < layout.dimensions.size(); d++ ) {
137 layout.dimensions[d] = std::max(layout.dimensions[d], other.dimensions[d]);
138 }
139 for ( size_t k = 0; k < layout.fields.size(); k++ ) {
140 widen(layout.fields[k].layout, other.fields[k].layout);
141 }
142 layout.arrange();
143}

155std::shared_ptr<Object> resized(const Object& object, const std::vector<std::string>& fields, size_t dimension, size_t length) {
156 auto layout = std::make_shared<Object::Layout>(*object.layout);
157 // the layouts from the base to the field whose dimension changes
158 std::vector<Object::Layout*> chain = { layout.get() };
159 for ( auto& name : fields ) {
160 auto& candidates = chain.back()->fields;
161 auto field = std::ranges::find_if(candidates, [&name](auto& candidate) { return candidate.name == name; });
162 if ( field == candidates.end() ) {
163 throw std::logic_error("Object: unknown field '" + name + "'");
164 }
165 chain.push_back(&field->layout);
166 }
167 chain.back()->dimensions.at(dimension) = length;
168 for ( auto it = chain.rbegin(); it != chain.rend(); ++it ) {
169 (*it)->arrange();
170 }
171 auto result = std::make_shared<Object>();
172 result->values.resize(layout->size());
173 correspond(*object.layout, 0, *layout, 0, [&](size_t source, size_t target) {
174 result->values[target] = object.values[source];
175 });
176 result->layout = std::move(layout);
177 return result;
178}

180std::shared_ptr<const Object> merge(const std::vector< std::shared_ptr<const Object> >& objects) {
181 assert( !objects.empty() );
182 if ( std::ranges::all_of(objects, [&objects](auto& object) { return object == objects.front(); }) ) {
183 return objects.front();
184 }
185 auto layout = std::make_shared<Object::Layout>(*objects.front()->layout);
186 for ( auto& object : objects ) {
187 widen(*layout, *object->layout);
188 }
189 auto result = std::make_shared<Object>();
190 result->values.resize(layout->size());
191 // a value is merged as a status value is: the first defined value is kept, and a conflict with it makes the
192 // value undefined for good
193 std::vector<bool> conflict(layout->size(), false);
194 for ( auto& object : objects ) {
195 correspond(*object->layout, 0, *layout, 0, [&](size_t source, size_t target) {
196 auto& value = object->values[source];
197 auto& merged = result->values[target];
198 if ( conflict[target] ) {
199 return;
200 }
201 if ( !merged.has_value() ) {
202 merged = value;
203 }
204 else if ( value.has_value() && value.value() != merged.value() ) {
205 merged = std::nullopt;
206 conflict[target] = true;
207 }
208 });
209 }
210 result->layout = std::move(layout);
211 return result;
212}

220/// Renders the values of a layout beginning at the given offset, from the given dimension on, as JSON.
221nlohmann::ordered_json json(const Object::Layout& layout, const std::vector<Value>& values, size_t offset, size_t dimension) {
222 if ( dimension < layout.dimensions.size() ) {
223 auto result = nlohmann::ordered_json::array();
224 size_t elementSpan = span(layout, dimension);
225 for ( size_t i = 0; i < layout.dimensions[dimension]; i++ ) {
226 result.push_back( json(layout, values, offset + i * elementSpan, dimension + 1) );
227 }
228 return result;
229 }
230 if ( layout.scalar.has_value() ) {
231 auto& value = values[offset];
232 if ( !value.has_value() ) {
233 return nullptr;
234 }
235 switch ( layout.scalar.value() ) {
236 case ValueType::BOOLEAN:
237 return value.value() != number(0);
238 case ValueType::INTEGER:
239 return (long)value.value();
240 case ValueType::DECIMAL:
241 return (double)value.value();
242 case ValueType::STRING:
243 return BPMNOS::to_string(value.value(), ValueType::STRING);
244 }
245 }
246 auto result = nlohmann::ordered_json::object();
247 for ( auto& field : layout.fields ) {
248 result[field.name] = json(field.layout, values, offset + field.offset, 0);
249 }
250 return result;
251}
