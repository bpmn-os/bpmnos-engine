17 : engine(engine)
18 , scenario(scenario)

20 , contributionsToObjective(other->contributionsToObjective)
21 , globals(other->globals)
22 , instantiationCounter(other->instantiationCounter)
23{
24 // Copy root state machine instances
25 for (const auto& otherInstance : other->instances) {
26 instances.push_back(std::make_shared<StateMachine>(this, nullptr, otherInstance.get()));
27
28 // Populate archive
29 auto key = (long unsigned int)otherInstance->instance.value();
30 if (other->archive.contains(key) && other->archive.at(key).lock().get() == otherInstance.get()) {
31 archive[key] = instances.back();
32 }
33 }
34
35 // Copy messages sent from MessageThrowEvent, messages sent from SendTask are created when copying the associated Token
36 for (const auto& otherMessage : other->messages) {
37 if (!otherMessage->waitingToken) {
38 messages.push_back(std::make_shared<Message>(otherMessage.get()));
39 }
40 }
41
42 // Helper to find new message corresponding to original message
43 auto findNewMessage = [this](const Message* otherMessage) -> std::shared_ptr<Message> {
44 for (const auto& message : messages) {
45 if (message->origin != otherMessage->origin) continue;
46 if (message->recipient != otherMessage->recipient) continue;
47 // Match waitingToken by node (both nullptr or same node)
48 if (otherMessage->waitingToken) {
49 if (!message->waitingToken || message->waitingToken->node != otherMessage->waitingToken->node) continue;
50 }
51 else {
52 if (message->waitingToken) continue;
53 }
54 return message;
55 }
56 assert(false && "Unable to find message!");
57 return nullptr;
58 };
59
60 // Populate outbox (keyed by origin node)
61 for (const auto& [node, otherMessages] : other->outbox) {
62 for (const auto& [otherMessageWeak] : otherMessages) {
63 if (auto otherMessage = otherMessageWeak.lock()) {
64 if (auto message = findNewMessage(otherMessage.get())) {
65 outbox[node].emplace_back(message);
66 }
67 }
68 }
69 }
70
71 // Populate unsent (keyed by recipient ID)
72 for (const auto& [recipientId, otherMessages] : other->unsent) {
73 for (const auto& [otherMessageWeak] : otherMessages) {
74 if (auto otherMessage = otherMessageWeak.lock()) {
75 if (auto message = findNewMessage(otherMessage.get())) {
76 unsent[recipientId].emplace_back(message);
77 }
78 }
79 }
80 }
81
82 // Populate inbox (keyed by StateMachine*, use archive for remapping)
83 for (const auto& [otherStateMachine, otherMessages] : other->inbox) {
84 auto instanceId = (long unsigned int)otherStateMachine->instance.value();
85 auto it = archive.find(instanceId);
86 if (it == archive.end()) continue;
87 auto newStateMachine = it->second.lock();
88 if (!newStateMachine) continue;
89
90 for (const auto& [otherMessageWeak] : otherMessages) {
91 if (auto otherMessage = otherMessageWeak.lock()) {
92 if (auto message = findNewMessage(otherMessage.get())) {
93 inbox[newStateMachine.get()].emplace_back(message);
94 }
95 }
96 }
97 }
98}

123BPMNOS::number SystemState::getWeightedObjective() const {
124 BPMNOS::number result = 0.0;
125 for ( auto& [attribute, value] : contributionsToObjective ) {
126 result += value * attribute->weight;
127 }
128
129 for ( auto& attribute : scenario->getModel()->attributes ) {
130 assert( attribute->category == BPMNOS::Model::Attribute::Category::GLOBAL );
131 auto value = globals[attribute->index];
132 if ( value.has_value() ) {
133 result += value.value() * attribute->weight;
134 }
135 }
136
137 return result;
138}
