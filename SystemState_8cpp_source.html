8SystemState::SystemState(const Engine* engine, const Scenario* scenario, BPMNOS::number currentTime)
9 : engine(engine)
10 , scenario(scenario)
11 , currentTime(currentTime)
12 , objective(0)
13 , stateMachine(std::make_shared<StateMachine>(this, scenario->dataProvider->getGlobals(*scenario)))
14{
15 // the values the globals are created with never pass through setValue, so the objective is seeded with
16 // them here; no change is notified, the run not having begun and no token being able to observe it
17 auto& globals = stateMachine->ownedData;
18 for ( auto& attribute : engine->getModel()->attributes ) {
19 if ( attribute->weight != 0 && globals[attribute->index].has_value() ) {
20 objective += globals[attribute->index].value() * attribute->weight;
21 }
22 }
23}

25SystemState::SystemState(const Engine* engine, const Scenario* scenario, const SystemState* other)
26 : engine(engine)
27 , scenario(scenario)
28 , currentTime(other->currentTime)
29 , objective(other->objective)
30 , instantiationCounter(other->instantiationCounter)
31{
32 // Copy the global state machine, which copies the token at the process of each instance together with
33 // the state machine of the instance it owns
34 stateMachine = std::make_shared<StateMachine>(this, nullptr, other->stateMachine.get());
35
36 // Populate archive with the state machines of the instances, the tokens being copied in their order
37 for ( size_t i = 0; i < stateMachine->tokens.size(); i++ ) {
38 auto otherInstance = other->stateMachine->tokens[i]->owned.get();
39 auto key = (long unsigned int)otherInstance->instance.value();
40 if (other->archive.contains(key) && other->archive.at(key).lock().get() == otherInstance) {
41 archive[key] = stateMachine->tokens[i]->owned;
42 }
43 }
44
45 // Copy messages sent from MessageThrowEvent, messages sent from SendTask are created when copying the associated Token
46 for (const auto& otherMessage : other->messages) {
47 if (!otherMessage->waitingToken) {
48 messages.push_back(std::make_shared<Message>(otherMessage.get()));
49 }
50 }
51
52 // Helper to find new message corresponding to original message
53 auto findNewMessage = [this](const Message* otherMessage) -> std::shared_ptr<Message> {
54 for (const auto& message : messages) {
55 if (message->origin != otherMessage->origin) continue;
56 if (message->recipient != otherMessage->recipient) continue;
57 // Match waitingToken by node (both nullptr or same node)
58 if (otherMessage->waitingToken) {
59 if (!message->waitingToken || message->waitingToken->node != otherMessage->waitingToken->node) continue;
60 }
61 else {
62 if (message->waitingToken) continue;
63 }
64 return message;
65 }
66 assert(false && "Unable to find message!");
67 return nullptr;
68 };
69
70 // Populate outbox (keyed by origin node)
71 for (const auto& [node, otherMessages] : other->outbox) {
72 for (const auto& [otherMessageWeak] : otherMessages) {
73 if (auto otherMessage = otherMessageWeak.lock()) {
74 if (auto message = findNewMessage(otherMessage.get())) {
75 outbox[node].emplace_back(message);
76 }
77 }
78 }
79 }
80
81 // Populate unsent (keyed by recipient ID)
82 for (const auto& [recipientId, otherMessages] : other->unsent) {
83 for (const auto& [otherMessageWeak] : otherMessages) {
84 if (auto otherMessage = otherMessageWeak.lock()) {
85 if (auto message = findNewMessage(otherMessage.get())) {
86 unsent[recipientId].emplace_back(message);
87 }
88 }
89 }
90 }
91
92 // Populate inbox (keyed by StateMachine*, use archive for remapping)
93 for (const auto& [otherStateMachine, otherMessages] : other->inbox) {
94 auto instanceId = (long unsigned int)otherStateMachine->instance.value();
95 auto it = archive.find(instanceId);
96 if (it == archive.end()) continue;
97 auto newStateMachine = it->second.lock();
98 if (!newStateMachine) continue;
99
100 for (const auto& [otherMessageWeak] : otherMessages) {
101 if (auto otherMessage = otherMessageWeak.lock()) {
102 if (auto message = findNewMessage(otherMessage.get())) {
103 inbox[newStateMachine.get()].emplace_back(message);
104 }
105 }
106 }
107 }
108}
