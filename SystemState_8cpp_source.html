7SystemState::SystemState(const Engine* engine, const BPMNOS::Model::Scenario* scenario, BPMNOS::number currentTime)
8 : engine(engine)
9 , scenario(scenario)
10 , currentTime(currentTime)
11 , globals(scenario->globals)
12{
13 // the values the globals are created with never pass through setValue, so the objective is seeded with
14 // them here; no data update is notified, the run not having begun and no token being able to observe it
15 for ( auto& attribute : scenario->getModel()->attributes ) {
16 assert( attribute->category == BPMNOS::Model::Attribute::Category::GLOBAL );
17 if ( attribute->weight != 0 && globals[attribute->index].has_value() ) {
18 globals[BPMNOS::Model::ExtensionElements::Index::Objective].value() +=
19 globals[attribute->index].value() * attribute->weight;
20 }
21 }
22}

24SystemState::SystemState(const Engine* engine, const BPMNOS::Model::Scenario* scenario, const SystemState* other)
25 : engine(engine)
26 , scenario(scenario)
27 , currentTime(other->currentTime)
28 , globals(other->globals)
29 , instantiationCounter(other->instantiationCounter)
30{
31 // Copy root state machine instances
32 for (const auto& otherInstance : other->instances) {
33 instances.push_back(std::make_shared<StateMachine>(this, nullptr, otherInstance.get()));
34
35 // Populate archive
36 auto key = (long unsigned int)otherInstance->instance.value();
37 if (other->archive.contains(key) && other->archive.at(key).lock().get() == otherInstance.get()) {
38 archive[key] = instances.back();
39 }
40 }
41
42 // Copy messages sent from MessageThrowEvent, messages sent from SendTask are created when copying the associated Token
43 for (const auto& otherMessage : other->messages) {
44 if (!otherMessage->waitingToken) {
45 messages.push_back(std::make_shared<Message>(otherMessage.get()));
46 }
47 }
48
49 // Helper to find new message corresponding to original message
50 auto findNewMessage = [this](const Message* otherMessage) -> std::shared_ptr<Message> {
51 for (const auto& message : messages) {
52 if (message->origin != otherMessage->origin) continue;
53 if (message->recipient != otherMessage->recipient) continue;
54 // Match waitingToken by node (both nullptr or same node)
55 if (otherMessage->waitingToken) {
56 if (!message->waitingToken || message->waitingToken->node != otherMessage->waitingToken->node) continue;
57 }
58 else {
59 if (message->waitingToken) continue;
60 }
61 return message;
62 }
63 assert(false && "Unable to find message!");
64 return nullptr;
65 };
66
67 // Populate outbox (keyed by origin node)
68 for (const auto& [node, otherMessages] : other->outbox) {
69 for (const auto& [otherMessageWeak] : otherMessages) {
70 if (auto otherMessage = otherMessageWeak.lock()) {
71 if (auto message = findNewMessage(otherMessage.get())) {
72 outbox[node].emplace_back(message);
73 }
74 }
75 }
76 }
77
78 // Populate unsent (keyed by recipient ID)
79 for (const auto& [recipientId, otherMessages] : other->unsent) {
80 for (const auto& [otherMessageWeak] : otherMessages) {
81 if (auto otherMessage = otherMessageWeak.lock()) {
82 if (auto message = findNewMessage(otherMessage.get())) {
83 unsent[recipientId].emplace_back(message);
84 }
85 }
86 }
87 }
88
89 // Populate inbox (keyed by StateMachine*, use archive for remapping)
90 for (const auto& [otherStateMachine, otherMessages] : other->inbox) {
91 auto instanceId = (long unsigned int)otherStateMachine->instance.value();
92 auto it = archive.find(instanceId);
93 if (it == archive.end()) continue;
94 auto newStateMachine = it->second.lock();
95 if (!newStateMachine) continue;
96
97 for (const auto& [otherMessageWeak] : otherMessages) {
98 if (auto otherMessage = otherMessageWeak.lock()) {
99 if (auto message = findNewMessage(otherMessage.get())) {
100 inbox[newStateMachine.get()].emplace_back(message);
101 }
102 }
103 }
104 }
105}
