24void TaskCompletionHandler::notice(const Observable* observable) {
25 if (observable->getObservableType() == Observable::Type::SystemState) {
26 // a freshly installed state (e.g. on resume): reset and rebuild from the tokens it lists as awaiting a completion event
27 auto systemState = static_cast<const SystemState*>(observable);
28 lastCheckedTime = std::numeric_limits<BPMNOS::number>::lowest();
29 awaitingTokens.clear();
30 pendingEvents.clear();
31 for (auto element : systemState->tokensAwaitingCompletionEvent) {
32 awaitingTokens.emplace_back(std::get<1>(element));
33 }
34 return;
35 }
36 if (observable->getObservableType() != Observable::Type::Token) {
37 return;
38 }
39
40 auto token = static_cast<const Token*>(observable);
41
42 // Filter: Task nodes in BUSY state, excluding special tasks
43 if (!token->node ||
44 !token->node->represents<BPMN::Task>() ||
45 token->state != Token::State::BUSY) {
46 return;
47 }
48
49 // Exclude tasks that complete via external events
50 if (token->node->represents<BPMN::SendTask>() ||
51 token->node->represents<BPMN::ReceiveTask>() ||
52 token->node->represents<BPMNOS::Model::DecisionTask>()) {
53 return;
54 }
55
56 auto systemState = token->owner->systemState;
57 auto currentTime = systemState->getTime();
58 auto token_ptr = const_cast<Token*>(token)->weak_from_this();
59
60 // If we've already checked at this time, check new token immediately
61 if (lastCheckedTime == currentTime) {
62 if (auto event = getEvent(token, systemState)) {
63 pendingEvents.emplace_back(token_ptr, event);
64 }
65 else {
66 awaitingTokens.emplace_back(token_ptr);
67 }
68 }
69 else {
70 awaitingTokens.emplace_back(token_ptr);
71 }
72}

86std::shared_ptr<Event> TaskCompletionHandler::dispatchEvent(const SystemState* systemState) {
87 // Dispatch from pending first
88 for (auto& [token_ptr, event] : pendingEvents) {
89 if (auto token = token_ptr.lock()) {
90 auto result = event; // Copy before remove destroys the tuple
91 pendingEvents.remove(token.get());
92 return result;
93 }
94 }
95

103 std::vector<Token*> completedTokens;

106 if (auto event = getEvent(token.get(), systemState)) {
107 completedTokens.push_back(token.get());
108 pendingEvents.emplace_back(token, event);

110 }
111 }
