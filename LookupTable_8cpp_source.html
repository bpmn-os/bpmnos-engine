24BPMNOS::CSVReader LookupTable::openCsv(const std::string& filename, const std::vector<std::string>& folders) {
25 // First, try to open the file using the given filename.
26 if (std::filesystem::exists(filename)) {
27 return CSVReader(filename);
28 }
29
30 // If the file is not found with the given filename, try each folder in the list.
31 for (const std::string& folder : folders) {
32 std::filesystem::path fullPath = std::filesystem::path(folder) / filename;
33 if (std::filesystem::exists(fullPath)) {
34 return CSVReader(fullPath.string());
35 }
36 }
37
38 // If the file is not found in any of the folders, throw an exception with all searched locations
39 std::string errorMsg = "LookupTable: CSV file '" + filename + "' not found in:\n" + std::filesystem::current_path().string();
40 for (const std::string& folder : folders) {
41 errorMsg += "\n" + std::filesystem::absolute(folder).string();
42 }
43 throw std::runtime_error(errorMsg);
44}

46void LookupTable::validateHeader(const std::string& sourceLabel, const CSVReader::Row& headerRow) const {
47 // Normalise a column name for comparison: trim surrounding whitespace and fold to lower case.
48 auto normalise = [](std::string s) {
49 BPMNOS::trim(s);
50 std::transform(s.begin(), s.end(), s.begin(), [](unsigned char c) { return (char)std::tolower(c); });
51 return s;
52 };
53
54 // Render a header cell as its column name (CSVReader may have parsed a numeric-looking name as a number).
55 auto cellName = [](const CSVReader::Row::value_type& cell) -> std::string {
56 if ( std::holds_alternative<std::string>(cell) ) {
57 return std::get<std::string>(cell);
58 }
59 return std::format("{}", (double)std::get<BPMNOS::number>(cell));
60 };
61
62 // Expected column names come from the semicolon separated header attribute.
63 auto expected = BPMNOS::split(header, ';');
64
65 if ( expected.size() != headerRow.size() ) {
66 throw std::runtime_error(std::format(
67 "LookupTable: table '{}' with source '{}' expects {} column(s) ('{}') but the CSV header has {}",
68 name, sourceLabel, expected.size(), header, headerRow.size()));
69 }
70
71 for ( size_t i = 0; i < expected.size(); i++ ) {
72 auto actual = cellName(headerRow[i]);
73 if ( normalise(expected[i]) != normalise(actual) ) {
74 throw std::runtime_error(
75 std::format("LookupTable: table '{}' with source '{}' expects column {} to be '{}' but the CSV header has '{}'", name, sourceLabel, i + 1, BPMNOS::trim_copy(expected[i]), BPMNOS::trim_copy(actual))
76 );
77 }
78 }
79}

81void LookupTable::populate(const std::string& sourceLabel, CSVReader::Table table) {
82 if ( table.empty() ) {
83 throw std::runtime_error(std::format("LookupTable: table '{}' with source '{}' is empty", name, sourceLabel));
84 }
85 // validate the CSV header line (index 0) against the expected column names
86 validateHeader(sourceLabel, table[0]);
87 // populate lookup map
88 for (size_t j = 1; j < table.size(); j++) { // assume a single header line at index 0
89 auto& row = table[j];
90 std::vector< double > inputs;
91 size_t columns = row.size();
92
93 for ( size_t i = 0; i < columns - 1; i++ ) {
94 auto& cell = row[i];
95 if ( !std::holds_alternative<BPMNOS::number>(cell) ) {
96 throw std::runtime_error(std::format("LookupTable: illegal input in table '{}' at row {}, column {}", name, j, i));
97 }
98//std::cerr << (double)std::get<BPMNOS::number>(cell) << ", ";
99 inputs.push_back( (double)std::get<BPMNOS::number>(cell) );
100 }
101 auto& cell = row[columns - 1];
102 if ( !std::holds_alternative<BPMNOS::number>(cell) ) {
103 std::visit([](auto&& value) { std::cerr << "Value: " << value << " "; }, cell);
104 throw std::runtime_error(std::format("LookupTable: illegal output in table '{}' at row {}, column {}", name, j, columns - 1));
105 }
106//std::cerr << "-> " << (double)std::get<BPMNOS::number>(cell) << std::endl;
107 auto result = (double)std::get<BPMNOS::number>(cell);
108 lookupMap.emplace( std::move(inputs), result );
109 }
