27std::vector< std::pair<std::string, BPMNOS::Schema> > LookupTable::parseColumns() const {
28 std::vector< std::pair<std::string, Schema> > result;
29 for ( auto& column : BPMNOS::split(header, ';') ) {
30 auto colon = column.find(':');
31 if ( colon == std::string::npos ) {
32 throw std::runtime_error(std::format("LookupTable: column '{}' of table '{}' requires a type, 'name: type'", BPMNOS::trim_copy(column), name));
33 }
34 auto columnName = BPMNOS::trim_copy(column.substr(0, colon));
35 Schema schema;
36 try {
37 schema = Schema::parse(BPMNOS::trim_copy(column.substr(colon + 1)));
38 }
39 catch ( const std::exception& error ) {
40 throw std::runtime_error(std::format("LookupTable: illegal type of column '{}' of table '{}'.\n{}", columnName, name, error.what()));
41 }
42 result.emplace_back(columnName, std::move(schema));
43 }
44 if ( result.size() < 2 ) {
45 throw std::runtime_error(std::format("LookupTable: table '{}' requires a key and a result column", name));
46 }
47 for ( size_t i = 0; i + 1 < result.size(); i++ ) {
48 if ( !std::holds_alternative<ValueType>(result[i].second) ) {
49 throw std::runtime_error(std::format("LookupTable: key column '{}' of table '{}' must have a scalar type", result[i].first, name));
50 }
51 }
52 return result;
53}

55BPMNOS::CSVReader LookupTable::openCsv(const std::string& filename, const std::vector<std::string>& folders) {
56 // First, try to open the file using the given filename.
57 if (std::filesystem::exists(filename)) {
58 return CSVReader(filename);
59 }
60
61 // If the file is not found with the given filename, try each folder in the list.
62 for (const std::string& folder : folders) {
63 std::filesystem::path fullPath = std::filesystem::path(folder) / filename;
64 if (std::filesystem::exists(fullPath)) {
65 return CSVReader(fullPath.string());
66 }
67 }
68
69 // If the file is not found in any of the folders, throw an exception with all searched locations
70 std::string errorMsg = "LookupTable: CSV file '" + filename + "' not found in:\n" + std::filesystem::current_path().string();
71 for (const std::string& folder : folders) {
72 errorMsg += "\n" + std::filesystem::absolute(folder).string();
73 }
74 throw std::runtime_error(errorMsg);

77void LookupTable::validateHeader(const std::string& sourceLabel, const CSVReader::Row& headerRow) const {
78 // Normalise a column name for comparison: trim surrounding whitespace and fold to lower case.
79 auto normalise = [](std::string s) {
80 BPMNOS::trim(s);
81 std::transform(s.begin(), s.end(), s.begin(), [](unsigned char c) { return (char)std::tolower(c); });
82 return s;
83 };
84
85 // Render a header cell as its column name (CSVReader may have parsed a numeric-looking name as a number).
86 auto cellName = [](const CSVReader::Row::value_type& cell) -> std::string {
87 if ( std::holds_alternative<std::string>(cell) ) {
88 return std::get<std::string>(cell);
89 }
90 return std::format("{}", (double)std::get<BPMNOS::number>(cell));
91 };
92
93 // Expected column names come from the semicolon separated header attribute.
94 std::vector<std::string> expected;
95 for ( auto& [columnName, _] : columns ) {
96 expected.push_back(columnName);
97 }
98
99 if ( expected.size() != headerRow.size() ) {
100 throw std::runtime_error(std::format(
101 "LookupTable: table '{}' with source '{}' expects {} column(s) ('{}') but the CSV header has {}",
102 name, sourceLabel, expected.size(), header, headerRow.size()));
103 }
104
105 for ( size_t i = 0; i < expected.size(); i++ ) {
106 auto actual = cellName(headerRow[i]);
107 if ( normalise(expected[i]) != normalise(actual) ) {
108 throw std::runtime_error(
109 std::format("LookupTable: table '{}' with source '{}' expects column {} to be '{}' but the CSV header has '{}'", name, sourceLabel, i + 1, BPMNOS::trim_copy(expected[i]), BPMNOS::trim_copy(actual))
110 );
111 }
