20std::vector<const Attribute*> Deferral::getObservables(XML::bpmnos::tDeferral* deferral, const AttributeRegistry& attributeRegistry) {
21 std::vector<const Attribute*> observables;
22 for ( XML::bpmnos::tParameter& parameter : deferral->parameter ) {
23 if ( parameter.name.value.value != "observable" ) {
24 throw std::runtime_error("Deferral: illegal parameter '" + parameter.name.value.value + "'");
25 }
26 std::string name = parameter.value.has_value() ? BPMNOS::trim_copy(parameter.value->get().value) : std::string();
27 if ( name.empty() ) {
28 throw std::runtime_error("Deferral: observable without attribute");
29 }
30 // an observed attribute is a data or global attribute in the scope of the decision point; a status attribute never
31 // reaches a data update, and whether data is immutable is known only once the whole model is read
32 if ( !attributeRegistry.contains(name) ) {
33 throw std::runtime_error("Deferral: observable '" + name + "' is no attribute in scope");
34 }
35 auto attribute = attributeRegistry[name];
36 if ( attribute->category != Attribute::Category::DATA ) {
37 throw std::runtime_error("Deferral: observable '" + name + "' is no data or global attribute");
38 }
39 if ( std::ranges::contains(observables, attribute) ) {
40 throw std::runtime_error("Deferral: observable '" + name + "' stated twice");
41 }
42 observables.push_back(attribute);
43 }
44 return observables;
45}
