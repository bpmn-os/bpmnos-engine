17/**
18 * @brief Iterable union of the contested entry and message-delivery candidate decisions, ordered by reward.
19 *
20 * Owns a candidate collection of sequential-ad-hoc-subprocess child entries and one of message deliveries,
21 * neither of which has precedence over the other. Iterating yields their two reward-ordered ranges as a single
22 * reward-ordered range by a lazy two-way merge: at each step the better of the two fronts is yielded and that
23 * side advanced, so the global best feasible decision is simply the front — without merging into a container or
24 * re-sorting. Each yielded element carries just the reward, the decision's weak Event, and its weak Evaluation,
25 * which is all a dispatcher needs to dispatch (and, for a rollout, to force) the decision.
26 */
27class CompetingCandidates {
28public:
29 CompetingCandidates(Evaluator* evaluator)
30 : sequentialEntries(evaluator)
31 , messageDeliveries(evaluator)
32 {
33 }
34
35 /// Connect both owned candidate collections so each registers itself for the observables it needs.
36 void connect(Notifier* notifier) {
37 sequentialEntries.connect(notifier);
38 messageDeliveries.connect(notifier);
39 }
40
41 /// Element yielded by the merge: reward, the decision's weak Event (second-to-last), its weak Evaluation (last).
42 using value_type = std::tuple< double, std::weak_ptr<Event>, std::weak_ptr<Evaluation> >;
43
44private:
45 SequentialEntries sequentialEntries;
46 MessageDeliveries messageDeliveries;
47
48 using EntryIterator = decltype( std::declval<SequentialEntries&>().begin() );
49 using MessageIterator = decltype( std::declval<MessageDeliveries&>().begin() );
50
51 /// Project a candidate tuple of either collection onto (reward, weak Event, weak Evaluation).
52 template <typename Tuple>
53 static value_type project(const Tuple& candidate) {
54 constexpr std::size_t size = std::tuple_size_v<Tuple>;
55 return value_type{ std::get<0>(candidate), std::get<size - 2>(candidate), std::get<size - 1>(candidate) };
56 }
57
58public:
59 /// End marker for the merge.
60 struct Sentinel {};
61
62 /// Lazy two-way merge of the two reward-ordered ranges; yields the better front, advances that side.
63 class Iterator {
64 public:
65 Iterator(EntryIterator entry, EntryIterator entryEnd, MessageIterator message, MessageIterator messageEnd)
66 : entry(entry), entryEnd(entryEnd), message(message), messageEnd(messageEnd)
67 {
68 }
69
70 bool operator!=(Sentinel) const { return entry != entryEnd || message != messageEnd; }
71
72 value_type operator*() const { return preferEntry() ? project(*entry) : project(*message); }
73
74 Iterator& operator++() {
75 if ( preferEntry() ) ++entry; else ++message;
76 return *this;
77 }
78
79 private:
80 /// Yield the entry front while it exists and is at least as good as the message front (reward-descending).
81 bool preferEntry() const {
82 if ( !(entry != entryEnd) ) return false;
83 if ( !(message != messageEnd) ) return true;
84 return std::get<0>(*entry) >= std::get<0>(*message);
85 }
86
87 EntryIterator entry, entryEnd;
88 MessageIterator message, messageEnd;
89 };
90
91 /// Iterating triggers each collection's lazy evaluation (via its begin) and merges the results.
92 Iterator begin() {
93 return Iterator( sequentialEntries.begin(), sequentialEntries.end(),
94 messageDeliveries.begin(), messageDeliveries.end() );
95 }
96 Sentinel end() { return {}; }
97};
