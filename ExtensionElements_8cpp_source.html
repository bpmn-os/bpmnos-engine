479template <typename DataType>
480BPMNOS::number ExtensionElements::applyOperators(BPMNOS::Status& status, DataType& data, std::shared_ptr<const BPMNOS::Object>* contentObject) const {
481 // the operators of a node stating a message or signal may refer to its content object, which is held in the slot
482 // following the status objects while they run
483 if ( content ) {
484 assert( contentObject && *contentObject );
485 assert( status.objects.size() == content->index );
486 status.objects.push_back( std::move(*contentObject) );
487 }
488 BPMNOS::number change = 0;
489 for ( auto& operator_ : operators ) {
490 change += operator_->apply(status,data);
491 }
492 if ( content ) {
493 *contentObject = std::move(status.objects.back());
494 status.objects.pop_back();
495 }
496 return change;
497}

528 std::vector<std::pair<const Attribute*, BPMNOS::number>> contributions;
529 // at most one contribution per attribute
530 contributions.reserve(attributes.size() + this->data.size());
531 for ( auto& attribute : attributes ) {
532 auto value = attributeRegistry.getValue(attribute.get(),status,data);
533 if ( value.has_value() ) {
534 contributions.emplace_back(attribute.get(),value.value());
535 }
536 }
537 for ( auto& attribute : this->data ) {
538 auto value = attributeRegistry.getValue(attribute.get(),status,data);
539 if ( value.has_value() ) {
540 contributions.emplace_back(attribute.get(),value.value());
541 }
542 }
543 return contributions;
