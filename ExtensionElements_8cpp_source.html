369template <typename DataType>
370bool ExtensionElements::satisfiesInheritedRestrictions(const BPMNOS::Values& status, const DataType& data, const BPMNOS::Values& globals) const {
371 auto base = baseElement->represents<BPMN::ChildNode>();
372
373 if ( !base ) return true;
374
375 // check restrictions within ancestor scopes
376 const BPMN::Node* ancestor = base->parent;
377 while ( ancestor ) {
378 assert( ancestor->extensionElements->represents<BPMNOS::Model::ExtensionElements>() );
379 if ( !ancestor->extensionElements->as<BPMNOS::Model::ExtensionElements>()->fullScopeRestrictionsSatisfied(status,data,globals) ) {
380 return false;
381 }
382 if ( auto eventSubProcess = ancestor->represents<BPMN::EventSubProcess>();
383 eventSubProcess &&
384 ( eventSubProcess->startEvent->represents<BPMN::ErrorStartEvent>() || eventSubProcess->startEvent->represents<BPMN::CompensateStartEvent>() )
385 ) {
386 // error and compensate event subprocesses do not inherit restrictions
387 break;
388 }
389 else if ( auto activity = ancestor->represents<BPMN::Activity>();
390 activity && activity->isForCompensation
391 ) {
392 // compensation activities do not inherit restrictions
393 break;
394 }
395 else if ( auto child = ancestor->represents<BPMN::ChildNode>() ) {
396 ancestor = child->parent;
397 }
398 else {
399 break;
400 }
401 }
402 return true;
403}
