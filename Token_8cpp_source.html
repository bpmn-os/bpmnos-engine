608 auto now = owner->systemState->getTime();

621 }

624 owner->systemState->engine->notify( DataUpdate( extensionElements->dataUpdate.attributes ) );

1053//std::cerr << "advanceToExiting: " << jsonify().dump() << std::endl;

1055

1058 }

1061

1063 if ( !exitIsFeasible() ) {

1065 return;
1066 }
1067
1068 auto extensionElements = node->extensionElements->represents<BPMNOS::Model::ExtensionElements>();
1069
1070 if ( extensionElements && ( extensionElements->attributes.size() || extensionElements->data.size() ) ) {
1071 // the objective is update with the final status attributes declared for the node
1072 BPMNOS::number DELTA = 0;
1073 for ( auto& attribute : extensionElements->attributes ) {
1074 if ( attribute->weight != 0 ) {
1075 if ( auto value = extensionElements->attributeRegistry.getValue(attribute.get(),status,*data,globals); value.has_value() ) {
1076 DELTA += value.value() * attribute->weight;
1077 }
1078 }
1079 }
1080 if ( DELTA != 0 ) {
1081 globals[BPMNOS::Model::ExtensionElements::Index::Objective].value() += DELTA;
1082 // dataUpdate indicating that objective has changed
1083 owner->systemState->engine->notify( DataUpdate( { extensionElements->attributeRegistry.globalAttributes[BPMNOS::Model::ExtensionElements::Index::Objective] } ) );
1084 }
1085//std::cerr << "objective updated" << std::endl;
1086 }
1087
1088 auto activity = node->represents<BPMN::Activity>();
1089
1090 if (
1091 owned &&
1092 !(
1093 activity &&
1094 activity->loopCharacteristics.has_value() &&
1095 activity->loopCharacteristics.value() != BPMN::Activity::LoopCharacteristics::Standard &&
1096 owner->systemState->tokenAtMultiInstanceActivity.contains(this)
1097 )
1098 ) {
1099 data = &const_cast<StateMachine*>(owner)->data;
1100 }
1101
1102 if ( activity && activity->loopCharacteristics.has_value() && activity->loopCharacteristics.value() == BPMN::Activity::LoopCharacteristics::Standard ) {
1103 auto& attributeRegistry = getAttributeRegistry();
1104

1106 if (extensionElements->loopCondition.has_value() && extensionElements->loopCondition.value()->expression) {
1107 auto value = extensionElements->loopCondition.value()->expression->execute(status, *data, globals);
1108 assert( value.has_value() );

1110 // do not loop if loop condition is violated
1111 return false;

1116 auto maximum = (double)extensionElements->loopMaximum.value()->expression->execute(status, *data, globals).value_or(0);
1117 assert( extensionElements->loopIndex.value()->expression );
1118 auto indexAttribute = extensionElements->loopIndex.value()->expression->isAttribute();
1119 assert( indexAttribute );
1120 auto index = attributeRegistry.getValue( indexAttribute, status, *data, globals).value();
1121
1122 if ( index >= maximum ) {
1123 // do not loop if loop maximum loop count is reached
1124 return false;
1125 }
1126 }
1127 return true;
1128 }();
1129
1130 if ( LOOP ) {
1131 awaitEntryEvent();
1132 return;
1133 }
1134 }
1135
1136 if ( extensionElements && extensionElements->attributes.size() ) {
1137 // remove attributes that are no longer needed
1138 assert( status.size() == extensionElements->attributeRegistry.statusAttributes.size() );
1139 status.resize( status.size() - extensionElements->attributes.size() );
1140 }

1142 if ( activity && activity->loopCharacteristics.has_value() && activity->loopCharacteristics.value() != BPMN::Activity::LoopCharacteristics::Standard ) {
1143 // delegate removal of token copies for multi-instance activity to owner
1144 auto stateMachine = const_cast<StateMachine*>(owner);
1145 engine->commands.emplace_back(std::bind(&StateMachine::deleteMultiInstanceActivityToken,stateMachine,this), this);
1146 return;
1147 }
1148
1149 if ( activity ) {
1150 auto stateMachine = const_cast<StateMachine*>(owner);
1151 if ( !activity->boundaryEvents.empty() ) {
1152 // remove tokens at boundary events
1153 engine->commands.emplace_back( std::bind(&StateMachine::deleteTokensAwaitingBoundaryEvent,stateMachine,this), stateMachine );
1154 }
1155 }

1157 // clear state machine owned by token
1158 if ( owned ) {
1159 owned->tokens.clear();
1160 owned->interruptingEventSubProcess.reset();
1161 owned->nonInterruptingEventSubProcesses.clear();
1162 owned->pendingEventSubProcesses.clear();
1163 owned.reset();
1164 }
1165
1166 if ( node->outgoing.empty() ) {
1167 engine->commands.emplace_back(std::bind(&Token::advanceToDone,this), this);
1168 return;
1169 }
1170 advanceToDeparting();
1171}

1499void Token::setSignalContent(BPMNOS::VariedValueMap& sourceMap) {
1500 auto& attributeRegistry = getAttributeRegistry();
1501 assert( node->extensionElements->represents<BPMNOS::Model::Signal>() );
1502 auto signalDefinition = node->extensionElements->as<BPMNOS::Model::Signal>();
1503
1504 auto oldObjective = globals[BPMNOS::Model::ExtensionElements::Index::Objective];
1505
1506 size_t counter = 0;
1507 for (auto& [key,contentValue] : sourceMap) {
1508 if ( auto it = signalDefinition->contentMap.find(key); it != signalDefinition->contentMap.end() ) {
1509 auto& [_,definition] = *it;
1510 auto attribute = definition->attribute;
1511//std::cerr << "Attribute: " << attribute.name << "/" << attribute.index << std::endl;
1512 if ( std::holds_alternative< std::optional<number> >(contentValue) && std::get< std::optional<number> >(contentValue).has_value() ) {
1513 // use attribute value of signal
1514 attributeRegistry.setValue(attribute, status, *data, globals, std::get< std::optional<number> >(contentValue).value() );
1515 }
1516 else if (std::holds_alternative<std::string>(contentValue)) {
1517 // use default value of emitter
1518 ValueVariant value = std::get< std::string >(contentValue);
1519 attributeRegistry.setValue(attribute, status, *data, globals, BPMNOS::to_number(value,attribute->type) );
1520 }
1521 else {
1522 attributeRegistry.setValue(attribute, status, *data, globals, std::nullopt );
1523 }
1524 }
1525 else {
1526 // key in signal content, but not in recipient content
1527 counter++;
1528 }
1529 }
1530
1531 if ( signalDefinition->contentMap.size() > sourceMap.size() - counter ) {
1532 // recipient has keys in content that are not in signal content
1533 for (auto& [key,definition] : signalDefinition->contentMap) {
1534 if ( !sourceMap.contains(key) ) {
1535 // key in recipient content, but not in message content
1536 attributeRegistry.setValue(definition->attribute, status, *data, globals, std::nullopt );
1537 }
1538 }
1539 }
1540
1541 if ( globals[BPMNOS::Model::ExtensionElements::Index::Objective] != oldObjective ) {
1542 // dataUpdate indicating that objective has changed
1543 owner->systemState->engine->notify( DataUpdate( { attributeRegistry.globalAttributes[BPMNOS::Model::ExtensionElements::Index::Objective] } ) );
1544 }
1545
1546 // notify about data update
1547 if ( signalDefinition->dataUpdate.global ) {
1548 owner->systemState->engine->notify( DataUpdate( signalDefinition->dataUpdate.attributes ) );
1549 }
1550 else {
1551 owner->systemState->engine->notify( DataUpdate( owner->root->instance.value(), signalDefinition->dataUpdate.attributes ) );
1552 }
