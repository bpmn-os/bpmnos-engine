195template<typename InstanceData>
196void DataProvider::ensureDefaultValue(InstanceData& instance, const std::string& attributeId,
197 std::optional<BPMNOS::number> value) {
198 assert(attributes.contains(instance.process));
199 auto it1 = attributes.at(instance.process).find(attributeId);
200 if (it1 == attributes.at(instance.process).end()) {
201 throw std::runtime_error("DataProvider: unable to find required attribute '" +
202 attributeId + "' for process '" + instance.process->id + "'");
203 }
204 auto attribute = it1->second;
205 if (auto it2 = instance.data.find(attribute); it2 == instance.data.end()) {
206 if (attribute->expression) {
207 throw std::runtime_error("DataProvider: initial value of default attribute '" +
208 attribute->id + "' must not be provided by expression");
209 }
210
211 if (value.has_value()) {
212 instance.data[attribute] = value.value();
213 }
214 else if (attributeId == BPMNOS::Keyword::Timestamp) {
215 instance.data[attribute] = 0;
216 }
217 else {
218 throw std::runtime_error("DataProvider: attribute '" + attribute->id + "' has no default value");
219 }
220 }
221}
