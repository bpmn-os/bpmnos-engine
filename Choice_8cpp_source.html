14 // The membership operator is written out before the condition is scanned, so that the scan meets the
15 // name `in` and reads the brackets of an enumeration as an index, which it copies. Were the operator
16 // written out afterwards, the brackets would follow no name and the enumeration would be read as a
17 // collection literal, which it is not: its alternatives may be expressions.
18 auto condition = decision->condition.value.value;
19 BPMNOS::replace_all( condition, "∈", " in ");
20 auto input = InputEncoder(condition).text();
21
22 if ( input.contains(" in ") ) {
23 parseEnumeration(input);
24 }
25 else if ( !input.contains("<") ) {
26 throw std::runtime_error("Choice: no enumeration or bounds given in '" + input + "'");
27 }
28 else {
29 BPMNOS::replace_all( input, "|", " divides ");
30 auto [bounds,discretizer] = [&input]() -> std::pair<std::string, std::string> {
31 if ( !input.contains(" divides ") ) {
32 // no discretizer provided
33 return { input, "" };
34 }
35
36 // discretizer is assumed to be provided following the bounds, separated by comma
37 auto pos = input.rfind(',');
38 if ( pos == std::string::npos ) {
39 throw std::runtime_error("Choice: illegal condition '" + input + "'");

41 return {

43 BPMNOS::trim_copy(input.substr(pos + 1))

64void Choice::parseEnumeration(const std::string& input) {
65 auto parts = BPMNOS::split(input," in ");
66 if ( parts.size() != 2 ) {
67 throw std::runtime_error("Choice: illegal enumeration '" + input + "'");
68 }
69
70 std::string attributeName = BPMNOS::trim_copy(parts.front());
71 if ( attributeName == "" ) {
72 throw std::runtime_error("Choice: unable to determine attribute name");
73 }
74 attribute = attributeRegistry[ attributeName ];
75
76 auto rhs = BPMNOS::trim_copy(parts.back());
77
78 if ( (rhs.front() == '[' && rhs.back() == ']') || (rhs.front() == '{' && rhs.back() == '}') ) {
79 // the condition has been scanned, so a literal among the alternatives is a number already and the
80 // commas that remain are the ones separating the alternatives
81 auto alternatives = BPMNOS::split( rhs.substr(1, rhs.size()-2), ',' );
82 for ( auto& alternative : alternatives ) {
83 enumeration.emplace_back( std::make_unique<Expression>(InputEncoder::fragment(BPMNOS::trim_copy(alternative)), attributeRegistry) );
84 for ( auto dependency : enumeration.back()->inputs ) {
85 dependencies.insert(dependency);
86 }
87 }
88 if ( enumeration.empty() ) {
89 throw std::runtime_error("Choice: empty enumeration");
90 }
91 }
92 else {
93 throw std::runtime_error("Choice: invalid enumeration '" + rhs + "'");
94 }

98 // check bounds
99 auto conditions = BPMNOS::split(input,'<');

165std::pair<BPMNOS::number,BPMNOS::number> Choice::getBounds(const BPMNOS::Values& status, const DataType& data, const BPMNOS::Values& globals) const {
166 assert( attribute->type != STRING );
167 assert( lowerBound.has_value() );
168 assert( upperBound.has_value() );
169 auto& [LB,strictLB] = lowerBound.value();
170 auto lb = LB->execute(status,data,globals);
171 BPMNOS::number min = lb.has_value() ? BPMNOS::number(lb.value()) : std::numeric_limits<BPMNOS::number>::lowest();
172 if ( strictLB ) {
173 min += BPMNOS_NUMBER_PRECISION;
174 }
175
176 auto& [UB,strictUB] = upperBound.value();
177 auto ub = UB->execute(status,data,globals);
178 BPMNOS::number max = ub.has_value() ? BPMNOS::number(ub.value()) : std::numeric_limits<BPMNOS::number>::max();
179 if ( strictUB ) {
180 max -= BPMNOS_NUMBER_PRECISION;
181 }
182
183 if ( attribute->type != DECIMAL ) {
184 min = std::ceil((double)min);
185 max = std::floor((double)max);
186 }
187
188 return {min,max};
189}

196std::vector<BPMNOS::number> Choice::getEnumeration(const BPMNOS::Values& status, const DataType& data, const BPMNOS::Values& globals) const {
197 assert( !enumeration.empty() || multipleOf );
198 std::vector<BPMNOS::number> allowedValues;
199 if ( !enumeration.empty() ) {
200 for ( auto& alternative : enumeration ) {
201 auto allowedValue = alternative->execute(status,data,globals);
202 if ( allowedValue.has_value() ) {
203 // an alternative is a value the attribute may take, so it is held as such
204 allowedValues.push_back( BPMNOS::number(allowedValue.value()) );
205 }
206 }
207 }
208 else {
209 auto [LB, UB] = getBounds(status, data, globals);
210 auto discretizer = multipleOf->execute(status, data, globals);
211 if ( !discretizer.has_value() ) {
212 if ( attribute->type == BPMNOS::ValueType::BOOLEAN || attribute->type == BPMNOS::ValueType::INTEGER ) {
213 discretizer = 1;
214 }
215 else {
216 throw std::runtime_error("Choice: cannot determine discretizer for '" + multipleOf->expression + "'");
217 }
218 }
219
220 double DELTA = std::abs(discretizer.value());
221 if ( !(DELTA > 0.0) ) {
222 throw std::runtime_error("Choice: discretizer for '" + multipleOf->expression + "' is not a positive step");
223 }
224
225 // The multiples are counted from zero
226 for ( double multiple = std::floor((double)LB / DELTA); ; multiple += 1.0 ) {
227 BPMNOS::number value = multiple * DELTA;
228 if ( value > UB ) {
229 break;
230 }
231 if ( value >= LB ) {
232 allowedValues.push_back(value);
233 }
234 }
235 }
236
237 return allowedValues;
238}
