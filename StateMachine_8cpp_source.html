412 auto extensionElements = token->node->extensionElements->represents<const BPMNOS::Model::ExtensionElements>();
413 assert( extensionElements != nullptr );
414
415 std::vector< std::map< const Model::Attribute*, std::optional<BPMNOS::number> > > valueMaps;
416
417 if ( extensionElements->loopCardinality.has_value() ) {
418 auto getLoopCardinality = [token,extensionElements]() -> std::optional<BPMNOS::number> {
419 if (!extensionElements->loopCardinality.value()->expression) {
420 return std::nullopt;
421 }
422 return extensionElements->loopCardinality.value()->expression->execute(token->status, *token->data, token->globals);
423 };
424 // use provided cardinality to determine number of tokens
425 if ( auto loopCardinality = getLoopCardinality();
426 loopCardinality.has_value()
427 ) {
428 valueMaps.resize( (size_t)loopCardinality.value() );
429 }
430 else {
431 throw std::runtime_error("StateMachine: cannot determine cardinality for multi-instance activity '" + token->node->id +"'" );
432 }
433 }
434
435 if ( extensionElements->messageDefinitions.size() ) {
436 if ( valueMaps.empty() ) {
437 valueMaps.resize(extensionElements->messageDefinitions.size());
438 }
439 else if ( valueMaps.size() != extensionElements->messageDefinitions.size() ) {
440 throw std::runtime_error("StateMachine: cardinality and number of messages inconsistent for multi-instance activity '" + token->node->id +"'" );
441 }
442 }
443
444 if ( valueMaps.empty() ) {
445 throw std::runtime_error("StateMachine: no instances created for multi-instance activity '" + token->node->id +"'" );
446 }
447
448 if ( extensionElements->loopIndex.has_value() && extensionElements->loopIndex.value()->expression ) {
449 // set value of loop index attribute for each instance
450 if ( auto attribute = extensionElements->loopIndex.value()->expression->isAttribute() ) {
451 for ( size_t i = 0; i < valueMaps.size(); i++ ) {
452 valueMaps[i][attribute] = i + 1;
453 }
454 }
455 else {
456 throw std::runtime_error("StateMachine: no attribute provided for loop index parameter of multi-instance activity '" + token->node->id +"'" );
457 }
458 }
459
460 auto activity = token->node->represents<BPMN::Activity>();
461 assert( activity );
462 assert( activity->loopCharacteristics.has_value() );
463
464 // create token copies
465 assert ( systemState->tokensAtActivityInstance.find(token) == systemState->tokensAtActivityInstance.end() );
466 assert ( systemState->exitStatusAtActivityInstance.find(token) == systemState->exitStatusAtActivityInstance.end() );
467
468 Token* tokenCopy = nullptr;
469 size_t counter = 0;
470 for ( auto valueMap : valueMaps ) {
471 counter++;
472 // disambiguate instance id
473 auto instanceId = BPMNOS::to_string(this->data[BPMNOS::Model::ExtensionElements::Index::Instance].get().value(),STRING) + BPMNOS::Model::Scenario::delimiters[0] + token->node->id + BPMNOS::Model::Scenario::delimiters[1] + std::to_string(counter);
474
475 tokens.push_back( std::make_shared<Token>( token ) );
476 if ( auto scope = token->node->represents<BPMN::Scope>() ) {
477 // create state machine for each multi-instance subprocess
478 auto data = systemState->getDataAttributes(root,token->node);
479 if ( !data.has_value() ) {
480 throw std::runtime_error("StateMachine: required data at '" + token->node->id +"' not yet available" );
481 }
482
483 // create child state machine with disambiguated instance identifier
484 createChild( tokens.back().get(), scope, std::move(data.value()), BPMNOS::to_number(instanceId,BPMNOS::ValueType::STRING) );
485 // ensure that data is set appropriately
486 tokens.back()->data = &tokens.back()->owned->data;
487//std::cerr << "MI:" << instanceId << "/" << tokens.back()->jsonify() << std::endl;
488 }
489 else {
490 // create child fake state machine with disambiguated instance identifier
491 createChild( tokens.back().get(), nullptr, {}, BPMNOS::to_number(instanceId,BPMNOS::ValueType::STRING) );
492 // ensure that data is set appropriately
493 tokens.back()->data = &tokens.back()->owned->data;
494//std::cerr << "Fake:" << tokens.back()->jsonify() << std::endl;
495 }

497 // update status of token copy
498 for ( auto [attribute,value] : valueMap ) {
499 tokens.back().get()->status[attribute->index] = value;
500 }
501
502 auto instanceToken = tokens.back().get();
503
504 // Register instance bookkeeping BEFORE any state notification, so that the ready-event
505 // handler recognises this as a multi-instance instance token and skips it (the single
506 // ready event is consumed by the main token, not re-raised for each instance).
507 const_cast<SystemState*>(systemState)->tokensAtActivityInstance[token].push_back(instanceToken);
508 const_cast<SystemState*>(systemState)->exitStatusAtActivityInstance[token] = {};

511 // Every instance emits its own CREATED state (birth); ARRIVED is skipped (no incoming flow).
512 instanceToken->update(Token::State::CREATED);
513
514 if ( activity->loopCharacteristics.value() == BPMN::Activity::LoopCharacteristics::MultiInstanceSequential ) {
515 if ( !tokenCopy ) {
516 // for sequential multi-instance activities only the first instance becomes ready immediately
517 instanceToken->update(Token::State::READY);
518//std::cerr << "Token awaiting entry:" << instanceToken->jsonify() << std::endl;
519 instanceToken->awaitEntryEvent();

521 else {
522 // subsequent instances remain in CREATED until their predecessor has exited
523 const_cast<SystemState*>(systemState)->tokenAwaitingMultiInstanceExit[tokenCopy] = instanceToken;
524 }
525 }
526 else if ( activity->loopCharacteristics.value() == BPMN::Activity::LoopCharacteristics::MultiInstanceParallel ) {
527 // for parallel multi-instance activities all instances become ready immediately
528 instanceToken->update(Token::State::READY);
529//std::cerr << "Token awaiting entry:" << instanceToken->jsonify() << std::endl;

542void StateMachine::deleteMultiInstanceActivityToken(Token* token) {
543 auto engine = const_cast<Engine*>(systemState->engine);
544 auto mainToken = const_cast<SystemState*>(systemState)->tokenAtMultiInstanceActivity.at(token);
545
546 auto activity = token->node->represents<BPMN::Activity>();
547 assert( activity );
548
549 // Instance has no outgoing flow: emit DONE as its terminal state before removal.
550 // Aggregation into the main token is handled below, not via the generic scope shutdown.
551 token->update(Token::State::DONE);
552
553 if ( activity->loopCharacteristics.value() == BPMN::Activity::LoopCharacteristics::MultiInstanceSequential ) {
554 // advance next token for sequential multi-instance activity
555 auto& tokenAwaitingMultiInstanceExit = const_cast<SystemState*>(systemState)->tokenAwaitingMultiInstanceExit;
556 auto it = tokenAwaitingMultiInstanceExit.find(token);
557
558 if ( it != tokenAwaitingMultiInstanceExit.end() ) {
559 auto waitingToken = it->second;
560 // successor leaves CREATED and becomes ready now that its predecessor has exited
561 waitingToken->update(Token::State::READY);
562 waitingToken->awaitEntryEvent();
563 tokenAwaitingMultiInstanceExit.erase(it);
564 }
565 }
566
567 // record exit status
568 const_cast<SystemState*>(systemState)->exitStatusAtActivityInstance[mainToken].push_back(token->status);
569 // remove token
570 const_cast<SystemState*>(systemState)->tokenAtMultiInstanceActivity.erase(token);
571 erase_ptr<Token>(const_cast<SystemState*>(systemState)->tokensAtActivityInstance[mainToken],token);
572 erase_ptr<Token>(tokens,token);
573
574 // advance main token when last multi-instance token exited
575 auto& tokensAtActivityInstance = const_cast<SystemState*>(systemState)->tokensAtActivityInstance;
576 if ( auto it1 = tokensAtActivityInstance.find(mainToken);
577 it1 != tokensAtActivityInstance.end()
578 ) {
579 if ( it1->second.empty() ) {
580 // last multi-instance token exited
581 tokensAtActivityInstance.erase(it1);
582
583 if ( !activity->boundaryEvents.empty() ) {
584 // remove tokens at boundary events
585 deleteTokensAwaitingBoundaryEvent( mainToken );
586 }
587
588 auto& exitStatusAtActivityInstance = const_cast<SystemState*>(systemState)->exitStatusAtActivityInstance;
589 if ( auto it2 = exitStatusAtActivityInstance.find(mainToken);
590 it2 != exitStatusAtActivityInstance.end()
591 ) {
592 // merge status
593 mainToken->status = BPMNOS::mergeValues(it2->second);
594 exitStatusAtActivityInstance.erase(it2);
595 }
596
597 // advance main token
598 if ( mainToken->node->outgoing.empty() ) {
599 engine->commands.emplace_back(std::bind(&Token::advanceToDone,mainToken), mainToken);
600 }
601 else {
602 engine->commands.emplace_back(std::bind(&Token::advanceToDeparting,mainToken), mainToken);
603 }
604 }
605 }
606 else {
607 assert(false && "cannot find tokens created for multi instance activity");
608 }

744void StateMachine::run(Values status) {
745 assert( status.size() >= 1 );
746 assert( data.size() >= 1 );
747 assert( data[BPMNOS::Model::ExtensionElements::Index::Instance].get().has_value() );
748 assert( status[BPMNOS::Model::ExtensionElements::Index::Timestamp].has_value() );
749
750//std::cerr << "Run " << scope->id << "/" << this << "/" << parentToken << std::endl;
751 if ( !parentToken ) {
752 // state machine without parent token represents a process
753//std::cerr << "Start process " << process->id << std::endl;
754 auto instanceId = (long unsigned int)data[BPMNOS::Model::ExtensionElements::Index::Instance].get().value();
755 const_cast<SystemState*>(systemState)->archive[ instanceId ] = weak_from_this();
756
757 tokens.push_back( std::make_shared<Token>(this,nullptr,std::move(status)) );
758 registerRecipient();
759
760 // create child that will own tokens flowing through the process
761 createChild(tokens.back().get(),scope, {});
762
763 }
764 else {
765 for ( auto startNode : scope->startNodes ) {
766 tokens.push_back( std::make_shared<Token>(this,startNode,std::move(status)) );
767 }
768 }
769
770 for ( auto token : tokens ) {
771 if ( token->node ) {
772 if ( auto startEvent = token->node->represents<BPMN::TypedStartEvent>() ) {
773 // get new attribute values
774 auto values = systemState->getStatusAttributes( root, token->node->parent );
775 if ( !values.has_value() ) {
776 throw std::runtime_error("StateMachine: status of event subprocess '" + token->node->parent->id + "' not known at initialization");
777 }
778
779 token->status.insert(token->status.end(), values.value().begin(), values.value().end());
780
781 if ( !startEvent->isInterrupting ) {
782 // token instantiates non-interrupting event subprocess
783 // get instantiation counter from context
784 auto context = const_cast<StateMachine*>(parentToken->owned.get());
785 auto counter = ++context->instantiations[token->node];
786 // disambiguate instance id
787 auto instanceId = BPMNOS::to_string((*parentToken->data)[BPMNOS::Model::ExtensionElements::Index::Instance].get().value(),STRING) + BPMNOS::Model::Scenario::delimiters[0] + scope->id + BPMNOS::Model::Scenario::delimiters[1] + std::to_string(counter);
788 data[BPMNOS::Model::ExtensionElements::Index::Instance].get() = BPMNOS::to_number(instanceId,BPMNOS::ValueType::STRING);
789 const_cast<SystemState*>(systemState)->archive[ (long unsigned int)data[BPMNOS::Model::ExtensionElements::Index::Instance].get().value() ] = weak_from_this();
790 registerRecipient();
791 }
792 }
793//std::cerr << "Initial token: >" << token->jsonify().dump() << "<" << std::endl;
794 }
795 // advance token
796 token->advanceFromCreated();
797 }

981void StateMachine::handleFailure(Token* token) {
982//std::cerr << scope->id << " handles failure at " << (token->node ? token->node->id : process->id ) << "/" << parentToken << "/" << token << std::endl;
983 auto engine = const_cast<Engine*>(systemState->engine);
984
985// assert( !token->owned );
986
987//std::cerr << "check whether failure is caught" << std::endl;
988
989 if ( token->node ) {
990 if ( auto activity = token->node->represents<BPMN::Activity>() ) {
991 if ( activity->isForCompensation ) {
992 // compensation activity failed, clear all other compensations
993 compensationTokens.clear();
994 }
995 else if ( token->state != Token::State::WAITING &&
996 activity->loopCharacteristics.has_value() &&
997 activity->loopCharacteristics.value() != BPMN::Activity::LoopCharacteristics::Standard
998 ) {
999 // handle failure at main token waiting for all instances
1000 auto mainToken = const_cast<SystemState*>(systemState)->tokenAtMultiInstanceActivity.at(token);
1001//std::cerr << "handle failure at main token waiting for all instances " << std::endl;
1002 mainToken->status = token->status;
1003 handleFailure(mainToken);
1004 return;
1005 }
1006 }
1007 }
1008
1009 // find error boundary event at token node
1010 if ( token->node ) {
1011 auto& tokensAwaitingBoundaryEvent = const_cast<SystemState*>(systemState)->tokensAwaitingBoundaryEvent[token];
1012 for ( auto eventToken : tokensAwaitingBoundaryEvent) {
1013 if ( eventToken->node->represents<BPMN::ErrorBoundaryEvent>() ) {
1014 eventToken->status = token->status;
1015 eventToken->advanceToCompleted();
1016 return;
1017 }
1018 }
1019 }
1020
1021 // find event-subprocess catching error
1022 auto it = std::find_if(pendingEventSubProcesses.begin(), pendingEventSubProcesses.end(), [](std::shared_ptr<StateMachine>& stateMachine) {
1023 auto eventSubProcess = stateMachine->scope->as<BPMN::EventSubProcess>();
1024 return eventSubProcess->startEvent->represents<BPMN::ErrorStartEvent>();
1025 });
1026 if ( it != pendingEventSubProcesses.end() ) {
1027 // trigger event subprocess
1028 auto eventToken = it->get()->tokens.front().get();
1029 // update status of event token with that of current token
1030 eventToken->status = token->status;
1031 // remove all tokens
1032 clearObsoleteTokens();
1033 engine->commands.emplace_back(std::bind(&Token::advanceToCompleted,eventToken), eventToken);
1034
1035 return;
1036 }
1037
1038 // failure is not caught
1039//std::cerr << "uncaught failure" << std::endl;
1040
1041
1042 if ( !parentToken ) {
1043//std::cerr << "process has failed" << std::endl;
1044 // failure at process before state machine containing flow elements has been created
1045 assert(tokens.size() == 1);
1046 assert(token = tokens.front().get());
1047 engine->commands.emplace_back(std::bind(&Engine::deleteInstance,engine,this), this);
1048 return;
1049 }
1050
1051 // bubble up error
1052
1053 // update status of parent token with that of current token
1054 parentToken->status = token->status;
1055 engine->commands.emplace_back(std::bind(&Token::advanceToFailed,parentToken), parentToken);
1056}

1084void StateMachine::shutdown() {
1085 auto engine = const_cast<Engine*>(systemState->engine);
1086//std::cerr << "shutdown: " << scope->id << std::endl;
1087 assert( tokens.size() );
1088 BPMNOS::Values mergedStatus = Token::mergeStatus(tokens);
1089
1090 if ( auto eventSubProcess = scope->represents<BPMN::EventSubProcess>() ) {
1091 // update global objective
1092 assert( scope->extensionElements->as<BPMNOS::Model::ExtensionElements>() );
1093 for ( auto& [attribute,value] : scope->extensionElements->as<BPMNOS::Model::ExtensionElements>()->getContributionsToObjective(mergedStatus,data,parentToken->globals) ) {
1094 const_cast<SystemState*>(systemState)->contributionsToObjective[attribute] += value;
1095 }
1096
1097 if (!eventSubProcess->startEvent->isInterrupting ) {
1098 // delete non-interrupting event subprocess
1099 auto context = const_cast<StateMachine*>(parentToken->owned.get());
1100 engine->commands.emplace_back(std::bind(&StateMachine::deleteNonInterruptingEventSubProcess,context,this), this);
1101 return;
1102 }
1103 else if ( eventSubProcess->startEvent->represents<BPMN::CompensateStartEvent>() ) {
1104 completeCompensationEventSubProcess();
1105 return;
1106 }
1107 }
1108
1109//std::cerr << "start shutdown: " << BPMNOS::to_string(instance.value(),STRING) << std::endl;
1110
1111 // update status of parent token (if it doesn't have a sequential performer)
1112 if ( parentToken &&
1113 !scope->extensionElements->as<BPMNOS::Model::ExtensionElements>()->hasSequentialPerformer
1114 ) {
1115 parentToken->status = mergedStatus;
1116 }
1117 tokens.clear();
1118
1119 // ensure that messages to state machine are removed
1120 unregisterRecipient();
1121
1122 if ( !parentToken ) {
