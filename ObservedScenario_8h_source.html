8/**
9 * @brief A scenario whose data is observed while the run proceeds.
10 *
11 * Nothing is known in advance. The scenario holds a log of what has been reported to it so far, and
12 * answers every query from that log, reporting `std::nullopt` for anything not yet observed, upon which
13 * the engine waits and asks again at the next clock tick.
14 *
15 * Whatever observes the world reports through @ref observeInstantiation, @ref observeValue,
16 * @ref observeReadyStatus and @ref observeCompletionStatus, and owns time, issuing the clock ticks itself.
17 * An observed world admits no duplicate of itself, so @ref clone throws.
18 */
19class ObservedScenario : public Scenario {
20public:
21 /**
22 * @brief Constructor for ObservedScenario.
23 *
24 * @param model The model to be executed.
25 * @param globalValueMap The observed initial values of global attributes. Globals change thereafter
26 * through the engine and are never re-observed.
27 */
28 ObservedScenario(const Model* model, const std::unordered_map<const Attribute*, BPMNOS::number>& globalValueMap);
