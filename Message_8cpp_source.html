102 auto& targetContentDefinition = node->extensionElements->as<BPMNOS::Model::ExtensionElements>()->getMessageDefinition(status)->contentMap;
103
104 size_t counter = 0;
105 for (auto& [key,contentValue] : contentValueMap) {
106 if ( auto it = targetContentDefinition.find(key); it != targetContentDefinition.end() ) {
107 auto& [_,definition] = *it;
108 auto attribute = definition->attribute;
109//std::cerr << "Attribute: " << attribute.name << "/" << attribute.index << std::endl;
110 if ( std::holds_alternative< std::optional<number> >(contentValue) && std::get< std::optional<number> >(contentValue).has_value() ) {
111 // use attribute value sent in message
112 attributeRegistry.setValue(attribute, status, data, globals, std::get< std::optional<number> >(contentValue).value() );
113 }
114 else if (std::holds_alternative<std::string>(contentValue)) {
115 // use default value of sender
116 ValueVariant value = std::get< std::string >(contentValue);
117 attributeRegistry.setValue(attribute, status, data, globals, BPMNOS::to_number(value,attribute->type) );
118 }
119 else {
120 attributeRegistry.setValue(attribute, status, data, globals, std::nullopt );
121 }
122 }
123 else {
124 // key in message content, but not in recipient content
125 counter++;
126 }
127 }
128
129 if ( targetContentDefinition.size() > contentValueMap.size() - counter ) {
130 // recipient has keys in content that are not in message content
131 for (auto& [key,definition] : targetContentDefinition) {
132 if ( !contentValueMap.contains(key) ) {
133 // key in recipient content, but not in message content
134 attributeRegistry.setValue(definition->attribute, status, data, globals, std::nullopt );
135 }
136 }
137 }
