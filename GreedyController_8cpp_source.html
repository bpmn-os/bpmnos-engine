14GreedyController::GreedyController(std::shared_ptr<Evaluator> evaluator, Config config)
15 : config(config)
16{
17 // Decisions come from Candidates collections, each owned by a generic GreedyDispatcher (templated on the
18 // collection type), tried in priority order. Exit, entry, and direct message delivery are unambiguous;
19 // the contested sequential-ad-hoc entries and message deliveries are merged into one reward-ordered
20 // collection so their best feasible decision competes within the same list.
21 dispatchers.push_back( std::make_unique<GreedyDispatcher<FirstFeasibleExit>>(evaluator) );
22 dispatchers.push_back( std::make_unique<GreedyDispatcher<FirstFeasibleEntry>>(evaluator) ); // non-sequential entries only (config.sequential=false)
23 dispatchers.push_back( std::make_unique<InstantDirectMessage>() );
24 if ( config.bisection ) {
25 dispatchers.push_back( std::make_unique<GreedyDispatcher<FirstBisectionalChoice>>(evaluator) );
26 }
27 else {
28 dispatchers.push_back( std::make_unique<GreedyDispatcher<FirstEnumeratedChoice>>(evaluator) );
29 }
30 dispatchers.push_back( std::make_unique<GreedyDispatcher<CompetingCandidates>>(evaluator) ); // sequential ad-hoc entries and message deliveries
31}

40std::shared_ptr<Event> GreedyController::dispatchEvent(const SystemState* systemState) {
41 // Dispatch the first feasible decision in priority order; forward any non-decision event immediately.
42 for ( auto& eventDispatcher : dispatchers ) {
43 if ( auto event = eventDispatcher->dispatchEvent(systemState) ) {
44 if ( auto decision = dynamic_pointer_cast<Decision>(event) ) {
45 if ( decision->reward().has_value() ) {
46 return event;
47 }
48 }
49 else {
50 // events are immediately forwarded
51 return event;
52 }
53 }
54 }
55
56 return nullptr;
57}
