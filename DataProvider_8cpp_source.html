98void DataProvider::evaluateGlobal(const std::string& initializationString,
99 const LIMEX::Handle<double>& handle) {
100 auto [attributeName, expressionString] = parseInitialization(initializationString);
101
102 // Find global attribute by name
103 const Attribute* attribute = nullptr;
104 for (auto& [id, globalAttribute] : attributes[nullptr]) {
105 if (globalAttribute->name == attributeName) {
106 attribute = globalAttribute;
107 break;
108 }
109 }
110 if (!attribute) {
111 throw std::runtime_error("DataProvider: unknown global attribute '" + attributeName + "'");
112 }
113 if (attribute->expression) {
114 throw std::runtime_error("DataProvider: global attribute '" + attributeName +
115 "' is initialized by model expression and must not be provided explicitly");
116 }
117
118 // Build globals vector from current globalValueMap
119 Values globals(model->attributes.size());
120 for (auto& [globalAttribute, value] : globalValueMap) {
121 globals[globalAttribute->index] = value;
122 }
123
124 // Compile and evaluate expression
125 Expression expression(handle, expressionString, model->attributeRegistry);
126
127 // Validate all referenced globals are already evaluated
128 for (auto* referencedAttribute : expression.variables) {
129 if (!globalValueMap.contains(referencedAttribute)) {
130 throw std::runtime_error("DataProvider: global '" + attributeName +
131 "' references unevaluated global '" + referencedAttribute->name + "'");
132 }
133 }
134
135 auto value = expression.execute(Values{}, Values{}, globals);
136 if (!value.has_value()) {
137 throw std::runtime_error("DataProvider: failed to evaluate global '" + attributeName + "'");
138 }
139
140 // Apply type conversion
141 BPMNOS::number convertedValue;
142 switch (attribute->type) {
143 case ValueType::INTEGER:
144 convertedValue = BPMNOS::number((int)value.value());
145 break;
146 case ValueType::BOOLEAN:
147 convertedValue = BPMNOS::number(value.value() != 0 ? 1 : 0);
148 break;
149 default:
150 convertedValue = value.value();
151 break;
152 }
153 globalValueMap[attribute] = convertedValue;
154}

176std::pair<std::string, std::string> DataProvider::parseInitialization(
177 const std::string& initialization) {
178 auto position = initialization.find(":=");
179 if (position == std::string::npos) {
180 throw std::runtime_error("DataProvider: initialization must be in format 'attribute := expression', got '" + initialization + "'");
181 }
182
183 std::string attributeName = initialization.substr(0, position);
184 std::string expression = initialization.substr(position + 2);
185
186 // Trim whitespace from attribute name
187 auto trimStart = attributeName.find_first_not_of(" \t");
188 auto trimEnd = attributeName.find_last_not_of(" \t");
189 if (trimStart == std::string::npos) {
190 throw std::runtime_error("DataProvider: empty attribute name in initialization '" + initialization + "'");
191 }
192 attributeName = attributeName.substr(trimStart, trimEnd - trimStart + 1);
193
194 // Trim whitespace from expression
195 trimStart = expression.find_first_not_of(" \t");
196 trimEnd = expression.find_last_not_of(" \t");
197 if (trimStart == std::string::npos) {
198 throw std::runtime_error("DataProvider: empty expression in initialization '" + initialization + "'");
199 }
200 expression = expression.substr(trimStart, trimEnd - trimStart + 1);
201
202 return {attributeName, expression};
203}
