17/// A candidate element: a tuple-like value whose first component is the reward and whose second-to-last
18/// component is the decision's weak Event — the only parts a dispatcher reads.
19template <typename T>
20concept CandidateElement = requires (T candidate) {
21 { std::get<0>(candidate) } -> std::convertible_to<double>;
22 { std::get<std::tuple_size_v<T> - 2>(candidate) } -> std::convertible_to<std::weak_ptr<Event>>;
23};
24
25/// A reward-ordered candidate collection a dispatcher can drive: connectable to a mediator and iterable over
26/// CandidateElements (best first).
27template <typename C>
28concept CandidateCollection = requires (C& collection, Mediator* mediator) {
29 collection.connect(mediator);
30 collection.begin() != collection.end();
31 ++std::declval< decltype(collection.begin())& >();
32 requires CandidateElement< std::remove_cvref_t<decltype(*collection.begin())> >;
33};
34
35/**
36 * @brief Generic greedy policy dispatcher owning a concrete Candidates collection.
37 *
38 * Templated on the Candidates type (any CandidateCollection), which it owns by value and constructs by
39 * forwarding the constructor arguments. On each dispatch it iterates the reward-ordered candidates for the
40 * current state (iterating them evaluates them lazily) and dispatches the best feasible one — the front of
41 * the set, unless that front is infeasible.
42 */
43template <CandidateCollection Candidates>
44class GreedyDispatcher : public EventDispatcher {
45public:
46 template <typename... Args>
47 GreedyDispatcher(Args&&... args) : candidates(std::forward<Args>(args)...) {}
48
49 std::shared_ptr<Event> dispatchEvent( [[maybe_unused]] const SystemState* systemState ) override {
50 for ( auto candidate : candidates ) {
51 constexpr std::size_t eventIndex = std::tuple_size<decltype(candidate)>::value - 2;
52 std::weak_ptr<Event>& event_ptr = std::get<eventIndex>(candidate);
53 if ( auto event = event_ptr.lock();
54 event && std::get<0>(candidate) > -std::numeric_limits<double>::infinity()
55 ) {
56 // dispatch the best feasible decision
57 return event;
58 }
59 else {
60 // best decision is infeasible, no need to inspect others
61 break;
62 }
63 }
64 return nullptr;
65 }
66
67 void connect(Mediator* mediator) override {
68 candidates.connect(mediator); // the candidates subscribe to their request type and DataUpdate
69 EventDispatcher::connect(mediator);
70 }
71
72protected:
73 Candidates candidates;
74};
