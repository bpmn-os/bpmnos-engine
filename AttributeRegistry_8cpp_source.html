11void AttributeRegistry::add(Attribute* attribute) {
12 if ( contains(attribute->name) ) {
13 throw std::runtime_error("AttributeRegistry: duplicate attribute name '" + attribute->name + "'");
14 }
15 // a matrix or an object input is read by its name as an attribute is, whereas a lookup is always called, so that
16 // only the former must not share its name with an attribute
17 if ( inputs.contains(attribute->name) ) {
18 throw std::runtime_error("AttributeRegistry: attribute name '" + attribute->name + "' is the name of an input");
19 }
20 attribute->declaration = declared++;
21 // objects are numbered separately from scalar attributes, so that the index of a scalar attribute never
22 // depends on objects
23 if ( attribute->category == Attribute::Category::STATUS ) {
24 auto& attributes = attribute->isObject() ? statusObjects : statusAttributes;
25 attribute->index = attributes.size();
26 attributes.push_back(attribute);
27 statusMap[attribute->name] = attribute;
28 }
29 else /* if ( attribute->category == Attribute::Category::DATA )*/ {
30 auto& attributes = attribute->isObject() ? dataObjects : dataAttributes;
31 attribute->index = attributes.size();
32 attributes.push_back(attribute);
33 dataMap[attribute->name] = attribute;
34 }
35}

141/// Returns the object a whole assignment of the value gives an object attribute holding the current object.
142std::shared_ptr<const BPMNOS::Object> assigned(const BPMNOS::Model::Attribute* object, const std::shared_ptr<const BPMNOS::Object>& current, const std::shared_ptr<const BPMNOS::Object>& value, const BPMNOS::Model::SchemaProvider& schemaProvider) {
143 BPMNOS::Schema target(schemaProvider.getSchema(object), *current->layout);
144 if ( target.supports(*value->layout->schema) ) {
145 return value;
146 }
147 try {
148 auto converted = std::make_shared<BPMNOS::Object>(target, *value);
149 if ( *converted->layout == *current->layout ) {
150 // the object keeps the layout of the current object, as after padding its fixed arrays
151 converted->layout = current->layout;
152 }
153 return converted;
154 }
155 catch ( const std::exception& error ) {
156 throw std::runtime_error("AttributeRegistry: illegal value of object '" + object->name + "'.\n" + error.what());
157 }
158}

172BPMNOS::number AttributeRegistry::setValue(const Attribute* attribute, Status& status, Data& data, std::optional<BPMNOS::number> value) const {
173 // the value of a scalar attribute; an object is held in the objects of the status or data
174 assert( !attribute->isObject() );
175 if ( attribute->category == Attribute::Category::STATUS ) {
176 assert(attribute->index < status.attributes.size());
177 status.attributes[attribute->index] = value;
178 // a status attribute is accounted in the objective when its scope ends
179 return 0;
180 }
181 else /* if ( attribute->category == Attribute::Category::DATA )*/ {
182 assert(attribute->index < data.attributes.size());
183 auto previous = data.attributes[attribute->index];
184 data.attributes[attribute->index] = value;
185 return ( value.value_or(0) - previous.value_or(0) ) * attribute->weight;
186 }
187}
