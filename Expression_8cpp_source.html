20Expression::Expression(const LIMEX::Handle<double>& handle, const InputEncoder& encoder,
21 const AttributeRegistry& attributeRegistry, bool newTarget)
22 : attributeRegistry(attributeRegistry)
23 , handle(handle)
24 , expression(encoder.text())
25 , compiled(getExpression(expression))
26 , type(getType())
27{
28 resize = ( preprocess(expression) != expression );
29 auto& assignment = std::get< LIMEX::Node<double> >(compiled.getRoot().operands[0]);
30 compound = ( (int)assignment.type > (int)LIMEX::Type::assign );
31 if ( auto name = compiled.getTarget(); name.has_value() ) {
32 if ( name.value() == BPMNOS::Keyword::Undefined ) {
33 throw std::runtime_error("Expression: illegal assignment '" + expression +"'");
34 }
35 if ( attributeRegistry.inputs.contains(name.value()) ) {
36 throw std::runtime_error("Expression: input '" + name.value() + "' is read only and cannot be assigned in '" + expression + "'");
37 }
38 if ( !newTarget ) {
39 target = attributeRegistry[ name.value() ];
40 objectTarget = target.value()->isObject();
41 }
42 }
43 if ( resize && !objectTarget ) {
44 throw std::runtime_error("Expression: resize requires an array in '" + expression + "'");
45 }
46 if ( compiled.getTargetPath().has_value() && !objectTarget ) {
47 throw std::runtime_error("Expression: assignment to an element of '" + compiled.getTarget().value() + "', which is not an object, in '" + expression + "'");
48 }
49 if ( objectTarget ) {
50 if ( type != Type::ASSIGN ) {
51 throw std::runtime_error("Expression: object '" + target.value()->name + "' cannot be undefined in '" + expression + "'");
52 }
53 if ( compound && !compiled.getTargetPath().has_value() ) {
54 throw std::runtime_error("Expression: object '" + target.value()->name + "' must be assigned by ':=' in '" + expression + "'");
55 }
56 analyseTarget(encoder.objects());
57 }
58
59 bindPaths();
60
61 for ( auto& name : compiled.getVariables() ) {
62 if ( auto input = attributeRegistry.inputs.find(name); input != attributeRegistry.inputs.end() ) {
63 // an input is read as a value only through a path, unless it is the object assigned
64 if ( input->second != sourceInput ) {
65 throw std::runtime_error("Expression: input '" + name + "' must be indexed to a value in '" + expression +"'");
66 }
67 variables.push_back(nullptr);
68 continue;
69 }
70 if ( name != BPMNOS::Keyword::Undefined ) {
71 auto attribute = attributeRegistry[ name ];
72 if ( attribute->isObject() && attribute != sourceObject ) {
73 throw std::runtime_error("Expression: object '" + name + "' must be indexed to a value in '" + expression +"'");
74 }
75 inputs.insert(attribute);
76 variables.push_back(attribute);
77 }
78 }
79 auto& pathDescriptions = compiled.getPaths();
80 for ( size_t path = 0; path < pathDescriptions.size(); path++ ) {
81 if ( auto input = attributeRegistry.inputs.find(pathDescriptions[path].name); input != attributeRegistry.inputs.end() ) {
82 // an input never changes, so that no expression depends on it
83 paths.push_back(nullptr);
84 pathInputs.push_back(input->second);
85 continue;
86 }
87 auto attribute = attributeRegistry[ pathDescriptions[path].name ];
88 inputs.insert(attribute);
89 paths.push_back(attribute);
90 pathInputs.push_back(nullptr);

96/// Returns the text of a path, an index being written `[]`.
97std::string pathText(const LIMEX::Path& path) {
98 std::string text = path.name;
99 for ( auto& step : path.steps ) {
100 text += step.has_value() ? "." + step.value() : "[]";
101 }
102 return text;
103}

105/// The part of an object a path addresses: the layout of the base or field reached, the position of its first
106/// value, and the number of its dimensions indexed.
107struct Location {
108 const BPMNOS::Object::Layout* layout;
109 size_t offset;
110 size_t dimension;
111};

113/// Thrown when a path reads an undefined value or is indexed by one, which makes the expression undefined.
114struct Undefined {};

116/// Returns the number of values one element of the given dimension of a layout spans.
117size_t span(const BPMNOS::Object::Layout& layout, size_t dimension) {
118 size_t result = layout.stride;
119 for ( size_t d = dimension + 1; d < layout.dimensions.size(); d++ ) {
120 result *= layout.dimensions[d];
121 }
122 return result;
123}

125/// Follows the steps of a path through an object, the indices counting from one.
126Location locate(const BPMNOS::Object& object, const LIMEX::Path& path, const std::vector<double>& indices) {
127 Location location{ object.layout.get(), 0, 0 };
128 size_t next = 0;
129 for ( auto& step : path.steps ) {
130 if ( step.has_value() ) {
131 // the fields of an element, every dimension being indexed, as the schema guarantees
132 auto& fields = location.layout->fields;
133 auto field = std::ranges::find_if(fields, [&step](auto& candidate) { return candidate.name == step.value(); });
134 assert( field != fields.end() );
135 location = Location{ &field->layout, location.offset + field->offset, 0 };
136 continue;
137 }
138 double index = indices[next++];
139 if ( std::isnan(index) ) {
140 throw Undefined{};
141 }
142 auto& layout = *location.layout;
143 if ( index < 1 || index != std::floor(index) || (size_t)index > layout.dimensions[location.dimension] ) {
144 throw std::runtime_error(std::format("Expression: illegal index {} for '{}' of size {}", index, pathText(path), layout.dimensions[location.dimension]));
145 }
146 location.offset += ( (size_t)index - 1 ) * span(layout, location.dimension);
147 location.dimension++;
148 }
149 return location;
150}

152/// Returns a view of an array of scalar values with a single dimension, an undefined value being a quiet NaN.
153LIMEX::View<double> valuesOf(const BPMNOS::Object* object, size_t offset, size_t count, size_t step) {
154 return LIMEX::View<double>( count, [object, offset, step](size_t k) -> double {
155 auto& value = object->values[offset + k * step];
156 return value.has_value() ? (double)value.value() : std::numeric_limits<double>::quiet_NaN();
157 });
158}

160/// Returns a view of the given length whose values are never read, for the argument of `size`.
161LIMEX::View<double> lengthOf(size_t count) {
162 return LIMEX::View<double>( count, [](size_t) -> double { return std::numeric_limits<double>::quiet_NaN(); });
163}

165/// Returns the layout of the part of an object a location addresses, which holds the dimensions not indexed.
166BPMNOS::Object::Layout elementLayout(const Location& location) {
167 auto& layout = *location.layout;
168 BPMNOS::Object::Layout element;
169 element.scalar = layout.scalar;
170 element.fields = layout.fields;
171 element.stride = layout.stride;
172 element.dimensions.assign(layout.dimensions.begin() + (long)location.dimension, layout.dimensions.end());
173 if ( !layout.fixed.empty() ) {
174 element.fixed.assign(layout.fixed.begin() + (long)location.dimension, layout.fixed.end());
175 }
176 return element;
177}

179/// Returns a copy of the part of an object a location addresses, whose values are contiguous.
180std::shared_ptr<const BPMNOS::Object> part(const BPMNOS::Object& object, const Location& location) {
181 auto result = std::make_shared<BPMNOS::Object>();
182 result->layout = std::make_shared<const BPMNOS::Object::Layout>( elementLayout(location) );
183 auto begin = object.values.begin() + (long)location.offset;
184 result->values.assign(begin, begin + (long)result->layout->size());

196void Expression::bindPaths() {
197 skipsUndefined.assign( compiled.getPaths().size(), false );
198 readsLengthOnly.assign( compiled.getPaths().size(), false );
199 auto& aggregatorNames = handle.getAggregatorNames();
200 // every path is checked against the schema for the way it is read, which only its position in the syntax tree
201 // tells: as an array or a value, as the argument of `size`, and inside an aggregation over an expression
202 std::function<void(const LIMEX::Node<double>&, bool, bool)> visit = [&](const LIMEX::Node<double>& node, bool aggregated, bool lengthOnly) {
203 if ( node.type == LIMEX::Type::path || node.type == LIMEX::Type::array_path ) {
204 size_t path = std::get<size_t>(node.operands[0]);
205 skipsUndefined[path] = aggregated;
206 readsLengthOnly[path] = lengthOnly;
207 // the target written and the source of an object assigned are bound by the analysis of the target
208 bool written = ( objectTarget && !compound && compiled.getTargetPath() == path );
209 if ( !written && sourcePath != path ) {
210 bind(path, node.type == LIMEX::Type::array_path, lengthOnly);
211 }
212 }
213 if ( node.type == LIMEX::Type::function_call && &node != sourceCall ) {
214 // a lookup returning an array may only be the entire value of an assignment to an array
215 auto& name = handle.getFunctionNames()[ std::get<size_t>(node.operands[0]) ];
216 if ( attributeRegistry.arrayLookups.contains(name) ) {
217 throw std::runtime_error("Expression: lookup '" + name + "' returns an array, which may only be assigned to an array, in '" + expression + "'");
218 }
219 }
220 bool lengthOnlyChild = ( node.type == LIMEX::Type::aggregation && aggregatorNames[ std::get<size_t>(node.operands[0]) ] == "size" );
221 bool aggregatedChild = aggregated || node.type == LIMEX::Type::aggregation_over;
222 for ( auto& operand : node.operands ) {
223 if ( std::holds_alternative< LIMEX::Node<double> >(operand) ) {
224 visit( std::get< LIMEX::Node<double> >(operand), aggregatedChild, lengthOnlyChild );
225 }
226 }
227 };
228 visit(compiled.getRoot(), false, false);
229}

238Expression::Step Expression::walk(size_t path) const {
239 auto& description = compiled.getPaths()[path];
240 auto text = pathText(description);
241 auto schema = schemaOf(description.name);
242 if ( !schema ) {
243 throw std::runtime_error("Expression: '" + text + "' does not address an object in '" + expression +"'");
244 }
245 // the steps are followed through the schema: every index takes the next dimension, and a field may only
246 // follow once every dimension is indexed
247 Step result{ schema, 0, {} };
248 for ( auto& step : description.steps ) {
249 if ( !step.has_value() ) {
250 if ( result.dimension == result.schema->dimensions.size() ) {
251 throw std::runtime_error("Expression: '" + text + "' has more indices than '" + description.name + "' has dimensions in '" + expression +"'");
252 }
253 result.dimension++;
254 continue;
255 }
256 if ( result.dimension < result.schema->dimensions.size() ) {
257 throw std::runtime_error("Expression: field '" + step.value() + "' of an array in '" + text + "', whose dimensions must be indexed first, in '" + expression +"'");
258 }
259 auto field = std::ranges::find_if(result.schema->fields, [&step](auto& candidate) { return candidate.first == step.value(); });
260 if ( field == result.schema->fields.end() ) {
261 throw std::runtime_error("Expression: unknown field '" + step.value() + "' in '" + text + "' in '" + expression +"'");
262 }
263 result.schema = &field->second;
264 result.dimension = 0;
265 result.fields.push_back(step.value());
266 }
267 return result;
268}

270void Expression::bind(size_t path, bool asArray, bool lengthOnly) const {
271 auto& description = compiled.getPaths()[path];
272 auto text = pathText(description);
273 if ( description.steps.empty() && !schemaOf(description.name) ) {
274 throw std::runtime_error("Expression: '" + description.name + "' is not an array in '" + expression +"'");
275 }
276 auto [schema, dimension, _] = walk(path);
277 if ( lengthOnly ) {
278 if ( dimension == schema->dimensions.size() ) {
279 throw std::runtime_error("Expression: '" + text + "' is not an array in '" + expression +"'");
280 }
281 }
282 else if ( asArray ) {
283 if ( !schema->scalar.has_value() || schema->dimensions.size() != dimension + 1 ) {
284 throw std::runtime_error("Expression: '" + text + "' is not an array of values with a single dimension in '" + expression +"'");
285 }
286 }
287 else if ( !schema->scalar.has_value() || dimension != schema->dimensions.size() ) {
288 throw std::runtime_error("Expression: '" + text + "' does not address a value in '" + expression +"'");
289 }

292void Expression::analyseTarget(const std::vector<size_t>& literals) {
293 auto attribute = target.value();
294 auto targetPath = compiled.getTargetPath();
295 if ( resize ) {
296 // the length of the dimension the target addresses is assigned a scalar value
297 if ( targetPath.has_value() ) {
298 auto step = walk(targetPath.value());
299 if ( step.dimension == step.schema->dimensions.size() ) {
300 throw std::runtime_error("Expression: '" + pathText(compiled.getPaths()[targetPath.value()]) + "' is not an array in '" + expression + "'");
301 }
302 resizeFields = step.fields;
303 resizeDimension = step.dimension;
304 }
305 else if ( attribute->schema->dimensions.empty() ) {
306 throw std::runtime_error("Expression: object '" + attribute->name + "' is not an array in '" + expression + "'");
307 }
308 return;
309 }
310 if ( compound ) {
311 // the target is read and written as a value, and bound as such
312 return;
313 }
314 if ( targetPath.has_value() ) {
315 auto step = walk(targetPath.value());
316 objectValue = !( step.schema->scalar.has_value() && step.dimension == step.schema->dimensions.size() );
317 targetFields = step.fields;
318 targetDimension = step.dimension;
319 }
320 else {
321 objectValue = true;
322 }
323 if ( !objectValue ) {
324 return;
325 }
326 // an array or object is assigned a literal, the object of an attribute, or the part of an object a path
327 // addresses
328 auto& assignment = std::get< LIMEX::Node<double> >(compiled.getRoot().operands[0]);
329 auto& value = std::get< LIMEX::Node<double> >(assignment.operands[0]);
330 auto illegal = [&]() {
331 return std::runtime_error("Expression: '" + attribute->name + "' must be assigned a literal, an object, or a path to an array or object in '" + expression + "'");
332 };
333 if ( value.type == LIMEX::Type::literal ) {
334 double number = std::get<double>(value.operands[0]);
335 if ( !std::ranges::contains(literals, (size_t)number) || number != std::floor(number) ) {
336 throw illegal();
337 }
338 sourceLiteral = (size_t)number;
339 }
340 else if ( value.type == LIMEX::Type::variable ) {
341 auto& name = compiled.getVariables()[ std::get<size_t>(value.operands[0]) ];
342 if ( auto input = attributeRegistry.inputs.find(name); input != attributeRegistry.inputs.end() ) {
343 // the object an input holds
344 sourceInput = input->second;
345 return;
346 }
347 auto source = attributeRegistry[ name ];
348 if ( !source->isObject() ) {
349 throw illegal();
350 }
351 sourceObject = source;
352 }
353 else if ( value.type == LIMEX::Type::function_call && attributeRegistry.arrayLookups.contains( handle.getFunctionNames()[ std::get<size_t>(value.operands[0]) ] ) ) {
354 // a lookup returning an array, which arrives as the index of a constant object
355 sourceCall = &value;
356 }
357 else if ( value.type == LIMEX::Type::path ) {
358 size_t path = std::get<size_t>(value.operands[0]);
359 auto step = walk(path);
360 if ( step.schema->scalar.has_value() && step.dimension == step.schema->dimensions.size() ) {
361 throw illegal();
362 }
363 sourcePath = path;
364 }
365 else {
366 throw illegal();
367 }
368}

370std::string Expression::preprocess(const std::string& text) {
371 // 'resize(path) := n' is read as the assignment 'path := n' of the length of the dimension the path addresses
372 static const std::regex form(R"(^\s*resize\s*\‍((.*)\)\s*:=(.*)$)");
373 std::smatch match;
374 if ( std::regex_match(text, match, form) ) {
375 return match[1].str() + " :=" + match[2].str();
376 }
377 return text;
378}

380/**
381 * @brief Resolves the paths of an expression in the objects they address, the object of each path given by its
382 * position.
383 *
384 * A resolver is constructed on the stack for one evaluation and refers to the objects of that evaluation, so that
385 * resolving allocates nothing.
386 */
387class Expression::PathResolver : public LIMEX::Resolver<double> {
388public:
389 PathResolver(const Expression& owner, const std::vector<const BPMNOS::Object*>& objects) : owner(owner), objects(objects) {}
390 double value(size_t path, const std::vector<double>& indices) const override {
391 try {
392 auto location = locate(*objects[path], owner.compiled.getPaths()[path], indices);
393 auto& value = objects[path]->values[location.offset];
394 if ( !value.has_value() ) {
395 throw Undefined{};
396 }
397 return (double)value.value();
398 }
399 catch ( const Undefined& ) {
400 // an aggregation skips an undefined value, which elsewhere makes the expression undefined
401 if ( owner.skipsUndefined[path] ) {
402 return std::numeric_limits<double>::quiet_NaN();
403 }
404 throw;
405 }
406 }
407 LIMEX::View<double> array(size_t path, const std::vector<double>& indices) const override {
408 auto location = locate(*objects[path], owner.compiled.getPaths()[path], indices);
409 auto& layout = *location.layout;
410 if ( owner.readsLengthOnly[path] ) {
411 return lengthOf(layout.dimensions[location.dimension]);
412 }
413 return valuesOf(objects[path], location.offset, layout.dimensions[location.dimension], span(layout, location.dimension));
414 }
415private:
416 const Expression& owner;
417 const std::vector<const BPMNOS::Object*>& objects;
418};

445Expression::Type Expression::getType() const {
446 auto variableNames = compiled.getVariables();
447 assert( compiled.getRoot().operands.size() == 1 );
448 assert( compiled.getRoot().type == LIMEX::Type::group );
449
450 auto& root = compiled.getRoot();
451 assert( !root.operands.empty() );
452 auto& node = std::get< LIMEX::Node<double> >(root.operands[0]);
453
454 // check if any of the variables is named "undefined"
455 if ( std::find( variableNames.begin(), variableNames.end(), BPMNOS::Keyword::Undefined ) != variableNames.end() ) {
456 // only lhs == undefined, lhs != undefined, and lhs := undefined are allowed
457
458 if ( node.type == LIMEX::Type::assign ) {
459 assert( node.operands.size() == 1 );
460 if (
461 !std::holds_alternative< LIMEX::Node<double> >(node.operands[0]) ||
462 std::get< LIMEX::Node<double> >(node.operands[0]).type != LIMEX::Type::variable
463 ) {
464 throw std::runtime_error("Expression: illegal assignment '" + expression +"'");
465 }
466 return Type::UNASSIGN;
467 }
468
469 if ( node.type != LIMEX::Type::equal_to && node.type != LIMEX::Type::not_equal_to ) {
470 throw std::runtime_error("Expression: illegal expression '" + expression +"'");
471 }
472
473 assert( node.operands.size() == 2 );
474 assert( std::holds_alternative< LIMEX::Node<double> >(node.operands[0]) );
475 assert( std::holds_alternative< LIMEX::Node<double> >(node.operands[1]) );
476 auto& lhs = std::get< LIMEX::Node<double> >(node.operands[0]);
477 auto& rhs = std::get< LIMEX::Node<double> >(node.operands[1]);
478 assert( !lhs.operands.empty() );
479 assert( !rhs.operands.empty() );
480
481 if (
482 lhs.type != LIMEX::Type::variable ||
483 variableNames.front() == BPMNOS::Keyword::Undefined ||
484 rhs.type != LIMEX::Type::variable ||
485 variableNames.back() != BPMNOS::Keyword::Undefined
486 ) {
487 throw std::runtime_error("Expression: illegal comparison '" + expression +"'");
488 }
489
490 return node.type == LIMEX::Type::equal_to ? Type::IS_NULL : Type::IS_NOT_NULL;
491 }
492 // all variables must be defined
493 return ( (int)node.type >= (int)LIMEX::Type::assign ) ? Type::ASSIGN : Type::OTHER;
494}

508template <typename DataType>
509std::optional<double> Expression::execute(const BPMNOS::Status& status, const DataType& data) const {
510 if ( type == Type::UNASSIGN ) {
511 return std::nullopt;
512 }
513 if ( type == Type::IS_NULL ) {
514 assert(variables.size() == 1);
515 auto value = attributeRegistry.getValue(variables[0],status,data);
516 return (double)!value.has_value();
517 }
518 if ( type == Type::IS_NOT_NULL ) {
519 assert(variables.size() == 1);
520 auto value = attributeRegistry.getValue(variables[0],status,data);
521 return (double)value.has_value();
522 }
523
524 // the buffers are reused by every evaluation on the thread, so that an evaluation allocates nothing; no evaluation
525 // starts another while they are in use
526 thread_local std::vector<double> variableValues;
527 thread_local std::vector<const BPMNOS::Object*> pathObjects;
528 variableValues.clear();
529 pathObjects.clear();
530 if ( !setValues(status, data, variableValues, pathObjects, false) ) {
531 // return nullopt because a required value is not given
532 return std::nullopt;
533 }
534
535 try {
536 return evaluate(variableValues,pathObjects);
537 }
538 catch (const std::runtime_error& e) {
539 throw std::runtime_error(std::format("Expression: failed to evaluate '{}' with {}\n{}", expression, stringify(status, data), e.what()));
540 }

557template <typename DataType>
558bool Expression::setValues(const BPMNOS::Status& status, const DataType& data, std::vector<double>& variableValues, std::vector<const BPMNOS::Object*>& pathObjects, bool undefinedAsNaN) const {
559 for ( auto attribute : variables ) {
560 if ( !attribute || attribute->isObject() ) {
561 // the object assigned, which LIMEX reads as a variable whose value is not used
562 variableValues.push_back(0);
563 continue;
564 }
565 auto value = attributeRegistry.getValue(attribute,status,data);
566 if ( !value.has_value() && !undefinedAsNaN ) {
567 return false;
568 }
569 variableValues.push_back( value.has_value() ? (double)value.value() : std::numeric_limits<double>::quiet_NaN() );
570 }
571 // the arrays and objects the paths address, a name used as an array being a path without steps
572 for ( size_t path = 0; path < paths.size(); path++ ) {
573 pathObjects.push_back( paths[path] ? objectOf(attributeRegistry, paths[path], status, data) : pathInputs[path]->object.get() );
574 if ( !pathObjects.back() ) {
575 return false;
576 }
577 }
578 return true;
579}

581template <typename DataType>
582void Expression::writeToObject(BPMNOS::Status& status, DataType& data) const {
583 assert( objectTarget );
584 // the buffers are reused by every write on the thread, so that a write allocates nothing for them; no write starts
585 // another while they are in use
586 thread_local std::vector<double> variableValues;
587 thread_local std::vector<const BPMNOS::Object*> pathObjects;
588 variableValues.clear();
589 pathObjects.clear();
590 if ( !setValues(status, data, variableValues, pathObjects, true) ) {
591 throw std::runtime_error("Expression: '" + expression + "' reads an array without a value");
592 }
593 auto& slot = attributeRegistry.getObjectSlot(target.value(), status, data);
594 auto targetPath = compiled.getTargetPath();
595 try {
596 PathResolver resolver(*this, pathObjects);
597 // the indices of the target, all of which must be defined
598 std::vector<double> indices;
599 try {
600 indices = compiled.evaluateTargetIndices(variableValues, resolver);
601 }
602 catch ( const Undefined& ) {
603 throw std::runtime_error("Expression: undefined index");
604 }
605 auto locateTarget = [&]() {
606 try {
607 return locate(*slot, compiled.getPaths()[targetPath.value()], indices);
608 }
609 catch ( const Undefined& ) {
610 throw std::runtime_error("Expression: undefined index");
611 }
612 };
613
614 if ( resize ) {
615 auto length = evaluate(variableValues, pathObjects);
616 if ( !length.has_value() || length.value() < 0 || length.value() != std::floor(length.value()) ) {
617 throw std::runtime_error("Expression: resize requires a non-negative integer");
618 }
619 if ( targetPath.has_value() ) {
620 // the indices of the path are checked, the dimension changing in every element alike
621 locateTarget();
622 }
623 slot = BPMNOS::resized(*slot, resizeFields, resizeDimension, (size_t)length.value());
624 return;
625 }
626
627 if ( !objectValue ) {
628 // a value is written to an element, which never changes a length
629 auto value = evaluate(variableValues, pathObjects);
630 auto location = locateTarget();
631 modifiable(slot).values[location.offset] = BPMNOS::to_value(value);
632 return;
633 }
634
635 // the array or object assigned
636 std::shared_ptr<const BPMNOS::Object> source;
637 if ( sourceLiteral.has_value() ) {
638 source = objectRegistry[sourceLiteral.value()];
639 }
640 else if ( sourceCall ) {
641 // the value of the assignment is the index of the constant object the lookup returns
642 double index = 0;
643 try {
644 index = compiled.evaluate(variableValues, resolver);
645 }
646 catch ( const Undefined& ) {
647 throw std::runtime_error("Expression: undefined argument");
648 }
649 source = objectRegistry[(size_t)index];
650 }
651 else if ( sourceObject ) {
652 source = attributeRegistry.getObject(sourceObject, status, data);
653 }
654 else if ( sourceInput ) {
655 source = sourceInput->object;
656 }
657 else {
658 // the indices of the source path are those LIMEX evaluates for the value of the assignment, which a
659 // resolver records instead of resolving the source path
660 class SourceResolver : public PathResolver {
661 public:
662 SourceResolver(const Expression& owner, const std::vector<const BPMNOS::Object*>& objects, size_t position) : PathResolver(owner, objects), position(position) {}
663 double value(size_t path, const std::vector<double>& pathIndices) const override {
664 if ( path == position ) {
665 sourceIndices = pathIndices;
666 return 0;
667 }
668 return PathResolver::value(path, pathIndices);
669 }
670 mutable std::vector<double> sourceIndices; ///< The indices of the source path
671 private:
672 size_t position; ///< The position of the source path
673 };
674 SourceResolver capturing(*this, pathObjects, sourcePath.value());
675 try {
676 compiled.evaluate(variableValues, capturing);
677 source = part(*pathObjects[sourcePath.value()], locate(*pathObjects[sourcePath.value()], compiled.getPaths()[sourcePath.value()], capturing.sourceIndices));
678 }
679 catch ( const Undefined& ) {
680 throw std::runtime_error("Expression: undefined index");
681 }
682 }
683
684 if ( !targetPath.has_value() ) {
685 // the whole object keeps the lengths of its fixed dimensions and takes those of its open ones
686 attributeRegistry.setObject(target.value(), status, data, source);
687 return;
688 }
689
690 // an element keeps its lengths, except those not yet determined, which the value determines for every
691 // element of the array
692 auto location = locateTarget();
693 auto element = elementLayout(location);
694 auto value = Schema::of(element, true).conform(source);
695 std::function<void(const BPMNOS::Object::Layout&, const BPMNOS::Object::Layout&, std::vector<std::string>, size_t)> determine =
696 [&](const BPMNOS::Object::Layout& current, const BPMNOS::Object::Layout& given, std::vector<std::string> fields, size_t first) {
697 for ( size_t d = 0; d < current.dimensions.size(); d++ ) {
698 if ( current.dimensions[d] == 0 && given.dimensions[d] > 0 ) {
699 slot = BPMNOS::resized(*slot, fields, first + d, given.dimensions[d]);
700 }
701 }
702 for ( size_t k = 0; k < current.fields.size(); k++ ) {
703 auto nested = fields;
704 nested.push_back(current.fields[k].name);
705 determine(current.fields[k].layout, given.fields[k].layout, nested, 0);
706 }
707 };
708 determine(element, *value->layout, targetFields, targetDimension);
709 location = locateTarget();
710 auto& values = modifiable(slot).values;
711 assert( location.offset + value->values.size() <= values.size() );
712 std::ranges::copy(value->values, values.begin() + (long)location.offset);
713 }
714 catch ( const std::runtime_error& error ) {
715 throw std::runtime_error(std::format("Expression: failed to evaluate '{}' with {}\n{}", expression, stringify(status, data), error.what()));
716 }
717}
