46void CachedCandidates<WeakPtrs...>::addEvaluation(WeakPtrs... weak_ptrs, std::shared_ptr<Decision> decision) {
47 assert ( decision );
48
49 // candidates are sorted in descending order of reward; decisions without reward are assumed infeasible (-infinity, sorting last).
50 // The decision is already owned in the base `decisions`, so this only lists the (weak) candidate.
51 auto reward = decision->reward();
52 this->candidates.emplace( (reward.has_value() ? (double)reward.value() : -std::numeric_limits<double>::infinity() ), weak_ptrs..., decision->weak_from_this(), decision->evaluation);
53
54 // weak-index the evaluation for invalidation; invariant decisions never invalidate, so they need no index
55 if ( decision->timeDependent && decision->dataDependencies.empty() ) {
56 timeDependentEvaluations.emplace_back(weak_ptrs..., decision );
