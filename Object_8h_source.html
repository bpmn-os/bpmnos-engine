13/**
14 * @brief An object, i.e. an array or a structured value, stored flat.
15 *
16 * The values of an object are held in one vector, and its layout states where each of them is: an element is
17 * one slot at an offset computed from the layout. An object is shared copy-on-write wherever it is held, as a
18 * `std::shared_ptr<const Object>`, so that copying a status or data copies no values.
19 */
20struct Object {
21 /**
22 * @brief The shape of an object with the sizes of its dimensions known.
23 *
24 * A layout is a base followed by its dimensions, read from left to right in the order of the indices. The
25 * base is a scalar type or a list of fields, each with a layout of its own, stored one after the other within
26 * an element. The elements of an array are uniform, so that they share their layout and are a stride apart:
27 * the element at the indices `i_1, ..., i_k`, counting from zero, begins at
28 * `((i_1 * d_2 + i_2) * d_3 + ... + i_k) * stride`.
29 */
30 struct Layout {
31 struct Field;
32 std::optional<ValueType> scalar; ///< The scalar type of the base, or nullopt if the base has fields
33 std::vector<Field> fields; ///< The fields of the base, in declaration order
34 std::vector<size_t> dimensions; ///< The sizes of the dimensions, in index order
35 size_t stride = 1; ///< The number of values of one element of the base
36 /// For each dimension whether its size is fixed, by the type or a size declaration, so that an assignment
37 /// pads a shorter value and rejects a longer one, whereas an open dimension takes the length of the value;
38 /// empty for a constant object
39 std::vector<bool> fixed;
40
41 /// @brief Returns the number of values of an object of this layout.
42 size_t size() const;
43 /// @brief Returns the type the layout states, in the syntax of attribute types.
44 std::string stringify() const;
45 /// @brief Recomputes the offsets of the fields and the stride from the layouts of the fields.
46 void arrange();
47 bool operator==(const Layout& other) const;
48 };
49
50 std::shared_ptr<const Layout> layout;
51 std::vector<Value> values;
52
53 /// @brief Returns true if the object is an array of scalar values with a single dimension.
54 bool isVector() const { return layout->scalar.has_value() && layout->dimensions.size() == 1; }
55};
