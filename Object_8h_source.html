41/**
42 * @brief The geometry of an object: a single value, an array or an object, the schema it lays out, and the number of
43 * positions it occupies in the values of the object.
44 *
45 * A layout and its schema agree: the layout has the structure of the schema, the schema states the value type of
46 * every single value and the name of every field, and every array the schema fixes has the size the schema states.
47 * The layout of an element or field refers to the corresponding part of the schema. A layout never changes once
48 * built, and refers to the layouts it contains through `std::shared_ptr<const Layout>`, so that a layout derived from
49 * another shares every part it does not change.
50 */
51struct Layout : std::variant<std::monostate, ArrayLayout, ObjectLayout> {
52 std::shared_ptr<const Schema> schema; ///< The schema of the object
53 size_t size; ///< The number of positions the layout occupies
54
55 /// @brief Constructs the layout of a single value of the given schema, a value type.
56 explicit Layout(std::shared_ptr<const Schema> schema);
57 /// @brief Constructs the layout of an array of the given schema, an array type, with the given element layout and
58 /// number of elements.
59 Layout(std::shared_ptr<const Schema> schema, std::shared_ptr<const Layout> layout, size_t size);
60 /// @brief Constructs the layout of an object of the given schema, an object type, with the given layouts of its
61 /// fields, the offsets following from their sizes.
62 Layout(std::shared_ptr<const Schema> schema, const std::vector< std::shared_ptr<const Layout> >& fields);
63
64 /// @brief Returns the schema in its notation with every array at its actual size.
65 std::string stringify() const;
66 /// @brief Returns true if the layouts have the same structure, sizes and offsets and their schemas are equal.
67 bool operator==(const Layout& other) const;
68};

70/**
71 * @brief An object, i.e. an array or a structured value, stored flat.
72 *
73 * The values of an object are held in one vector, and its layout states where each of them is and, through its
74 * schema, what value type it has. An object is shared copy-on-write wherever it is held, as a
75 * `std::shared_ptr<const Object>`, so that copying a status or data copies no values.
76 */
77struct Object {
78 std::shared_ptr<const Layout> layout;
79 std::vector<Value> values;
80
81 Object() = default;
82 /**
83 * @brief Constructs an object of the given schema with every value undefined, every open array having size 0.
84 */
85 explicit Object(const Schema& schema);
86 /**
87 * @brief Constructs an object of the given schema holding the values of the given JSON value.
88 *
89 * A JSON array is an array, whose elements have equal sizes at an open array and at most the declared size at a
90 * fixed one, a JSON object is an object, whose keys must be fields of the schema, and a number, truth value or
91 * text is a single value of the value type the schema states, `null` being undefined. A fixed array is padded with
92 * undefined values, an open array takes the size of the JSON array, and a field the JSON lacks is undefined.
93 */
94 Object(const Schema& schema, const nlohmann::json& json);
95 /**
96 * @brief Constructs an object of the given schema, the target, from the given source object, whose schema is the
97 * schema of its layout.
98 *
99 * At a value type, the source holds a single value of the same value type; arrays and objects are not cast. At an
100 * array type, the source is an array. A fixed array of size n has size n, a shorter source being padded with
101 * undefined elements and a longer source being cut off if every value beyond the n-th element is undefined and
102 * rejected otherwise; an open array has the size of the source. At an object type, the source is an object whose
103 * fields are fields of the target; a field the source lacks is undefined, every open array in it having size 0.
104 * Every part of the layout determined that equals the corresponding part of the layout of the source is that
105 * part, so that it is shared. A source that cannot be converted is rejected with an error.
106 */
107 Object(const Schema& schema, const Object& source);
108};
