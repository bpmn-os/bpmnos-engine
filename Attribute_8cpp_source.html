98std::unique_ptr<const Expression> Attribute::getExpression(std::string& input, AttributeRegistry& attributeRegistry) {
99 if ( !input.contains(":=") ) {
100 return nullptr;
101 }
102
103 if ( schema ) {
104 // an object initialised with a literal, which is registered as a constant object and checked against the
105 // schema, is shared by every instance and needs no expression; an object initialised otherwise is written by
106 // an expression compiled once the attribute is registered, when its scope is created
107 auto value = BPMNOS::trim_copy( input.substr( input.find(":=") + 2 ) );
108 InputEncoder encoder(value);
109 if ( !encoder.object().has_value() ) {
110 initialExpression = input;
111 return nullptr;
112 }
113 initialObject = encoder.object().value();
114 try {
115 schema->conform( objectRegistry[initialObject.value()] );
116 }
117 catch ( const std::exception& error ) {
118 throw std::runtime_error("Attribute: illegal initial value of object '" + id + "'.\n" + error.what());
119 }
120 return nullptr;
121 }
122
123 auto expression = std::make_unique<const Expression>(InputEncoder(input),attributeRegistry,true);
124 auto& root = expression->compiled.getRoot();
125 assert( root.operands.size() == 1 );
126 assert( root.type == LIMEX::Type::group );
127 auto& node = std::get< LIMEX::Node<double> >(root.operands[0]);
128 if ( node.type != LIMEX::Type::assign ) {
129 throw std::runtime_error("Attribute: illegal initialization '" + input + "' for attribute '" + id + "'");
130 }
131 return expression;
132}
