98std::variant<std::monostate, size_t, std::unique_ptr<const Expression>> Attribute::getInitialization(std::string& input, AttributeRegistry& attributeRegistry) {
99 if ( !input.contains(":=") ) {
100 return std::monostate{};
101 }
102
103 if ( isObject() ) {
104 // an object initialised with a literal, which is registered as a constant object and checked against the
105 // type, is shared by every instance and needs no expression; an object initialised otherwise is written by
106 // an expression compiled once the attribute is registered, when its scope is created
107 auto value = BPMNOS::trim_copy( input.substr( input.find(":=") + 2 ) );
108 InputEncoder encoder(value);
109 if ( !encoder.getObjectIndex().has_value() ) {
110 return std::monostate{};
111 }
112 size_t literal = encoder.getObjectIndex().value();
113 try {
114 // the conversion rejects a literal that does not fit the type
115 auto& constant = *objectRegistry[literal];
116 if ( !type.supports(*constant.layout->schema) ) {
117 BPMNOS::Object(type, constant);
118 }
119 }
120 catch ( const std::exception& error ) {
121 throw std::runtime_error("Attribute: illegal initial value of object '" + id + "'.\n" + error.what());
122 }
123 return literal;
124 }
125
126 auto expression = std::make_unique<const Expression>(InputEncoder(input),attributeRegistry,true);
127 auto& root = expression->compiled.getRoot();
128 assert( root.operands.size() == 1 );
129 assert( root.type == LIMEX::Type::group );
130 auto& node = std::get< LIMEX::Node<double> >(root.operands[0]);
131 if ( node.type != LIMEX::Type::assign ) {
132 throw std::runtime_error("Attribute: illegal initialization '" + input + "' for attribute '" + id + "'");
133 }
134 return expression;
135}
