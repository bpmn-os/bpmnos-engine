25 if ( observable->getObservableType() == Observable::Type::EntryRequest ) {

28 if ( !config.sequential ) {
29 assert( request->token->node->parent );
30 if ( request->token->node->parent->represents<BPMNOS::Model::SequentialAdHocSubProcess>() ) {
31 // entries of sequential ad-hoc subprocess children are left to a competing source
32 return;

34 }

36 addDecision( request->token->weak_from_this(), request->weak_from_this(), decision );
37 }
38 else if ( observable->getObservableType() == Observable::Type::SystemState ) {
39 clear(); // start from a clean cache, then rebuild from the installed state

46 }
