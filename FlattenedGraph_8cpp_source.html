573 }
574
575 auto& [flowNodeEntry,flowNodeExit] = vertexMap.at({instanceId,scopeEntry->loopIndices,flowNode});
576
577 if ( flowNode->represents<BPMN::UntypedStartEvent>() || flowNode->represents<BPMN::TypedStartEvent>() ) {
578 flowNodeEntry->inflows.emplace_back(nullptr,scopeEntry);
579 scopeEntry->outflows.emplace_back(nullptr,flowNodeEntry);
580 }
581
582 if ( flowNode->represents<BPMN::Activity>() && flowNode->incoming.empty() ) {
583 if( !flowNode->parent->represents<BPMNOS::Model::SequentialAdHocSubProcess>() ) {
584 throw std::runtime_error("FlattenedGraph: only activities within adhoc subprocesses maybe without incoming sequence flow");
585 }
586 flowNodeEntry->inflows.emplace_back(nullptr,scopeEntry);
587 scopeEntry->outflows.emplace_back(nullptr,flowNodeEntry);
588 }
589
590 if ( flowNode->outgoing.empty() ) {
591 // flow node without outgoing sequence flow
592 scopeExit->inflows.emplace_back(nullptr,flowNodeExit);
593 flowNodeExit->outflows.emplace_back(nullptr,scopeExit);
594 }
595
596 // flatten child scopes
597 if ( auto subprocess = flowNode->represents<BPMN::SubProcess>();
598 subprocess &&
599 subprocess->loopCharacteristics.has_value()
600 ) {
601 // nothing to be done because loops of each subprocess are already flattened
602 }
603 else if ( auto childScope = flowNode->represents<BPMN::Scope>() ) {
604 flatten( flowNodeEntry->instanceId, childScope, flowNodeEntry, flowNodeExit );
605 }
606
607 }
608
609//std::cerr << "sequence flows: " << scope->sequenceFlows.size() << std::endl;
610 // sequence flows
611 for ( auto& sequenceFlow : scope->sequenceFlows ) {
612 // create unique flow from origin to destination
613 Vertex* origin = vertexMap.at({instanceId,scopeEntry->loopIndices,sequenceFlow->source}).second;
614 Vertex* destination = vertexMap.at({instanceId,scopeEntry->loopIndices,sequenceFlow->target}).first;
615 origin->outflows.emplace_back(sequenceFlow.get(),destination);
616 destination->inflows.emplace_back(sequenceFlow.get(),origin);
617 }
618
619 // boundary events
620 for ( auto& flowNode : scope->flowNodes ) {
621 if ( auto boundaryEvent = flowNode->represents<BPMN::BoundaryEvent>() ) {
622 throw std::runtime_error("FlattenedGraph: Boundary event '" + boundaryEvent->id + "' is not supported");
623 }
624 }
625
626 // event-subprocesses
627 for ( auto& eventSubProcess : scope->eventSubProcesses ) {
628 if ( eventSubProcess->startEvent->isInterrupting ) {
629 // interrupting event-subprocesses
630 throw std::runtime_error("FlattenedGraph: Interrupting event-subprocess '" + eventSubProcess->id + "' is not supported");
631// flatten( instanceId, eventSubProcess, scopeEntry, scopeExit );
632 }
633 else {
634 // non-interrupting event-subprocesses
635 if ( eventSubProcess->startEvent->represents<BPMN::MessageStartEvent>() ) {
636 addNonInterruptingEventSubProcess(eventSubProcess, scopeEntry, scopeExit);
637 }
638 else {
639 throw std::runtime_error("FlattenedGraph: Type of non-interrupting event-subprocess '" + eventSubProcess->id + "' is not supported");
640 }
641 }
642 }
643
644 // compensation nodes
645 for ( auto& flowNode : scope->flowNodes ) {
646 auto activity = flowNode->represents<BPMN::Activity>();
647 if ( activity && activity->isForCompensation ) {
648 throw std::runtime_error("FlattenedGraph: Compensation activity '" + activity->id + "' is not supported");
649 }
650 }
651}

667std::vector< const FlattenedGraph::Vertex* > FlattenedGraph::sortVertices() const {
668 std::vector< const Vertex* > sortedVertices;
669 sortedVertices.reserve( vertices.size() );
670 for ( auto initialVertex : initialVertices ) {
671 std::unordered_map<const Vertex*, size_t> inDegree;
672
673 std::deque<const Vertex*> queue;
674 queue.push_back(initialVertex);
675
676 // determine vertices in topological order
677 while ( !queue.empty() ) {
678//std::cerr << "Queue " << queue.size() << std::endl;
679 const Vertex* current = queue.front();
680 queue.pop_front();
681 sortedVertices.push_back(current);
682 inDegree.erase(current);
683
684 for ( auto& [_,vertex] : current->outflows ) {
685 if ( !inDegree.contains(vertex) ) {
686 // initialize in degree
687 inDegree[vertex] = vertex->inflows.size() + vertex->predecessors.size();
688 }
689 // decrement in degree and add vertex to queue if zero
690 if ( --inDegree.at(vertex) == 0 ) {
691 queue.push_back(vertex);
692 }
693 }
694 for ( auto vertex : current->successors ) {
695 if ( !inDegree.contains(vertex) ) {
696 // initialize in degree
697 inDegree[vertex] = vertex->inflows.size() + vertex->predecessors.size();
698 }
699//std::cerr << "Vertex " << vertex.reference() << " has inDegree " << inDegree.at(&vertex) << "/" << vertex.inflows.size() << "/" << vertex.predecessors.size() << std::endl;
700 // decrement in degree and add vertex to queue if zero
701 if ( --inDegree.at(vertex) == 0 ) {
702 queue.push_back(vertex);
703 }
704 }
705 }
706
707 if ( inDegree.size() ) {
708 throw std::runtime_error("FlattenedGraph: cycle detected in '" + BPMNOS::to_string(initialVertex->rootId, STRING) + "'");
709 }
710 }
711//std::cerr << "sortedVertices:" << sortedVertices.size() << std::endl;
712 return sortedVertices;
713}

715const FlattenedGraph::Vertex* FlattenedGraph::getVertex( const Token* token ) const {
716//std::cerr << "getVertex(" << token->jsonify() << ")" << std::endl;
717 auto node = token->node ? token->node->as<BPMN::Node>() : token->owner->process->as<BPMN::Node>();
718 if( !loopIndexAttributes.contains(node) ) {
719 // unreachable typed start event
720 assert( node->represents<BPMN::TypedStartEvent>() );
721 return nullptr;
722 }
723
724 // determine loop indices from token
725 std::vector< size_t > loopIndices;
726 for ( auto attribute : loopIndexAttributes.at(node) ) {
727//std::cerr << attribute->id << std::endl;
728 if ( !token->status.at(attribute->index).has_value() ) {
729 // loop index has not yet been set
730 break;
731 }
732 loopIndices.push_back( (size_t)token->status.at(attribute->index).value() );
733 }
734
735 auto instanceId = token->data->at(BPMNOS::Model::ExtensionElements::Index::Instance).get().value();
736//std::cerr << stringRegistry[(size_t)instanceId] << "/" << node->id << "/" << token->jsonify() << std::endl;
737 if( !vertexMap.contains({instanceId,loopIndices,node}) ) {
738 return nullptr;
739 }
740 assert( vertexMap.contains({instanceId,loopIndices,node}) );
741 auto& [entry,exit] = vertexMap.at({instanceId,loopIndices,node});
742 return (token->state == Token::State::ENTERED) ? entry : exit;
743}
