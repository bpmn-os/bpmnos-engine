60std::vector< std::tuple<const BPMN::Process*, BPMNOS::Values, BPMNOS::Values> > DynamicScenario::getCurrentInstantiations(const BPMNOS::number currentTime) const {
61 std::vector< std::tuple<const BPMN::Process*, BPMNOS::Values, BPMNOS::Values> > result;
62 for ( auto& [id, instance] : instances ) {
63 // Effective instantiation time is max(instantiationTime, processDisclosure).
64 // If process data disclosure is later than instantiation time, instantiation is delayed
65 // until all process data is disclosed.
66 BPMNOS::number effectiveInstantiationTime = instance.instantiationTime;
67 if ( disclosure.contains(instance.id) && disclosure.at(instance.id).contains(instance.process) ) {
68 effectiveInstantiationTime = std::max(effectiveInstantiationTime, disclosure.at(instance.id).at(instance.process));
69 }
70 if ( effectiveInstantiationTime == currentTime ) {
71 auto status = getKnownInitialStatus(&instance, currentTime);
72 // If instantiation was delayed due to disclosure, update timestamp to reflect actual instantiation time
73 if ( effectiveInstantiationTime > instance.instantiationTime ) {
74 status[ExtensionElements::Index::Timestamp] = currentTime;
75 }
76 result.push_back({instance.process, std::move(status), getKnownInitialData(&instance, currentTime)});
77 }
78 }
79 return result;
80}

98std::optional<BPMNOS::number> DynamicScenario::getValue(const Scenario::InstanceData* instance, const BPMNOS::Model::Attribute* attribute, [[maybe_unused]] const BPMNOS::number currentTime) const {
99 // Node-level disclosure is checked by caller (getCurrentInstantiations, getStatus, getData)
100 if ( attribute->expression && attribute->expression->type == Expression::Type::ASSIGN ) {
101 // Value is computed from an expression
102 std::vector<double> variableValues;
103 for ( auto input : attribute->expression->variables ) {
104 if ( !input->isImmutable ) {
105 return std::nullopt;
106 }
107 auto value = getValue(instance, input, currentTime);
108 if ( !value.has_value() ) {
109 return std::nullopt;
110 }
111 variableValues.push_back( (double)value.value() );
112 }
113
114 std::vector<std::vector<double>> collectionValues;
115 for ( auto input : attribute->expression->collections ) {
116 if ( !input->isImmutable ) {
117 return std::nullopt;
118 }
119 collectionValues.push_back( {} );
120 auto collection = getValue(instance, input, currentTime);
121 if ( !collection.has_value() ) {
122 return std::nullopt;
123 }
124 for ( auto value : collectionRegistry[(size_t)collection.value()] ) {
125 collectionValues.back().push_back( value );
126 }
127 }
128
129 return number(attribute->expression->compiled.evaluate(variableValues, collectionValues));
130 }
131 else {
132 // Return stored value directly
133 if ( instance->values.contains(attribute) ) {
134 return instance->values.at(attribute);
135 }
136 // Check pending disclosures
137 if ( pendingDisclosures.contains(instance->id) ) {
138 for ( auto& pending : pendingDisclosures.at(instance->id) ) {
139 if ( pending.attribute == attribute && currentTime >= pending.disclosureTime ) {
140 return pending.value;
141 }
142 }
143 }
144 }
145 return std::nullopt;
146}

196void DynamicScenario::revealData(BPMNOS::number currentTime) const {
197 for (auto& [instanceId, pendings] : pendingDisclosures) {
198 auto& instance = instances.at(instanceId);
199
200 // Process pending disclosures that are due
201 auto it = pendings.begin();
202 while (it != pendings.end()) {
203 if (currentTime >= it->disclosureTime) {
204 // Reveal pre-computed value
205 instance.values[it->attribute] = it->value;
206 disclosedAttributes.insert({instanceId, it->attribute});
207 it = pendings.erase(it);
208 }
209 else {
210 ++it;
211 }

218 BPMNOS::number instanceId,
219 const BPMN::Node* activity,
220 BPMNOS::number currentTime
221) const {
222 // disclosure is keyed by the root id (specified pre-runtime)
223 if (disclosure.contains((size_t)rootId) && disclosure.at((size_t)rootId).contains(activity)) {
224 if (currentTime < disclosure.at((size_t)rootId).at(activity)) {

226 }
227 }
228
229 // status keyed by the full instance id; declared instance/values keyed by root id
230 auto key = std::make_pair((size_t)instanceId, activity);
231 if (!activityArrivalStatus.contains(key)) {
232 return std::nullopt;
233 }
234
235 auto& instance = instances.at((size_t)rootId);
236 auto extensionElements = activity->extensionElements->as<const BPMNOS::Model::ExtensionElements>();
237
238 // Start with parent's status
239 Values result = activityArrivalStatus.at(key);
240
241 // Add node's own attributes (same approach as getStatus)
242 for (auto& attribute : extensionElements->attributes) {
243 result.push_back(getValue(&instance, attribute.get(), currentTime));
