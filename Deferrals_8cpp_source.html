28void Deferrals::notice(const Observable* observable) {
29 switch ( observable->getObservableType() ) {
30 case Observable::Type::SystemState:
31 // none of the deferred tokens exists in an installed state
32 systemState = static_cast<const SystemState*>(observable);
33 deferrals.clear();
34 break;
35 case Observable::Type::EntryRequest:
36 case Observable::Type::ChoiceRequest:
37 case Observable::Type::MessageDeliveryRequest:
38 // a request arriving at a deferred decision point is a new candidate, and the first request of a token taking
39 // the address of a destroyed token ends the stale deferral of the latter
40 if ( !deferrals.empty() ) {
41 deferrals.erase( decisionPoint(static_cast<const DecisionRequest*>(observable)->token) );
42 }
43 break;
44 case Observable::Type::Message:
45 message(static_cast<const Message*>(observable));
46 break;
47 case Observable::Type::DataUpdate:
48 dataUpdate(static_cast<const DataUpdate*>(observable));
49 break;
50 default:
51 break;
52 }
53}

55void Deferrals::defer(const Token* token) {
56 auto point = decisionPoint(token);
57 auto deferral = deferralOf(point);
58 if ( !deferral ) {
59 throw std::runtime_error("Deferrals: no deferral stated for the decision point at '" + point->node->id + "'");
60 }
61
62 std::vector< std::weak_ptr<const DecisionRequest> > requests;
63 if ( point->decisionRequest && point->decisionRequest->type != Observable::Type::EntryRequest ) {
64 requests.push_back(point->decisionRequest);
65 }
66 else {
67 // the entry requests of the tokens awaiting entry at the performer
68 for ( auto& [ token_ptr ] : point->pendingSequentialEntries ) {
69 if ( auto waitingToken = token_ptr.lock(); waitingToken && waitingToken->decisionRequest ) {
70 requests.push_back(waitingToken->decisionRequest);
71 }
72 }
73 }
74 if ( requests.empty() ) {
75 throw std::runtime_error("Deferrals: no pending request at the decision point at '" + point->node->id + "'");
76 }
77
78 deferrals.insert_or_assign(point, Entry{ std::move(requests), deferral });
79}

94void Deferrals::message(const Message* message) {
95 if ( deferrals.empty() || message->state != Message::State::CREATED ) {
96 return;
97 }
98 // a message which may be delivered to a deferred catch event is a new candidate, as message deliveries pair them
99 auto& recipientCandidates = message->origin->extensionElements->as<BPMNOS::Model::ExtensionElements>()->messageCandidates;
100 std::erase_if(deferrals, [message, &recipientCandidates](const auto& item) {
101 auto& [token, entry] = item;
102 auto request = std::get<0>(entry).front().lock();
103 return
104 request &&
105 request->type == Observable::Type::MessageDeliveryRequest &&
106 std::ranges::contains(recipientCandidates, token->node) &&
107 message->matches(static_cast<const MessageDeliveryRequest*>(request.get())->recipientHeader)
108 ;
109 });
110}

112void Deferrals::dataUpdate(const DataUpdate* update) {
113 if ( deferrals.empty() ) {
114 return;
115 }
116
117 // a written global attribute concerns the deferrals of every instance, any other written attribute only those of the
118 // instance that wrote it
119 std::vector<const BPMNOS::Model::Attribute*> globalAttributes;
120 std::ranges::copy_if(update->attributes, std::back_inserter(globalAttributes), [](const BPMNOS::Model::Attribute* attribute) { return attribute->isGlobal; });
121
122 std::erase_if(deferrals, [this, update, &globalAttributes](const auto& item) {
123 auto& [token, entry] = item;
124 if ( !isAlive(entry) ) {
125 return true;
126 }
127 auto deferral = std::get<const BPMNOS::Model::Deferral*>(entry);
128 auto& attributes = ( instanceOf(token) == update->instanceId ) ? update->attributes : globalAttributes;
129 if ( std::ranges::any_of(attributes, [deferral](auto attribute) { return std::ranges::contains(deferral->observables, attribute); }) ) {
130 return true;
131 }
132 if (
133 std::ranges::any_of(attributes, [deferral](auto attribute) { return deferral->condition.inputs.contains(attribute); }) &&
134 !deferral->conditionHolds(token->status, *token->data)
135 ) {
136 return true;
137 }
138 // the update of the time at a clock tick, which names the instance 0, ends a deferral no reconsideration of which
139 // can come any more
140 return update->instanceId == BPMNOS::number(0) && !isAdmissible(deferral);
141 });
142}

156const BPMNOS::Model::Deferral* Deferrals::deferralOf(const Token* token) {
157 auto extensionElements = token->node->extensionElements->as<BPMNOS::Model::ExtensionElements>();
158 if ( token->decisionRequest && token->decisionRequest->type == Observable::Type::ChoiceRequest ) {
159 return extensionElements->choiceGuidance.has_value() ? extensionElements->choiceGuidance.value()->deferral.get() : nullptr;
160 }
161 if ( token->decisionRequest && token->decisionRequest->type == Observable::Type::MessageDeliveryRequest ) {
162 return extensionElements->messageDeliveryGuidance.has_value() ? extensionElements->messageDeliveryGuidance.value()->deferral.get() : nullptr;
163 }
164 return extensionElements->performerDeferral.get();
