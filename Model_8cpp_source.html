89/// @brief Returns the {name, source, header} of every lookup table a data store declares.
90/// @throws std::runtime_error if a source contains a path separator (a source must be a bare file name).
91std::vector<std::tuple<std::string, std::string, std::string>> lookupTablesOf(const XML::bpmn::tDataStore& dataStore) {
92 std::vector<std::tuple<std::string, std::string, std::string>> lookups;
93 auto extensionElements = dataStore.getOptionalChild<XML::bpmn::tExtensionElements>();
94 if ( !extensionElements.has_value() ) {
95 return lookups;
96 }
97 auto tables = extensionElements->get().getOptionalChild<XML::bpmnos::tTables>();
98 if ( !tables.has_value() ) {
99 return lookups;
100 }
101 for ( const XML::bpmnos::tTable& table : tables->get().find<XML::bpmnos::tTable>() ) {
102 std::string name = table.getRequiredAttributeByName("name").value;
103 std::string source = table.getRequiredAttributeByName("source").value;
104 std::string header = table.getRequiredAttributeByName("header").value;
105 if ( source.find('/') != std::string::npos || source.find('\\') != std::string::npos ) {
106 throw std::runtime_error("Model: lookup table source '" + source + "' must be a file name, not a path");
107 }
108 lookups.emplace_back( std::move(name), std::move(source), std::move(header) );
109 }
110 return lookups;
111}

320std::unique_ptr<BPMN::FlowNode> Model::createMessageStartEvent(XML::bpmn::tStartEvent* startEvent, BPMN::Scope* parent) {
321 auto baseElement = BPMN::Model::createMessageStartEvent(startEvent,parent);
322 auto extensionElements = std::make_unique<BPMNOS::Model::ExtensionElements>(startEvent,parent->extensionElements->as<ExtensionElements>()->attributeRegistry,parent);
323
324 // Lambda function to compare unique_ptrs by their raw pointers
325 auto contains = [](const std::vector<std::unique_ptr<Attribute>>& attributes, Attribute* attribute) {
326 return std::any_of(attributes.begin(), attributes.end(),[attribute](const std::unique_ptr<Attribute>& ptr) {
327 return ptr.get() == attribute;
328 });
329 };
330
331 for ( auto& messageDefinition : extensionElements->messageDefinitions ) {
332 for ( auto& [_,content] : messageDefinition->contentMap ) {
333 Attribute* attribute = content->attribute;
334 auto parentExtension = parent->extensionElements->as<BPMNOS::Model::ExtensionElements>();
335 if ( attribute->category == Attribute::Category::GLOBAL ) {
336 throw std::runtime_error("Model: Message start event '" + baseElement->id + "' attempts to modify global attribute '" + attribute->id + "'");
337 }
338 else if ( attribute->category == Attribute::Category::DATA ) {
339 if ( !contains(parentExtension->data,attribute) ) {
340 throw std::runtime_error("Model: Message start event '" + baseElement->id + "' attempts to modify data attribute '" + attribute->id + "' which is not owned by event-subprocess");
341 }
342 // data attributes owned by event-subprocesses are considered immutable even if they are modified by the message start event
343 }
344 else if ( attribute->category == Attribute::Category::STATUS ) {
345 // status attributes owned by event-subprocesses are considered immutable even if they are modified by the message start event
346 if ( !contains(parentExtension->attributes,attribute) ) {
347 attribute->isImmutable = false;
348 }
349 }
350 }
351 }
352 // bind attributes, restrictions, and operators to all event subprocesses
353 return bind<BPMN::FlowNode>( std::move(baseElement), std::move(extensionElements) );
354}

404void Model::createMessageFlows() {
405 BPMN::Model::createMessageFlows();
406
407 // Messages can only flow between message extensions with the same name and header.
408 // If message flows are given in the model, the message flow closest to the meesage
409 // event restricts the candidate catching or throwing message events.
410 for ( auto& sendingProcess : processes ) {
411 // find all throwing message events of the sending process
412 auto throwingMessageEvents = sendingProcess->find_all(
413 [](const BPMN::Node* node) { return node->represents<BPMN::MessageThrowEvent>();}
414 );
415
416 for ( auto& receivingProcess : processes ) {
417 // only consider node pairs belonging to different processes
418 if ( sendingProcess.get() != receivingProcess.get() ) {
419 // find all catching message events of receiving process
420 auto catchingMessageEvents = receivingProcess->find_all(
421 [](const BPMN::Node* node) { return node->represents<BPMN::MessageCatchEvent>();}
422 );
423
424 for ( auto throwingMessageEvent : throwingMessageEvents ) {
425 for ( auto catchingMessageEvent : catchingMessageEvents ) {
426 createMessageCandidates(sendingProcess.get(), throwingMessageEvent->as<BPMN::FlowNode>(), receivingProcess.get(), catchingMessageEvent->as<BPMN::FlowNode>());
427 }
428 }
429 }

431 }

436std::vector<BPMN::MessageFlow*>& Model::determineMessageFlows(BPMN::FlowNode* messageEvent, auto getMessageFlows) {
437 auto& relevantFlows = getMessageFlows(messageEvent);
438 if ( relevantFlows.empty() ) {
439 BPMN::ChildNode* node = messageEvent;
440 BPMN::Scope* scope = nullptr;
441 do {
442 // get next scope that may have message flows
443 scope = node->parent;
444 while ( auto eventSubProcess = scope->represents<BPMN::EventSubProcess>() ) {
445 // skip event-subprocesses
446 node = eventSubProcess;
447 scope = eventSubProcess->parent;
448 }
449
450 relevantFlows = getMessageFlows(scope);
451 node = scope->represents<BPMN::SubProcess>();
452
453 } while ( relevantFlows.empty() && node);
454 }
455 return relevantFlows;

457bool Model::messageMayBeCaught( [[maybe_unused]] BPMN::Process* sendingProcess, BPMN::FlowNode* throwingMessageEvent, BPMN::Process* receivingProcess, BPMN::FlowNode* catchingMessageEvent ) {
458 // determine relevant message flows for throwing events
459 auto& outgoingMessageFlows = determineMessageFlows(
460 throwingMessageEvent,
461 [](BPMN::Node* node) -> std::vector<BPMN::MessageFlow*>& {
462 return node->sending;
463 }
464 );
465
466 if ( outgoingMessageFlows.empty() ) {
467 // no message flow is provided that imposes a restriction
468 return true;
469 }
470
471 // determine whether catching message event is in message flow target
472 bool found = false;
473 for ( auto messageFlow : outgoingMessageFlows ) {
474 auto& [process,flowNode] = messageFlow->target;
475 if ( process == receivingProcess ) {
476 if ( flowNode == catchingMessageEvent ) {
477 return true;
478 }
479 else {
480 if ( flowNode ) {
481 found = flowNode->find_all(
482 [catchingMessageEvent](const BPMN::Node* node) { return node == catchingMessageEvent;}
483 ).size();
484 }
485 else {
486 found = process->find_all(
487 [catchingMessageEvent](const BPMN::Node* node) { return node == catchingMessageEvent;}
488 ).size();
489 }
490 }
491 if ( found ) {
492 return true;
493 }
494 }
495 }
496 return false;
497}

499bool Model::messageMayBeThrown( BPMN::Process* sendingProcess, BPMN::FlowNode* throwingMessageEvent, [[maybe_unused]] BPMN::Process* receivingProcess, BPMN::FlowNode* catchingMessageEvent ) {
500 // determine relevant message flows for catching event
501 auto incomingMessageFlows = determineMessageFlows(
502 catchingMessageEvent,
503 [](BPMN::Node* node) -> std::vector<BPMN::MessageFlow*>& {
504 return node->receiving;
505 }
506 );
507
508 if ( incomingMessageFlows.empty() ) {
509 // no message flow is provided that imposes a restriction
510 return true;
511 }
512
513 // determine whether throwing message event is in message flow source
514 bool found = false;
515 for ( auto messageFlow : incomingMessageFlows ) {
516 auto& [process,flowNode] = messageFlow->source;
517 if ( process == sendingProcess ) {
518 if ( flowNode == throwingMessageEvent ) {
519 return true;
520 }
521 else {
522 if ( flowNode ) {
523 found = flowNode->find_all(
524 [throwingMessageEvent](const BPMN::Node* node) { return node == throwingMessageEvent;}
525 ).size();
526 }
527 else {
528 found = process->find_all(
529 [throwingMessageEvent](const BPMN::Node* node) { return node == throwingMessageEvent;}
530 ).size();
531 }
532 if ( found ) {
533 return true;
534 }
535 }
536 }
537 }
538 return false;

541void Model::createMessageCandidates( BPMN::Process* sendingProcess, BPMN::FlowNode* throwingMessageEvent, BPMN::Process* receivingProcess, BPMN::FlowNode* catchingMessageEvent ) {
542 auto senderExtension = throwingMessageEvent->extensionElements->represents<BPMNOS::Model::ExtensionElements>();
543 for ( auto& outgoingMessageDefinition : senderExtension->messageDefinitions ) {
544 auto recipientExtension = catchingMessageEvent->extensionElements->represents<BPMNOS::Model::ExtensionElements>();
545 for ( auto& incomingMessageDefinition : recipientExtension->messageDefinitions) {
546
547 assert( outgoingMessageDefinition.get() );
548 assert( incomingMessageDefinition.get() );
549
550 if ( outgoingMessageDefinition->name != incomingMessageDefinition->name ) {
551 continue;
552 }
553 if ( outgoingMessageDefinition->header != incomingMessageDefinition->header ) {
554 continue;
555 }
556
557 if ( messageMayBeCaught(sendingProcess, throwingMessageEvent, receivingProcess, catchingMessageEvent) &&
558 messageMayBeThrown(sendingProcess, throwingMessageEvent, receivingProcess, catchingMessageEvent)
559 ) {
560 // add message events to collection of candidates of each other
561 if( find(
562 senderExtension->messageCandidates.begin(),
563 senderExtension->messageCandidates.end(),
564 catchingMessageEvent->as<BPMN::FlowNode>()
565 ) == senderExtension->messageCandidates.end()
566 ) {
567//std::cerr << throwingMessageEvent->id << " -> " << catchingMessageEvent->id << std::endl;
568 senderExtension->messageCandidates.push_back(catchingMessageEvent->as<BPMN::FlowNode>());
569 }
570
571 if( find(
572 recipientExtension->messageCandidates.begin(),
573 recipientExtension->messageCandidates.end(),
574 throwingMessageEvent->as<BPMN::FlowNode>()
575 ) == recipientExtension->messageCandidates.end()
576 ) {
577//std::cerr << throwingMessageEvent->id << " -> " << catchingMessageEvent->id << std::endl;
578 recipientExtension->messageCandidates.push_back(throwingMessageEvent->as<BPMN::FlowNode>());
579 }
580 }
581 }
582 }
